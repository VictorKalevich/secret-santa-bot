
  // ---------- служебный просмотр владельца (010) ----------
  // Вход — вводом слова ОТКРЫТЬ (как СБРОС и УДАЛИТЬ), чтобы владелец, который сам играет, не увидел пары случайно;
  // введённое слово уходит в каждом запросе служебного просмотра — его проверяет и сервер (025)
  function openVaultGate(evId, s, users, meMember) {
    const word = 'ОТКРЫТЬ';
    const inp = h('input', { class: 'ds-input', type: 'text', placeholder: `Введи слово ${word}`, autocomplete: 'off' });
    const go = ui.btn('🔐 Войти', '', 'ds-btn--block');
    go.disabled = true;
    inp.addEventListener('input', () => { go.disabled = inp.value.trim().toUpperCase() !== word; });
    go.addEventListener('click', () => openVault(evId, s, users, meMember, inp.value.trim()));
    openSheet('🔐 Служебный просмотр', [
      meMember ? h('div', { class: 'ds-notice ds-notice--error' }, h('b', {}, 'Ты участник этого события. '),
        'В полной таблице пар ты увидишь своего Санту. Лучше смотри по одному участнику и не открывай себя.') : null,
      h('div', { class: 'ds-notice' }, 'Здесь видно, кто кому дарит, письма, адреса и переписку. Каждое открытие участника или всех пар ' +
        'записывается в журнал события — его видят владелец и администраторы. Статусы участников просмотр не меняет.'),
      clearable(inp, 'Слово подтверждения'), go].filter(Boolean));
  }

  function openVault(evId, s, users, meMember, word) {
    const vapi = (action, body) => api(action, { ...(body || {}), event: evId, confirm: word });
    const sel = h('select', { class: 'ds-input' }, h('option', { value: '' }, 'Участник…'),
      ...users.map((u) => h('option', { value: u.id }, (u.name || 'без имени') + (u.id === state.me.id ? ' (ты)' : ''))));
    const showBtn = ui.btn('👁 Открыть участника', '', 'ds-btn--block');
    const personBox = h('div', { class: 'ds-stack' });
    showBtn.addEventListener('click', async () => {
      const u = users.find((x) => String(x.id) === sel.value);
      if (!u) return toast('Выбери участника');
      if (u.id === state.me.id && !(await confirmBox('Это ты: увидишь своего Санту. Открыть?'))) return;
      busy(showBtn, async () => personBox.replaceChildren(...vaultPerson(await vapi('admin.vault.person', { id: u.id }), s)));
    });
    const pairsBtn = ui.btn('📋 Показать все пары', 'secondary', 'ds-btn--block');
    const pairsBox = h('div', { class: 'ds-stack' });
    pairsBtn.addEventListener('click', async () => {
      const warn = meMember ? ' Ты участник — увидишь и своего Санту.' : '';
      if (!(await confirmBox(`Показать все пары события «${s.title}»?${warn} Открытие запишется в журнал.`))) return;
      busy(pairsBtn, async () => {
        const r = await vapi('admin.vault.pairs');
        const list = h('div', { class: 'ds-card ds-card--flush' });
        if (!r.pairs.length) list.append(h('div', { class: 'ds-list-item ds-text-muted' }, 'Пар нет: жеребьёвки ещё не было.'));
        for (const p of r.pairs) {
          list.append(h('div', { class: 'ds-list-item' }, h('div', { class: 'ds-grow' }, h('b', {}, p.santa.name), ' → ', h('b', {}, p.ward.name))));
        }
        pairsBox.replaceChildren(ui.section(`Все пары · ${r.pairs.length}`), list);
      });
    });
    openSheet('🔐 Служебный просмотр', [
      ui.hint(s.draw_done
        ? 'По одному участнику — безопаснее: не увидишь лишнего. Каждое открытие записывается в журнал.'
        : 'Жеребьёвки ещё не было — пар нет. Письма, адреса и переписку можно смотреть по участникам.'),
      sel, showBtn, personBox,
      s.draw_done ? pairsBtn : null, pairsBox,
    ].filter(Boolean));
  }

  // Карточка участника в служебном просмотре: подопечный, Санта, письмо, адрес, переписка с датами
  function vaultPerson(r, s) {
    const rows = [
      ['Дарит', r.ward ? r.ward.name : (s.draw_done ? 'не в жеребьёвке' : 'жеребьёвки ещё не было')],
      ['Санта', r.santa ? r.santa.name : (s.draw_done ? 'нет' : 'жеребьёвки ещё не было')],
      ...('gift_sent' in r ? [['Отправка', r.gift_sent ? 'Подарок отправлен' : 'Не отмечена']] : []),
    ];
    // mineSide — сторона самого участника в этой переписке
    const chat = (title, msgs, mineSide, otherLabel) => [
      ui.section(title),
      h('div', { class: 'ds-card ds-chat' }, msgs.length
        ? msgs.map((m) => h('div', { class: 'ds-chat__msg ' + (m.from === mineSide ? 'ds-chat__msg--mine' : 'ds-chat__msg--theirs') },
          m.text, h('div', { class: 'ds-chat__meta' }, `${m.from === mineSide ? r.person.name : otherLabel} · ${fmtTime(m.created_at)}`)))
        : h('div', { class: 'ds-chat__empty' }, 'Сообщений нет.')),
    ];
    return [
      ui.section(r.person.name),
      h('div', { class: 'ds-card' }, h('dl', { class: 'ds-info-grid' }, rows.flatMap(([k, v]) => [h('dt', {}, k), h('dd', {}, v)]))),
      ui.section('Письмо Санте'),
      r.wishlist.trim() ? h('div', { class: 'ds-card ds-letter' }, r.wishlist) : h('div', { class: 'ds-card ds-text-muted' }, 'Письмо пустое.'),
      ...('delivery' in r ? [ui.section('Адрес доставки'),
        r.delivery.trim() ? h('div', { class: 'ds-card ds-letter' }, r.delivery) : h('div', { class: 'ds-card ds-text-muted' }, 'Адрес не указан.')] : []),
      ...chat('Переписка с Сантой', r.chat_with_santa, 'ward', r.santa ? `Санта (${r.santa.name})` : 'Санта'),
      ...chat('Переписка с подопечным', r.chat_with_ward, 'santa', r.ward ? `Подопечный (${r.ward.name})` : 'Подопечный'),
    ];
  }

  // ---------- управление жеребьёвкой (048): обмен подопечных у двух участников, только владелец ----------
  // Владелец может играть сам: в шторке только имена, сервер отвечает кодами и текстами правил без имён. Обмен
  // симметричен — порядок отметок не важен; повтор запроса после успеха сервер отклонит по версии пар
  function openSwapSheet(p) {
    const { me, xapi } = p;
    const inDraw = p.players.filter((u) => u.in_draw).sort((x, y) => (x.name || '').localeCompare(y.name || '', 'ru'));
    const picked = [];
    const result = h('div');
    const go = ui.btn('🔀 Поменять подопечных', '', 'ds-btn--block');
    // Метка «не открыл подопечного» (049) — своим узлом: после обмена её ставят обоим
    const marks = new Map(inDraw.map((u) => [u, h('div')]));
    const mark = (u) => marks.get(u).replaceChildren(...(u.ward_seen === false ? [h('div', { class: 'ds-chips' }, unseenChip())] : []));
    const boxes = inDraw.map((u) => {
      const box = h('input', { type: 'checkbox' });
      box.addEventListener('change', () => {
        if (box.checked) picked.push(u);
        else picked.splice(picked.indexOf(u), 1);
        result.replaceChildren();
        syncBoxes();
      });
      return [u, box];
    });
    // Отмечены двое — остальные недоступны; кнопка — только при двух
    const syncBoxes = () => {
      for (const [u, box] of boxes) box.disabled = picked.length === 2 && !picked.includes(u);
      go.disabled = picked.length !== 2;
    };
    // Плашка проверки — под списком: в шторке она может оказаться ниже края экрана
    const show = (node) => {
      result.replaceChildren(node);
      node.scrollIntoView({ block: 'nearest', behavior: 'smooth' });
    };
    const details = (rules) => h('details', { class: 'ds-details' }, h('summary', { class: 'ds-details__summary' }, 'Подробнее'),
      h('ul', { class: 'ds-bullets' }, rules.map((r) => h('li', {}, r.text))));
    // Новые пары — с чистого листа (048): новых подопечных оба ещё не открывали
    const success = async (u, t) => {
      haptic('success');
      toast('🔀 Подопечные поменялись');
      for (const x of [u, t]) {
        x.ward_seen = false;
        mark(x);
      }
      picked.length = 0;
      for (const [, box] of boxes) box.checked = false;
      result.replaceChildren();
      await p.again({ quiet: true });
    };
    // Нарушения — абстрактный текст, правила в «Подробнее», обмен — только с галочкой; отказ сервера убирает плашку
    const warning = (u, t, c) => {
      const agree = h('input', { type: 'checkbox' });
      const btn = ui.btn('🔀 Поменять подопечных', 'danger', 'ds-btn--block');
      btn.disabled = true;
      agree.addEventListener('change', () => { btn.disabled = !agree.checked; });
      btn.addEventListener('click', async () => {
        let ok = false;
        await busy([btn, go], async () => {
          await xapi('admin.draw.swap', { a: u.id, b: t.id, version: c.version, agree: c.rules.map((r) => r.rule) });
          ok = true;
          await success(u, t);
        });
        if (!ok) result.replaceChildren();
        syncBoxes();
      });
      return h('div', { class: 'ds-notice ds-notice--error ds-stack ds-stack--sm' },
        h('b', {}, c.rules.length > 1 ? '⚠️ Обмен нарушает несколько принципов жеребьёвки.' : '⚠️ Обмен нарушает один из принципов жеребьёвки.'),
        details(c.rules),
        h('div', {}, 'Если поменять, тем, кто уже открыл подопечного, бот напишет, что он сменился. Отметки подарка, маски и переписка этих двух пар начнутся заново.'),
        h('label', { class: 'ds-check' }, agree, 'Согласен — всё равно поменять'),
        btn);
    };
    go.addEventListener('click', async () => {
      await busy(go, async () => {
        const [u, t] = picked;
        const c = await xapi('admin.draw.swap.check', { a: u.id, b: t.id });
        if (!c.possible) {
          haptic('error');
          show(h('div', { class: 'ds-notice ds-notice--error ds-stack ds-stack--sm' },
            h('b', {}, 'Этих двоих поменять нельзя — выбери другую пару.'), details(c.rules)));
          return;
        }
        if (c.rules.length) return show(warning(u, t, c));
        if (!(await confirmBox(`Поменять подопечных у «${u.name}» и «${t.name}»? Тем, кто уже открыл подопечного, бот напишет, что он сменился. Отметки подарка, маски и переписка этих двух пар начнутся заново.`))) return;
        await xapi('admin.draw.swap', { a: u.id, b: t.id, version: c.version });
        await success(u, t);
      });
      syncBoxes(); // busy вернул кнопке доступность — после обмена отметок нет
    });
    for (const u of inDraw) mark(u);
    syncBoxes();
    const meInDraw = inDraw.some((u) => u.id === me.id);
    openSheet('🔀 Управление жеребьёвкой', [
      meInDraw ? h('div', { class: 'ds-notice ds-notice--error' }, h('b', {}, 'Ты участник этого события. '),
        'Если выберешь себя, предупреждение о правилах или отказ могут подсказать, кто твой Санта.') : null,
      ui.hint('Выбери двух участников — они обменяются подопечными. Кто кому дарит, здесь не видно.'),
      h('div', { class: 'ds-card ds-card--flush' }, boxes.map(([u, box]) =>
        h('label', { class: 'ds-list-item ds-check' }, box,
          h('div', { class: 'ds-grow' }, h('div', {}, u.name || 'без имени', u.id === me.id ? ' (ты)' : ''), marks.get(u))))),
      go, result,
    ].filter(Boolean));
  }

  // Журнал события, общий журнал компании открытого экрана или журнал бота (bot, 051: только владельцу бота)
  function openAudit(eventId, bot = false) {
    const box = h('div', { class: 'ds-loader' }, h('div', { class: 'ds-spinner' }), 'Загрузка…');
    openSheet(bot ? 'Журнал бота' : eventId ? 'Журнал события' : 'Общий журнал', [box]);
    (bot ? api('bot.audit') : eventId ? api('admin.audit', { event: eventId }) : capi('admin.audit')).then((d) => {
      const list = h('div', { class: 'ds-card ds-card--flush' });
      if (!d.entries.length) list.append(h('div', { class: 'ds-list-item ds-text-muted' }, 'Записей пока нет.'));
      for (const e of d.entries) {
        list.append(h('div', { class: 'ds-list-item' }, h('div', { class: 'ds-grow' },

      }
      box.replaceWith(ui.hint('Последние 100 записей, новые сверху.', 'ds-hint--inset'), list);
    }).catch((e) => box.replaceWith(ui.hint('⚠️ ' + e.message, 'ds-hint--inset')));
