
  // ---------- служебный просмотр владельца (010) ----------
  // Вход — вводом слова ОТКРЫТЬ (как СБРОС и УДАЛИТЬ), чтобы владелец, который сам играет, не увидел пары случайно;
  // введённое слово уходит в каждом запросе служебного просмотра — его проверяет и сервер (025)
  function openVaultGate(evId, s, users, meMember) {
    const word = t('app.core.word.open');
    const inp = h('input', { class: 'ds-input', type: 'text', placeholder: t('app.sheet.typed.placeholder', { word }), autocomplete: 'off' });
    const go = ui.btn(t('app.vault.gate.enter'), '', 'ds-btn--block');
    go.disabled = true;
    inp.addEventListener('input', () => { go.disabled = inp.value.trim().toUpperCase() !== word; });
    go.addEventListener('click', () => openVault(evId, s, users, meMember, inp.value.trim()));
    openSheet(t('app.vault.title'), [
      meMember ? h('div', { class: 'ds-notice ds-notice--error' }, h('b', {}, t('app.vault.youPlay')), t('app.vault.gate.youPlay')) : null,
      h('div', { class: 'ds-notice' }, t('app.vault.gate.notice')),
      clearable(inp, t('app.sheet.typed.field')), go].filter(Boolean));
  }

  // Имя в списке участников служебного просмотра и обмена: без имени — «без имени», своё — с «(ты)»
  function vaultName(u, meId) {
    const name = u.name || t('app.vault.noName');
    return u.id === meId ? t('app.vault.nameYou', { name }) : name;
  }

  function openVault(evId, s, users, meMember, word) {
    const vapi = (action, body) => api(action, { ...(body || {}), event: evId, confirm: word });
    const sel = h('select', { class: 'ds-input' }, h('option', { value: '' }, t('app.vault.pick')),
      ...users.map((u) => h('option', { value: u.id }, vaultName(u, state.me.id))));
    const showBtn = ui.btn(t('app.vault.open'), '', 'ds-btn--block');
    const personBox = h('div', { class: 'ds-stack' });
    showBtn.addEventListener('click', async () => {
      const u = users.find((x) => String(x.id) === sel.value);
      if (!u) return toast(t('app.vault.pickNeeded'));
      if (u.id === state.me.id && !(await confirmBox(t('app.vault.openSelf')))) return;
      busy(showBtn, async () => personBox.replaceChildren(...vaultPerson(await vapi('admin.vault.person', { id: u.id }), s)));
    });
    const pairsBtn = ui.btn(t('app.vault.pairs.show'), 'secondary', 'ds-btn--block');
    const pairsBox = h('div', { class: 'ds-stack' });
    pairsBtn.addEventListener('click', async () => {
      if (!(await confirmBox(meMember ? t('app.vault.pairs.confirmPlayer', { title: s.title }) : t('app.vault.pairs.confirm', { title: s.title })))) return;
      busy(pairsBtn, async () => {
        const r = await vapi('admin.vault.pairs');
        const list = h('div', { class: 'ds-card ds-card--flush' });
        if (!r.pairs.length) list.append(h('div', { class: 'ds-list-item ds-text-muted' }, t('app.vault.pairs.none')));
        for (const p of r.pairs) {
          list.append(h('div', { class: 'ds-list-item' }, h('div', { class: 'ds-grow' }, h('b', {}, p.santa.name), ' → ', h('b', {}, p.ward.name))));
        }
        pairsBox.replaceChildren(ui.section(t('app.vault.pairs.title', { count: r.pairs.length })), list);
      });
    });
    openSheet(t('app.vault.title'), [
      ui.hint(s.draw_done ? t('app.vault.hintDrawn') : t('app.vault.hintNoDraw')),
      sel, showBtn, personBox,
      s.draw_done ? pairsBtn : null, pairsBox,
    ].filter(Boolean));
  }

  // Карточка участника в служебном просмотре: подопечный, Санта, письмо, адрес, переписка с датами
  function vaultPerson(r, s) {
    const rows = [
      [t('app.vault.person.gives'), r.ward ? r.ward.name : (s.draw_done ? t('app.vault.person.notInDraw') : t('app.vault.person.noDraw'))],
      [t('app.vault.person.santa'), r.santa ? r.santa.name : (s.draw_done ? t('app.vault.person.noSanta') : t('app.vault.person.noDraw'))],
      ...('gift_sent' in r ? [[t('app.vault.person.shipping'), r.gift_sent ? t('app.vault.person.giftSent') : t('app.vault.person.notMarked')]] : []),
    ];
    // mineSide — сторона самого участника в этой переписке
    const chat = (title, msgs, mineSide, otherLabel) => [
      ui.section(title),
      h('div', { class: 'ds-card ds-chat' }, msgs.length
        ? msgs.map((m) => h('div', { class: 'ds-chat__msg ' + (m.from === mineSide ? 'ds-chat__msg--mine' : 'ds-chat__msg--theirs') },
          m.text, h('div', { class: 'ds-chat__meta' }, `${m.from === mineSide ? r.person.name : otherLabel} · ${fmtTime(m.created_at)}`)))
        : h('div', { class: 'ds-chat__empty' }, t('app.sheet.noMessages'))),
    ];
    return [
      ui.section(r.person.name),
      h('div', { class: 'ds-card' }, h('dl', { class: 'ds-info-grid' }, rows.flatMap(([k, v]) => [h('dt', {}, k), h('dd', {}, v)]))),
      ui.section(t('app.vault.person.letter')),
      r.wishlist.trim() ? h('div', { class: 'ds-card ds-letter' }, r.wishlist) : h('div', { class: 'ds-card ds-text-muted' }, t('app.vault.person.letterEmpty')),
      ...('delivery' in r ? [ui.section(t('app.vault.person.address')),
        r.delivery.trim() ? h('div', { class: 'ds-card ds-letter' }, r.delivery) : h('div', { class: 'ds-card ds-text-muted' }, t('app.vault.person.addressEmpty'))] : []),
      ...chat(t('app.vault.person.chatSanta'), r.chat_with_santa, 'ward', r.santa ? t('app.vault.person.santaNamed', { name: r.santa.name }) : t('app.vault.person.santa')),
      ...chat(t('app.sheet.chat.withWard'), r.chat_with_ward, 'santa', r.ward ? t('app.vault.person.wardNamed', { name: r.ward.name }) : t('app.vault.person.ward')),
    ];
  }

  // ---------- управление жеребьёвкой (048): обмен подопечных у двух участников, только владелец ----------
  // Владелец может играть сам: в шторке только имена, сервер отвечает кодами и текстами правил без имён. Обмен
  // симметричен — порядок отметок не важен; повтор запроса после успеха сервер отклонит по версии пар
  function openSwapSheet(p) {
    const { me, xapi } = p;
    const inDraw = p.players.filter((u) => u.in_draw).sort((x, y) => (x.name || '').localeCompare(y.name || '', 'ru'));
    const picked = [];
    const result = h('div');
    const go = ui.btn(t('app.vault.swap.go'), '', 'ds-btn--block');
    // Метка «не открыл подопечного» (049) — своим узлом: после обмена её ставят обоим
    const marks = new Map(inDraw.map((u) => [u, h('div')]));
    const mark = (u) => marks.get(u).replaceChildren(...(u.ward_seen === false ? [h('div', { class: 'ds-chips' }, unseenChip())] : []));
    const boxes = inDraw.map((u) => {
      const box = h('input', { type: 'checkbox' });
      box.addEventListener('change', () => {
        if (box.checked) picked.push(u);
        else picked.splice(picked.indexOf(u), 1);
        result.replaceChildren();
        syncBoxes();
      });
      return [u, box];
    });
    // Отмечены двое — остальные недоступны; кнопка — только при двух
    const syncBoxes = () => {
      for (const [u, box] of boxes) box.disabled = picked.length === 2 && !picked.includes(u);
      go.disabled = picked.length !== 2;
    };
    // Плашка проверки — под списком: в шторке она может оказаться ниже края экрана
    const show = (node) => {
      result.replaceChildren(node);
      node.scrollIntoView({ block: 'nearest', behavior: 'smooth' });
    };
    const details = (rules) => h('details', { class: 'ds-details' }, h('summary', { class: 'ds-details__summary' }, t('app.vault.swap.more')),
      h('ul', { class: 'ds-bullets' }, rules.map((r) => h('li', {}, r.text))));
    // Новые пары — с чистого листа (048): новых подопечных оба ещё не открывали
    const success = async (u, v) => {
      haptic('success');
      toast(t('app.vault.swap.done'));
      for (const x of [u, v]) {
        x.ward_seen = false;
        mark(x);
      }
      picked.length = 0;
      for (const [, box] of boxes) box.checked = false;
      result.replaceChildren();
      await p.again({ quiet: true });
    };
    // Нарушения — абстрактный текст, правила в «Подробнее», обмен — только с галочкой; отказ сервера убирает плашку
    const warning = (u, v, c) => {
      const agree = h('input', { type: 'checkbox' });
      const btn = ui.btn(t('app.vault.swap.go'), 'danger', 'ds-btn--block');
      btn.disabled = true;
      agree.addEventListener('change', () => { btn.disabled = !agree.checked; });
      btn.addEventListener('click', async () => {
        let ok = false;
        await busy([btn, go], async () => {
          await xapi('admin.draw.swap', { a: u.id, b: v.id, version: c.version, agree: c.rules.map((r) => r.rule) });
          ok = true;
          await success(u, v);
        });
        if (!ok) result.replaceChildren();
        syncBoxes();
      });
      return h('div', { class: 'ds-notice ds-notice--error ds-stack ds-stack--sm' },
        h('b', {}, c.rules.length > 1 ? t('app.vault.swap.breaksMany') : t('app.vault.swap.breaksOne')),
        details(c.rules),
        h('div', {}, t('app.vault.swap.warning')),
        h('label', { class: 'ds-check' }, agree, t('app.vault.swap.agree')),
        btn);
    };
    go.addEventListener('click', async () => {
      await busy(go, async () => {
        const [u, v] = picked;
        const c = await xapi('admin.draw.swap.check', { a: u.id, b: v.id });
        if (!c.possible) {
          haptic('error');
          show(h('div', { class: 'ds-notice ds-notice--error ds-stack ds-stack--sm' },
            h('b', {}, t('app.vault.swap.impossible')), details(c.rules)));
          return;
        }
        if (c.rules.length) return show(warning(u, v, c));
        if (!(await confirmBox(t('app.vault.swap.confirm', { a: u.name, b: v.name })))) return;
        await xapi('admin.draw.swap', { a: u.id, b: v.id, version: c.version });
        await success(u, v);
      });
      syncBoxes(); // busy вернул кнопке доступность — после обмена отметок нет
    });
    for (const u of inDraw) mark(u);
    syncBoxes();
    const meInDraw = inDraw.some((u) => u.id === me.id);
    openSheet(t('app.vault.swap.title'), [
      meInDraw ? h('div', { class: 'ds-notice ds-notice--error' }, h('b', {}, t('app.vault.youPlay')), t('app.vault.swap.youPlay')) : null,
      ui.hint(t('app.vault.swap.hint')),
      h('div', { class: 'ds-card ds-card--flush' }, boxes.map(([u, box]) =>
        h('label', { class: 'ds-list-item ds-check' }, box,
          h('div', { class: 'ds-grow' }, h('div', {}, vaultName(u, me.id)), marks.get(u))))),
      go, result,
    ].filter(Boolean));
  }

  // ---------- журнал действий (007) ----------
  // Подпись записи — app.vault.audit.<код> (коды — docs/api.md, «Коды журнала»); код без подписи — как есть
  const auditLabel = (action) => (own(TEXTS, 'app.vault.audit.' + action) ? t('app.vault.audit.' + action) : action);
  // Журнал события, общий журнал компании открытого экрана или журнал бота (bot, 051: только владельцу бота)
  function openAudit(eventId, bot = false) {
    const box = h('div', { class: 'ds-loader' }, h('div', { class: 'ds-spinner' }), t('app.boot.loading'));
    openSheet(bot ? t('app.vault.log.bot') : eventId ? t('app.vault.log.event') : t('app.vault.log.common'), [box]);
    (bot ? api('bot.audit') : eventId ? api('admin.audit', { event: eventId }) : capi('admin.audit')).then((d) => {
      const list = h('div', { class: 'ds-card ds-card--flush' });
      if (!d.entries.length) list.append(h('div', { class: 'ds-list-item ds-text-muted' }, t('app.vault.log.empty')));
      for (const e of d.entries) {
        list.append(h('div', { class: 'ds-list-item' }, h('div', { class: 'ds-grow' },
          h('div', { class: 'ds-list-item__title' }, auditLabel(e.action)),
          e.details ? h('div', { class: 'ds-text-sm' }, e.details) : null,
          ui.hint(`${e.actor_name} · ${fmtTime(e.created_at)}`))));
      }
      box.replaceWith(ui.hint(t('app.vault.log.hint'), 'ds-hint--inset'), list);
    }).catch((e) => box.replaceWith(ui.hint('⚠️ ' + e.message, 'ds-hint--inset')));
  }
