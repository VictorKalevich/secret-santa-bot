
  // ---------- компании (051) ----------
  // Экран «Компании» — корневой у человека с несколькими компаниями: строки компаний (вход, своё название), у владельца
  // бота — шторка «Компания» (кто ведёт, ссылка владельца, удаление) и инструменты: создать компанию, журнал бота, передать
  // права владельца бота. Контракт — specs/051-companies/contracts/companies.md, раздел 5. 052 (открытый бот,
  // specs/052-open-bot/contracts/open-bot.md, раздел 4): создать компанию может каждый — не больше двух своих; у своей
  // компании — общее название и удаление пустой
  function renderCompanies() {
    const me = state.me;
    heroBack.hidden = true;
    heroTitle.textContent = t('app.boot.title');
    const greet = () => { heroSub.textContent = t('app.home.greet', { name: me.name }); };
    greet();
    document.getElementById('tabs').hidden = true;
    const nodes = [ui.section(t('app.companies.title'))];
    if (state.companies.length) nodes.push(h('div', { class: 'ds-card ds-card--flush' }, state.companies.map((c) => companyRow(c, true))));
    else nodes.push(ui.hint(state.botOwner ? t('app.companies.emptyBotOwner') : t('app.companies.empty'), 'ds-hint--inset'));
    if (state.botOwner) nodes.push(...companiesTools());
    else if (state.create) nodes.push(ownCreate());
    nodes.push(...personalSections(state.nav, greet));
    main.replaceChildren(...nodes);
  }

  // Корневой экран (052): «Компании», если их несколько или их нет, а список доступен (withList); иначе главный экран
  // единственной компании (или без компании)
  const listFirst = () => state.companies.length >= 2 || (!state.companies.length && withList());
  function goRoot() {
    return listFirst() ? goCompanies() : openCompany(state.companies.length ? state.companies[0].id : null);
  }

  // Строка компании: нажатие — вход (link), «✏️» и «⚙️» компанию не открывают
  function companyRow(c, link) {
    const own = (fn) => (e) => { e.stopPropagation(); fn(); };
    const about = t(c.events && !c.active ? 'app.companies.aboutIdle' : 'app.companies.about', { owner: c.owner_name || '—', events: c.events });
    return h('div', { class: 'ds-list-item' + (link ? ' ds-list-item--link' : ''), role: link ? 'button' : null, onclick: link ? () => openCompany(c.id) : null },
      h('div', { class: 'ds-grow' },
        h('div', { class: 'ds-list-item__title' }, c.title),
        h('div', { class: 'ds-chips' }, ui.chip(t('app.core.role.' + c.role))),
        ui.hint(about)),
      link && c.unread ? h('span', { class: 'ds-badge' }, String(c.unread)) : null,
      h('div', { class: 'ds-row ds-row--tight' },
        h('button', { class: 'ds-icon-btn', title: c.rename ? t('app.companies.roomName') : t('app.companies.ownName'), onclick: own(() => openLabelSheet(c)) }, '✏️'),
        state.botOwner || c.deletable
          ? h('button', { class: 'ds-icon-btn', title: t('app.companies.settings'), onclick: own(() => (state.botOwner ? openCompanySheet(c) : openOwnCompanySheet(c))) }, '⚙️')
          : null),
      link ? h('span', { class: 'ds-list-item__chevron' }, '›') : null);
  }

  // «Название для себя»: владельцу бота — название компании (видит только он; общее — 052 — видят её люди), владельцу
  // компании с общим названием (c.rename, 052) — оно же, остальным — своё (пустое — по умолчанию)
  function openLabelSheet(c) {
    const common = state.botOwner || c.rename;
    const input = h('input', { class: 'ds-input', type: 'text', maxlength: 40, value: c.named || c.rename ? c.title : '', placeholder: c.title, autocomplete: 'off' });
    const save = ui.btn(t('app.core.save'), '', 'ds-btn--block');
    save.addEventListener('click', () => busy(save, async () => {
      if (common && !input.value.trim()) return toast(t('app.core.needTitle'));
      const r = await api('company.label.set', { company: c.id, label: input.value });
      c.title = r.title;
      c.named = state.botOwner || (!c.rename && !!input.value.trim());
      closeSheet();
      toast(t('app.companies.label.saved'));
      if (state.view !== 'event') render();
    }));
    openSheet(c.rename ? t('app.companies.roomName') : t('app.companies.ownName'), [
      ui.hint(c.shared && common ? t('app.companies.label.shared')
        : state.botOwner ? t('app.companies.label.botOwner') : t('app.companies.label.own')),
      clearable(input, t('app.companies.label.field')), save]);
  }

  // Инструменты владельца бота — только на экране «Компании» (052): создать компанию, журнал бота, передать права
  function companiesTools() {
    const create = ui.btn(t('app.companies.create'), 'secondary', 'ds-btn--block');
    create.addEventListener('click', openCreateCompany);
    const log = ui.btn(t('app.companies.botLog'), 'secondary', 'ds-btn--block');
    log.addEventListener('click', () => openAudit(null, true));
    const pass = ui.btn(t('app.companies.botOwner.pass'), 'secondary', 'ds-btn--block');
    pass.addEventListener('click', () => openBotOwnerSheet(pass));
    return [ui.card(create, log, pass)];
  }

  // Своя компания человека открытого бота (052): две своих — кнопка недоступна, под ней — почему
  function ownCreate() {
    const create = ui.btn(t('app.companies.create'), 'secondary', 'ds-btn--block');
    const full = state.create.left === 0;
    create.disabled = full;
    create.addEventListener('click', openCreateCompany);
    return ui.card(create, full ? ui.hint(t('app.companies.ownLimit')) : null);
  }

  // Новая компания: владельцу бота — компания, которую ведёт он (051); человеку открытого бота — своя, он её владелец
  // (052). После создания — её главный экран
  function openCreateCompany() {
    const own = !state.botOwner;
    const name = h('input', { class: 'ds-input', type: 'text', maxlength: 40, placeholder: t('app.companies.new.placeholder'), autocomplete: 'off' });
    const ok = ui.btn(t('app.core.create'), '', 'ds-btn--block');
    ok.addEventListener('click', () => busy(ok, async () => {
      if (!name.value.trim()) return toast(t('app.core.needTitle'));
      const r = await api(own ? 'company.create' : 'bot.company.create', { name: name.value });
      closeSheet();
      toast(t('app.companies.new.done'));
      await openCompany(r.company);
    }));
    openSheet(t('app.companies.new.title'), [
      ui.hint(own ? t('app.companies.new.hintOwn') : t('app.companies.new.hintBot')),
      clearable(name, t('app.companies.roomName')), ok]);
  }

  // Шторка «Компания» (владелец бота): кто ведёт и «Вести самому», ссылка владельца компании, удаление пустой компании
  function openCompanySheet(c) {
    const items = [ui.section(t('app.companies.lead.title'))];
    if (c.bot_run) items.push(ui.card(h('div', { class: 'ds-text-strong' }, t('app.companies.lead.you')), ui.hint(t('app.companies.lead.botRun'))));
    else {
      const self = ui.btn(t('app.companies.lead.self'), 'secondary', 'ds-btn--block');
      self.addEventListener('click', async () => {
        if (!(await confirmBox(t('app.companies.lead.selfConfirm', { name: c.owner_name })))) return;
        busy(self, async () => {
          await capi('admin.owner.transfer', { company: c.id, id: state.me.id });
          closeSheet();
          toast(t('app.companies.lead.selfDone'));
          await reloadMe();
          render();
        });
      });
      items.push(ui.card(h('div', { class: 'ds-text-strong' }, c.owner_name), ui.hint(t('app.companies.lead.owner')), self));
    }
    items.push(ui.section(t('app.companies.link.title')), companyLinkCard(c));
    items.push(...companyDeletion(c, 'bot.company.delete', t('app.companies.delete.keepBot')));
    openSheet(c.title, items);
  }

  // Шторка своей компании человека открытого бота (052): только удаление пустой
  function openOwnCompanySheet(c) {
    openSheet(c.title, companyDeletion(c, 'company.delete', t('app.companies.delete.keepOwn')));
  }

  // Раздел «Удаление»: есть события — почему нельзя; иначе шторка со словом УДАЛИТЬ (action — действие API), после —
  // корневой экран
  function companyDeletion(c, action, keep) {
    if (c.events) return [ui.section(t('app.companies.delete.section')), ui.hint(t('app.companies.delete.hasEvents'), 'ds-hint--inset')];
    const del = ui.btn(t('app.companies.delete.button'), 'danger', 'ds-btn--block');
    del.addEventListener('click', () => typedSheet(t('app.companies.delete.title'), {
      lines: tl('app.companies.delete.lines'),
      keep,
      word: t('app.core.word.delete'),
      label: t('app.companies.delete.button'),
      sure: true,
      run: (word) => api(action, { company: c.id, confirm: word }),
      after: async () => { toast(t('app.companies.delete.done')); await reloadMe(); await goRoot(); },
    }));
    return [ui.section(t('app.companies.delete.section')), ui.card(del)];
  }

  // Ссылка владельца: только что созданная — с «Скопировать» и «Отправить в чат»; действующая повторно не показывается
  function companyLinkCard(c) {
    const card = ui.card();
    const draw = (link) => {
      const shown = [];
      if (link) {
        const input = h('input', { class: 'ds-input', type: 'text', value: link, readonly: true });
        const copy = ui.btn(t('app.companies.link.copy'), 'secondary', 'ds-btn--sm');
        copy.addEventListener('click', async () => {
          try { await navigator.clipboard.writeText(link); toast(t('app.core.link.copied')); }
          catch (e) { input.select(); toast(t('app.core.link.copyByHand')); }
        });
        const share = ui.btn(t('app.core.link.share'), 'secondary', 'ds-btn--sm');
        share.addEventListener('click', () => shareLink(link, t('app.companies.link.shareText')));
        shown.push(input, h('div', { class: 'ds-row' }, copy, share));
      } else if (c.link) shown.push(ui.hint(t('app.companies.link.active')));
      const make = ui.btn(c.link ? t('app.companies.link.renew') : t('app.companies.link.create'), 'secondary', 'ds-btn--sm');
      const off = c.link ? ui.btn(t('app.companies.link.revoke'), 'danger', 'ds-btn--sm') : null;
      const row = [make, off].filter(Boolean);
      make.addEventListener('click', () => busy(row, async () => {
        const r = await api('bot.company.link', { company: c.id, on: true });
        c.link = !!r.link;
        draw(r.link);
      }));
      if (off) off.addEventListener('click', () => busy(row, async () => {
        await api('bot.company.link', { company: c.id, on: false });
        c.link = false;
        toast(t('app.companies.link.revoked'));
        draw('');
      }));
      card.replaceChildren(ui.hint(t('app.companies.link.hint')),
        ...shown, h('div', { class: 'ds-row' }, ...row));
    };
    draw('');
    return card;
  }

  // Передать права владельца бота: кому — человеку компаний, которые ведёт владелец бота (как раньше — администратору
  // или участнику; сервер проверяет то же)
  function openBotOwnerSheet(btn) {
    busy(btn, async () => {
      const lists = await Promise.all(state.companies.filter((c) => c.bot_run).map((c) => capi('admin.users', { company: c.id })));
      const byId = new Map();
      for (const d of lists) for (const u of d.users) if (u.id !== state.me.id) byId.set(u.id, u);
      const people = [...byId.values()].sort((a, b) => a.name.localeCompare(b.name, 'ru'));
      const sel = h('select', { class: 'ds-input' }, h('option', { value: '' }, t('app.companies.botOwner.choose')), ...people.map((u) => h('option', { value: u.id }, u.name)));
      const go = ui.btn(t('app.companies.botOwner.pass'), 'danger', 'ds-btn--block');
      go.addEventListener('click', async () => {
        const u = people.find((x) => String(x.id) === sel.value);
        if (!u) return toast(t('app.core.pickPerson'));
        if (!(await confirmBox(t('app.companies.botOwner.confirm', { name: u.name })))) return;
        busy(go, async () => {
          await api('bot.owner.transfer', { id: u.id });
          closeSheet();
          toast(t('app.companies.botOwner.done'));
          await reloadMe();
          await goRoot();
        });
      });
      openSheet(t('app.companies.botOwner.title'), [ui.hint(t('app.companies.botOwner.hint')), sel, go]);
    });
  }
