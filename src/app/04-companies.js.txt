
  // ---------- компании (051) ----------
  // Экран «Компании» — корневой у человека с несколькими компаниями: строки компаний (вход, своё название), у владельца
  // бота — шторка «Компания» (кто ведёт, ссылка владельца, удаление) и инструменты: создать компанию, журнал бота, передать
  // права владельца бота. Контракт — specs/051-companies/contracts/companies.md, раздел 5
  const ROLE_CHIPS = { owner: '👑 владелец', admin: '🛠 администратор', member: 'участник', guest: '👁 владелец бота' };

  function renderCompanies() {
    const me = state.me;
    heroBack.hidden = true;
    heroTitle.textContent = 'Тайный Санта';
    const greet = () => { heroSub.textContent = 'Привет, ' + me.name + '!'; };
    greet();
    document.getElementById('tabs').hidden = true;
    const nodes = [ui.section('Компании'), h('div', { class: 'ds-card ds-card--flush' }, state.companies.map((c) => companyRow(c, true)))];
    if (state.botOwner) nodes.push(...companiesTools(null));
    nodes.push(...personalSections(state.nav, greet));
    main.replaceChildren(...nodes);
  }

  // Корневой экран: «Компании», если их несколько; иначе главный экран единственной компании (или без компании)
  function goRoot() {
    return state.companies.length >= 2 ? goCompanies() : openCompany(state.companies.length ? state.companies[0].id : null);
  }

  // Строка компании: нажатие — вход (link), «✏️» и «⚙️» компанию не открывают
  function companyRow(c, link) {
    const own = (fn) => (e) => { e.stopPropagation(); fn(); };
    const about = `Владелец — ${c.owner_name || '—'} · событий: ${c.events}` + (c.events && !c.active ? ' · идущих нет' : '');
    return h('div', { class: 'ds-list-item' + (link ? ' ds-list-item--link' : ''), role: link ? 'button' : null, onclick: link ? () => openCompany(c.id) : null },
      h('div', { class: 'ds-grow' },
        h('div', { class: 'ds-list-item__title' }, c.title),
        h('div', { class: 'ds-chips' }, ui.chip(ROLE_CHIPS[c.role] || c.role)),
        ui.hint(about)),
      link && c.unread ? h('span', { class: 'ds-badge' }, String(c.unread)) : null,
      h('div', { class: 'ds-row ds-row--tight' },
        h('button', { class: 'ds-icon-btn', title: 'Название для себя', onclick: own(() => openLabelSheet(c)) }, '✏️'),
        state.botOwner ? h('button', { class: 'ds-icon-btn', title: 'Компания', onclick: own(() => openCompanySheet(c)) }, '⚙️') : null),
      link ? h('span', { class: 'ds-list-item__chevron' }, '›') : null);
  }

  // «Название для себя»: владельцу бота — название компании (видит только он), остальным — своё (пустое — по умолчанию)
  function openLabelSheet(c) {
    const input = h('input', { class: 'ds-input', type: 'text', maxlength: 40, value: c.named ? c.title : '', placeholder: c.title, autocomplete: 'off' });
    const save = ui.btn('Сохранить', '', 'ds-btn--block');
    save.addEventListener('click', () => busy(save, async () => {
      if (state.botOwner && !input.value.trim()) return toast('Напиши название');
      const r = await api('company.label.set', { company: c.id, label: input.value });
      c.title = r.title;
      c.named = state.botOwner || !!input.value.trim();
      closeSheet();
      toast('Название сохранено');
      if (state.view !== 'event') render();
    }));
    openSheet('Название для себя', [
      ui.hint(state.botOwner ? 'Название компании видишь только ты — владелец бота.' : 'Это название видишь только ты.'),
      clearable(input, 'Название'), save]);
  }

  // Инструменты владельца бота: под списком «Компаний» и на главном экране, если компания одна (single — она)
  function companiesTools(single) {
    const create = ui.btn('🏢 Создать компанию', 'secondary', 'ds-btn--block');
    create.addEventListener('click', openCreateCompany);
    const log = ui.btn('📜 Журнал бота', 'secondary', 'ds-btn--block');
    log.addEventListener('click', () => openAudit(null, true));
    const pass = ui.btn('👑 Передать права владельца бота', 'secondary', 'ds-btn--block');
    pass.addEventListener('click', () => openBotOwnerSheet(pass));
    const nodes = state.view === 'companies' ? [] : [ui.section('Компании')];
    if (single) nodes.push(h('div', { class: 'ds-card ds-card--flush' }, companyRow(single, false)));
    nodes.push(ui.card(create, log, pass));
    return nodes;
  }

  function openCreateCompany() {
    const name = h('input', { class: 'ds-input', type: 'text', maxlength: 40, placeholder: 'Например: Работа', autocomplete: 'off' });
    const ok = ui.btn('Создать', '', 'ds-btn--block');
    ok.addEventListener('click', () => busy(ok, async () => {
      if (!name.value.trim()) return toast('Напиши название');
      await api('bot.company.create', { name: name.value });
      closeSheet();
      toast('🏢 Компания создана');
      await reloadMe();
      await goRoot();
    }));
    openSheet('Новая компания', [
      ui.hint('У компании свои люди, события, семьи и администраторы. Её можно вести самому или отдать владельцу по ссылке.'),
      clearable(name, 'Название компании'), ok]);
  }

  // Шторка «Компания» (владелец бота): кто ведёт и «Вести самому», ссылка владельца компании, удаление пустой компании
  function openCompanySheet(c) {
    const items = [ui.section('Кто ведёт')];
    if (c.bot_run) items.push(ui.card(h('div', { class: 'ds-text-strong' }, 'Ты'), ui.hint('Своего владельца у компании нет: её ведёшь ты.')));
    else {
      const self = ui.btn('Вести самому', 'secondary', 'ds-btn--block');
      self.addEventListener('click', async () => {
        if (!(await confirmBox(`Вести компанию самому? «${c.owner_name}» останется администратором.`))) return;
        busy(self, async () => {
          await capi('admin.owner.transfer', { company: c.id, id: state.me.id });
          closeSheet();
          toast('Компанию снова ведёшь ты');
          await reloadMe();
          render();
        });
      });
      items.push(ui.card(h('div', { class: 'ds-text-strong' }, c.owner_name), ui.hint('Владелец компании: администраторы, сброс и удаление событий, служебный просмотр пар.'), self));
    }
    items.push(ui.section('🔗 Ссылка владельца компании'), companyLinkCard(c));
    items.push(ui.section('Удаление'));
    if (c.events) items.push(ui.hint('В компании есть события — сначала удали их.', 'ds-hint--inset'));
    else {
      const del = ui.btn('🗑 Удалить компанию', 'danger', 'ds-btn--block');
      del.addEventListener('click', () => typedSheet('Удалить компанию', {
        lines: ['её люди, администраторы и семьи;', 'общий журнал компании.'],
        keep: 'Событий в компании нет. Её владелец и администраторы получат сообщение.',
        word: 'УДАЛИТЬ',
        label: '🗑 Удалить компанию',
        sure: true,
        run: (word) => api('bot.company.delete', { company: c.id, confirm: word }),
        after: async () => { toast('Компания удалена'); await reloadMe(); await goRoot(); },
      }));
      items.push(ui.card(del));
    }
    openSheet(c.title, items);
  }

  // Ссылка владельца: только что созданная — с «Скопировать» и «Отправить в чат»; действующая повторно не показывается
  function companyLinkCard(c) {
    const card = ui.card();
    const draw = (link) => {
      const shown = [];
      if (link) {
        const input = h('input', { class: 'ds-input', type: 'text', value: link, readonly: true });
        const copy = ui.btn('📋 Скопировать', 'secondary', 'ds-btn--sm');
        copy.addEventListener('click', async () => {
          try { await navigator.clipboard.writeText(link); toast('Ссылка скопирована'); }
          catch (e) { input.select(); toast('Выдели и скопируй ссылку'); }
        });
        const share = ui.btn('📤 Отправить в чат', 'secondary', 'ds-btn--sm');
        share.addEventListener('click', () => shareLink(link, '👑 Ссылка владельца в «Тайном Санте»: открой её, чтобы вести свою компанию.'));
        shown.push(input, h('div', { class: 'ds-row' }, copy, share));
      } else if (c.link) shown.push(ui.hint('Ссылка действует. Показать её ещё раз нельзя — можно создать новую.'));
      const make = ui.btn(c.link ? 'Создать новую' : 'Создать ссылку', 'secondary', 'ds-btn--sm');
      const off = c.link ? ui.btn('Отозвать', 'danger', 'ds-btn--sm') : null;
      const row = [make, off].filter(Boolean);
      make.addEventListener('click', () => busy(row, async () => {
        const r = await api('bot.company.link', { company: c.id, on: true });
        c.link = !!r.link;
        draw(r.link);
      }));
      if (off) off.addEventListener('click', () => busy(row, async () => {
        await api('bot.company.link', { company: c.id, on: false });
        c.link = false;
        toast('Ссылка отозвана');
        draw('');
      }));
      card.replaceChildren(ui.hint('Ссылка одноразовая: кто первым перейдёт по ней, станет владельцем компании и сможет открывать пары её событий. Отправляй её лично.'),
        ...shown, h('div', { class: 'ds-row' }, ...row));
    };
    draw('');
    return card;
  }

  // Передать права владельца бота: кому — человеку компаний, которые ведёт владелец бота (как раньше — администратору
  // или участнику; сервер проверяет то же)
  function openBotOwnerSheet(btn) {
    busy(btn, async () => {
      const lists = await Promise.all(state.companies.filter((c) => c.bot_run).map((c) => capi('admin.users', { company: c.id })));
      const byId = new Map();
      for (const d of lists) for (const u of d.users) if (u.id !== state.me.id) byId.set(u.id, u);
      const people = [...byId.values()].sort((a, b) => a.name.localeCompare(b.name, 'ru'));
      const sel = h('select', { class: 'ds-input' }, h('option', { value: '' }, 'Кому передать…'), ...people.map((u) => h('option', { value: u.id }, u.name)));
      const go = ui.btn('👑 Передать права владельца бота', 'danger', 'ds-btn--block');
      go.addEventListener('click', async () => {
        const u = people.find((x) => String(x.id) === sel.value);
        if (!u) return toast('Выбери человека');
        if (!(await confirmBox(`Передать права владельца бота «${u.name}»? Он(а) будет создавать и удалять компании и назначать их владельцев. Ты останешься администратором своих компаний.`))) return;
        busy(go, async () => {
          await api('bot.owner.transfer', { id: u.id });
          closeSheet();
          toast('👑 Права владельца бота переданы');
          await reloadMe();
          await goRoot();
        });
      });
      openSheet('Права владельца бота', [ui.hint('Владелец бота один. Передать права можно человеку из компаний, которые ведёшь ты.'), sel, go]);
    });
  }
