
  // ---------- компании (051) ----------
  // Экран «Компании» — корневой у человека с несколькими компаниями: строки компаний (вход, своё название), у владельца
  // бота — шторка «Компания» (кто ведёт, ссылка владельца, удаление) и инструменты: создать компанию, журнал бота, передать
  // права владельца бота. Контракт — specs/051-companies/contracts/companies.md, раздел 5. 052 (открытый бот,
  // specs/052-open-bot/contracts/open-bot.md, раздел 4): создать компанию может каждый — не больше двух своих; у своей
  // компании — общее название и удаление пустой
  const ROLE_CHIPS = { owner: '👑 владелец', admin: '🛠 администратор', member: 'участник', guest: '👁 владелец бота' };
  const OWN_LIMIT = 'Вести можно не больше двух своих комнат — передай или удали одну из них.';

  function renderCompanies() {
    const me = state.me;
    heroBack.hidden = true;
    heroTitle.textContent = 'Тайный Санта';
    const greet = () => { heroSub.textContent = 'Привет, ' + me.name + '!'; };
    greet();
    document.getElementById('tabs').hidden = true;
    const nodes = [ui.section('Комнаты')];
    if (state.companies.length) nodes.push(h('div', { class: 'ds-card ds-card--flush' }, state.companies.map((c) => companyRow(c, true))));
    else nodes.push(ui.hint(state.botOwner ? 'Комнат пока нет.' : 'Комнат пока нет. Создай свою — или попроси пригласительную ссылку на событие.', 'ds-hint--inset'));
    if (state.botOwner) nodes.push(...companiesTools());
    else if (state.create) nodes.push(ownCreate());
    nodes.push(...personalSections(state.nav, greet));
    main.replaceChildren(...nodes);
  }

  // Корневой экран (052): «Компании», если их несколько или их нет, а список доступен (withList); иначе главный экран
  // единственной компании (или без компании)
  const listFirst = () => state.companies.length >= 2 || (!state.companies.length && withList());
  function goRoot() {
    return listFirst() ? goCompanies() : openCompany(state.companies.length ? state.companies[0].id : null);
  }

  // Строка компании: нажатие — вход (link), «✏️» и «⚙️» компанию не открывают
  function companyRow(c, link) {
    const own = (fn) => (e) => { e.stopPropagation(); fn(); };
    const about = `Владелец — ${c.owner_name || '—'} · событий: ${c.events}` + (c.events && !c.active ? ' · идущих нет' : '');
    return h('div', { class: 'ds-list-item' + (link ? ' ds-list-item--link' : ''), role: link ? 'button' : null, onclick: link ? () => openCompany(c.id) : null },
      h('div', { class: 'ds-grow' },
        h('div', { class: 'ds-list-item__title' }, c.title),
        h('div', { class: 'ds-chips' }, ui.chip(ROLE_CHIPS[c.role] || c.role)),
        ui.hint(about)),
      link && c.unread ? h('span', { class: 'ds-badge' }, String(c.unread)) : null,
      h('div', { class: 'ds-row ds-row--tight' },
        h('button', { class: 'ds-icon-btn', title: c.rename ? 'Название комнаты' : 'Название для себя', onclick: own(() => openLabelSheet(c)) }, '✏️'),
        state.botOwner || c.deletable
          ? h('button', { class: 'ds-icon-btn', title: 'Комната', onclick: own(() => (state.botOwner ? openCompanySheet(c) : openOwnCompanySheet(c))) }, '⚙️')
          : null),
      link ? h('span', { class: 'ds-list-item__chevron' }, '›') : null);
  }

  // «Название для себя»: владельцу бота — название компании (видит только он; общее — 052 — видят её люди), владельцу
  // компании с общим названием (c.rename, 052) — оно же, остальным — своё (пустое — по умолчанию)
  function openLabelSheet(c) {
    const common = state.botOwner || c.rename;
    const input = h('input', { class: 'ds-input', type: 'text', maxlength: 40, value: c.named || c.rename ? c.title : '', placeholder: c.title, autocomplete: 'off' });
    const save = ui.btn('Сохранить', '', 'ds-btn--block');
    save.addEventListener('click', () => busy(save, async () => {
      if (common && !input.value.trim()) return toast('Напиши название');
      const r = await api('company.label.set', { company: c.id, label: input.value });
      c.title = r.title;
      c.named = state.botOwner || (!c.rename && !!input.value.trim());
      closeSheet();
      toast('Название сохранено');
      if (state.view !== 'event') render();
    }));
    openSheet(c.rename ? 'Название комнаты' : 'Название для себя', [
      ui.hint(c.shared && common ? 'Это название видят все люди комнаты.'
        : state.botOwner ? 'Название комнаты видишь только ты — владелец бота.' : 'Это название видишь только ты.'),
      clearable(input, 'Название'), save]);
  }

  // Инструменты владельца бота — только на экране «Компании» (052): создать компанию, журнал бота, передать права
  function companiesTools() {
    const create = ui.btn('🚪 Создать комнату', 'secondary', 'ds-btn--block');
    create.addEventListener('click', openCreateCompany);
    const log = ui.btn('📜 Журнал бота', 'secondary', 'ds-btn--block');
    log.addEventListener('click', () => openAudit(null, true));
    const pass = ui.btn('👑 Передать права владельца бота', 'secondary', 'ds-btn--block');
    pass.addEventListener('click', () => openBotOwnerSheet(pass));
    return [ui.card(create, log, pass)];
  }

  // Своя компания человека открытого бота (052): две своих — кнопка недоступна, под ней — почему
  function ownCreate() {
    const create = ui.btn('🚪 Создать комнату', 'secondary', 'ds-btn--block');
    const full = state.create.left === 0;
    create.disabled = full;
    create.addEventListener('click', openCreateCompany);
    return ui.card(create, full ? ui.hint(OWN_LIMIT) : null);
  }

  // Новая компания: владельцу бота — компания, которую ведёт он (051); человеку открытого бота — своя, он её владелец
  // (052). После создания — её главный экран
  function openCreateCompany() {
    const own = !state.botOwner;
    const name = h('input', { class: 'ds-input', type: 'text', maxlength: 40, placeholder: 'Например: Работа', autocomplete: 'off' });
    const ok = ui.btn('Создать', '', 'ds-btn--block');
    ok.addEventListener('click', () => busy(ok, async () => {
      if (!name.value.trim()) return toast('Напиши название');
      const r = await api(own ? 'company.create' : 'bot.company.create', { name: name.value });
      closeSheet();
      toast('🚪 Комната создана');
      await openCompany(r.company);
    }));
    openSheet('Новая комната', [
      ui.hint(own
        ? 'Ты станешь владельцем комнаты: создаёшь события, приглашаешь людей, назначаешь администраторов. Название увидят все люди комнаты.'
        : 'У комнаты свои люди, события, семьи и администраторы. Её можно вести самому или отдать владельцу по ссылке.'),
      clearable(name, 'Название комнаты'), ok]);
  }

  // Шторка «Компания» (владелец бота): кто ведёт и «Вести самому», ссылка владельца компании, удаление пустой компании
  function openCompanySheet(c) {
    const items = [ui.section('Кто ведёт')];
    if (c.bot_run) items.push(ui.card(h('div', { class: 'ds-text-strong' }, 'Ты'), ui.hint('Своего владельца у комнаты нет: её ведёшь ты.')));
    else {
      const self = ui.btn('Вести самому', 'secondary', 'ds-btn--block');
      self.addEventListener('click', async () => {
        if (!(await confirmBox(`Вести комнату самому? «${c.owner_name}» останется администратором.`))) return;
        busy(self, async () => {
          await capi('admin.owner.transfer', { company: c.id, id: state.me.id });
          closeSheet();
          toast('Комнату снова ведёшь ты');
          await reloadMe();
          render();
        });
      });
      items.push(ui.card(h('div', { class: 'ds-text-strong' }, c.owner_name), ui.hint('Владелец комнаты: администраторы, сброс и удаление событий, служебный просмотр пар.'), self));
    }
    items.push(ui.section('🔗 Ссылка владельца комнаты'), companyLinkCard(c));
    items.push(...companyDeletion(c, 'bot.company.delete', 'Событий в комнате нет. Её владелец и администраторы получат сообщение.'));
    openSheet(c.title, items);
  }

  // Шторка своей компании человека открытого бота (052): только удаление пустой
  function openOwnCompanySheet(c) {
    openSheet(c.title, companyDeletion(c, 'company.delete', 'Событий в комнате нет. Её администраторы получат сообщение.'));
  }

  // Раздел «Удаление»: есть события — почему нельзя; иначе шторка со словом УДАЛИТЬ (action — действие API), после —
  // корневой экран
  function companyDeletion(c, action, keep) {
    if (c.events) return [ui.section('Удаление'), ui.hint('В комнате есть события — сначала удали их.', 'ds-hint--inset')];
    const del = ui.btn('🗑 Удалить комнату', 'danger', 'ds-btn--block');
    del.addEventListener('click', () => typedSheet('Удалить комнату', {
      lines: ['её люди, администраторы и семьи;', 'общий журнал комнаты.'],
      keep,
      word: 'УДАЛИТЬ',
      label: '🗑 Удалить комнату',
      sure: true,
      run: (word) => api(action, { company: c.id, confirm: word }),
      after: async () => { toast('Комната удалена'); await reloadMe(); await goRoot(); },
    }));
    return [ui.section('Удаление'), ui.card(del)];
  }

  // Ссылка владельца: только что созданная — с «Скопировать» и «Отправить в чат»; действующая повторно не показывается
  function companyLinkCard(c) {
    const card = ui.card();
    const draw = (link) => {
      const shown = [];
      if (link) {
        const input = h('input', { class: 'ds-input', type: 'text', value: link, readonly: true });
        const copy = ui.btn('📋 Скопировать', 'secondary', 'ds-btn--sm');
        copy.addEventListener('click', async () => {
          try { await navigator.clipboard.writeText(link); toast('Ссылка скопирована'); }
          catch (e) { input.select(); toast('Выдели и скопируй ссылку'); }
        });
        const share = ui.btn('📤 Отправить в чат', 'secondary', 'ds-btn--sm');
        share.addEventListener('click', () => shareLink(link, '👑 Ссылка владельца в «Тайном Санте»: открой её, чтобы вести свою комнату.'));
        shown.push(input, h('div', { class: 'ds-row' }, copy, share));
      } else if (c.link) shown.push(ui.hint('Ссылка действует. Показать её ещё раз нельзя — можно создать новую.'));
      const make = ui.btn(c.link ? 'Создать новую' : 'Создать ссылку', 'secondary', 'ds-btn--sm');
      const off = c.link ? ui.btn('Отозвать', 'danger', 'ds-btn--sm') : null;
      const row = [make, off].filter(Boolean);
      make.addEventListener('click', () => busy(row, async () => {
        const r = await api('bot.company.link', { company: c.id, on: true });
        c.link = !!r.link;
        draw(r.link);
      }));
      if (off) off.addEventListener('click', () => busy(row, async () => {
        await api('bot.company.link', { company: c.id, on: false });
        c.link = false;
        toast('Ссылка отозвана');
        draw('');
      }));
      card.replaceChildren(ui.hint('Ссылка одноразовая: кто первым перейдёт по ней, станет владельцем комнаты и сможет открывать пары её событий. Отправляй её лично.'),
        ...shown, h('div', { class: 'ds-row' }, ...row));
    };
    draw('');
    return card;
  }

  // Передать права владельца бота: кому — человеку компаний, которые ведёт владелец бота (как раньше — администратору
  // или участнику; сервер проверяет то же)
  function openBotOwnerSheet(btn) {
    busy(btn, async () => {
      const lists = await Promise.all(state.companies.filter((c) => c.bot_run).map((c) => capi('admin.users', { company: c.id })));
      const byId = new Map();
      for (const d of lists) for (const u of d.users) if (u.id !== state.me.id) byId.set(u.id, u);
      const people = [...byId.values()].sort((a, b) => a.name.localeCompare(b.name, 'ru'));
      const sel = h('select', { class: 'ds-input' }, h('option', { value: '' }, 'Кому передать…'), ...people.map((u) => h('option', { value: u.id }, u.name)));
      const go = ui.btn('👑 Передать права владельца бота', 'danger', 'ds-btn--block');
      go.addEventListener('click', async () => {
        const u = people.find((x) => String(x.id) === sel.value);
        if (!u) return toast('Выбери человека');
        if (!(await confirmBox(`Передать права владельца бота «${u.name}»? Он(а) будет создавать и удалять комнаты и назначать их владельцев. Ты останешься администратором своих комнат.`))) return;
        busy(go, async () => {
          await api('bot.owner.transfer', { id: u.id });
          closeSheet();
          toast('👑 Права владельца бота переданы');
          await reloadMe();
          await goRoot();
        });
      });
      openSheet('Права владельца бота', [ui.hint('Владелец бота один. Передать права можно человеку из комнат, которые ведёшь ты.'), sel, go]);
    });
  }
