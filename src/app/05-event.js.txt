  function tabHasDot(id) {
    const u = (state.member && state.member.unread) || {};
    if (id === 'my') return state.dirty || u.from_santa > 0;
    if (id === 'ward') return u.from_ward > 0;
    // 012: непрочитанный ответ организатора; у админа — ничьи обращения и новые сообщения во взятых им
    if (id === 'info') return !!(state.support && state.support.unread > 0);
    if (id === 'admin') return !!(state.supportAdmin && state.supportAdmin.waiting + state.supportAdmin.unread > 0);
    return false;
  }

  function tabsList() {
    const tabs = [];
    if (state.member) tabs.push(['my', '✉️', t('app.event.tab.my')], ['ward', '🎁', t('app.event.tab.ward')]);

  }

  function renderTabs() {
    const nav = document.getElementById('tabs');
    nav.hidden = false;
    nav.replaceChildren(...tabsList().map(([id, icon, label]) =>
      h('button', {
        class: 'ds-tab' + (state.tab === id ? ' ds-tab--active' : ''),

          if (!(await mayLeave())) return;

          render();

  }

  // Черновик письма (041; 058 — реже и сразу при уходе): пока пишут — сохранение не чаще раза в DRAFT_GAP, первое после
  // паузы — через DRAFT_PAUSE (draftDelay, мс); при уходе приложения с экрана и уходе с вкладки — сразу (saveLetterNow,
  // mayLeave). Сохранения всех редакторов — по одному (letterQueue): редактор, открытый заново, пока дописывается
  // прежний, ждёт его и берёт версию из его ответа (letterRevisions — по ключу копии) — иначе вторая запись с той же
  // версией получила бы отказ «письмо изменилось»
  const DRAFT_PAUSE = 450;
  const DRAFT_GAP = 5000;
  const draftDelay = (now, last) => Math.max(DRAFT_PAUSE, last + DRAFT_GAP - now);
  let letterQueue = Promise.resolve();
  const letterRevisions = new Map();
  let saveLetterNow = null;
  onVisibility((shown) => { if (!shown && saveLetterNow) saveLetterNow(); });

  // Локальная копия страхует ввод до ответа сервера, включая мгновенное закрытие Telegram.
  function letterDraftEditor(m, markDirty) {
    const event = state.evId, draft = m.draft, editable = state.can.edit_letter;
    const storageKey = ['santa-letter', state.me.id, event, draft.joined_at].join('/');
    const savedText = () => draft.stale ? m.wishlist : (draft.text ?? m.wishlist);
    let backup;
    try { backup = JSON.parse(localStorage.getItem(storageKey)); } catch {}
    if (typeof backup?.text !== 'string' || backup.text.length > state.limits.chars) backup = null;
    const resume = editable && backup?.revision === draft.revision;
    let recovery = backup && !resume && backup.text !== savedText() ? backup.text : null;
    let pending = false, timer = null, failure = '', saving = false, urgent = false, last = 0;
    const field = ui.textarea({ class: 'ds-input ds-input--letter', 'aria-label': t('app.event.draft.label'),
      placeholder: t('app.event.letter.placeholderShort'), value: resume ? backup.text : savedText(),
      maxlength: state.limits.chars, readonly: !editable });
    const title = h('strong'), detail = ui.hint('');
    const status = h('div', { class: 'ds-notice ds-notice--letter', role: 'status' }, title, detail);
    const preview = ui.btn(t('app.event.draft.preview'), 'secondary', 'ds-btn--sm');
    preview.onclick = () => openSheet(t('app.event.draft.previewTitle'), h('div', { class: 'ds-letter' }, m.wishlist));
    const restore = ui.btn(t('app.event.draft.restore'), 'secondary');
    const retry = ui.btn(t('app.event.draft.retry'), 'secondary');
    const publish = ui.btn('', '', 'ds-btn--block ds-btn--letter-publish');
    const paper = h('div', {}, postcard(m.card, field));
    const card = ui.btn(t('app.event.card.button', { title: cardTitle(m.card) }), 'secondary', 'ds-btn--block');
    card.onclick = () => openPostcardSheet((id) => {
      paper.replaceChildren(postcard(id, field));
      card.textContent = t('app.event.card.button', { title: cardTitle(id) });
    });
    const counter = ui.hint('');
    function paint() {
      const published = !!m.wishlist.trim(), changed = field.value !== m.wishlist;
      title.textContent = t(changed ? (published ? 'app.event.draft.changes' : 'app.event.draft.draft')
        : published ? 'app.event.draft.published' : 'app.event.draft.empty');
      const visibility = t(changed && published ? 'app.event.draft.seesPrevious' : published
        ? 'app.event.draft.seesLetter' : 'app.event.draft.seesNothing');
      detail.textContent = visibility + (failure ? ' ' + failure : pending ? ' ' + t('app.event.draft.saving') : '');
      preview.hidden = !published || !changed;
      restore.hidden = recovery === null && !draft.stale;
      restore.textContent = t(editable ? 'app.event.draft.restore' : 'app.event.draft.viewPrevious');
      retry.hidden = !failure || !pending;
      card.hidden = publish.hidden = !editable;
      publish.textContent = t(published ? 'app.event.draft.publishChanges' : 'app.event.draft.publishLetter');
      publish.disabled = saving || !changed || !field.value.trim();
      counter.textContent = field.value.length + ' / ' + state.limits.chars;
      if (field.isConnected) markDirty(pending || saving);
    }
    function remember(text) {
      try { localStorage.setItem(storageKey, JSON.stringify({ text, revision: draft.revision })); } catch {}
    }
    function forget(text, revision) {
      try {
        const cached = JSON.parse(localStorage.getItem(storageKey));
        if (cached?.text === text && cached.revision === revision) localStorage.removeItem(storageKey);
      } catch {}
    }
    // Сохранение по расписанию: уже запланированное новая правка не откладывает
    function schedule() {
      if (timer || saving || !pending || failure) return;
      timer = setTimeout(() => { timer = null; write(false); }, urgent ? 0 : draftDelay(Date.now(), last));
    }
    function edited() {
      pending = true;
      failure = '';
      remember(field.value);
      schedule();
      paint();
    }
    // Сразу: приложение ушло с экрана, уход с вкладки; идёт сохранение — сразу после него
    function saveNow() {
      if (!pending || failure) return;
      urgent = true;
      clearTimeout(timer);
      timer = null;
      if (!saving) write(false);
    }
    // Один запрос за раз во всём приложении. Текст, набранный во время записи, уходит со следующей версией.
    async function write(publishing) {
      if (saving || (!pending && !publishing)) return;
      clearTimeout(timer);
      timer = null;
      urgent = false;
      saving = true;
      paint();
      const turn = letterQueue.then(async () => {
        draft.revision = Math.max(draft.revision, letterRevisions.get(storageKey) || 0);
        const text = field.value, revision = draft.revision;
        last = Date.now();
        try {
          const result = await api(publishing ? 'wishlist.publish' : 'wishlist.draft',
            { event, text, revision, joined_at: draft.joined_at });
          draft.revision = result.revision;
          letterRevisions.set(storageKey, result.revision);
          draft.stale = false;
          if (publishing) { m.wishlist = text; m.wishlist_updated_at = result.updated_at; }
          draft.text = text === m.wishlist ? null : text;
          pending = field.value !== text;
          if (pending) remember(field.value); else forget(text, revision);
          failure = '';
        } catch (e) {
          failure = t(publishing ? 'app.event.draft.publishFailed' : 'app.event.draft.saveFailed', { error: e.message });
        }
      });
      letterQueue = turn;
      await turn;
      saving = false;
      paint();
      schedule();
    }
    restore.onclick = () => {
      const previous = recovery ?? draft.text;
      if (!editable) { openSheet(t('app.event.draft.previousTitle'), h('div', { class: 'ds-letter' }, previous)); return; }
      recovery = null;
      field.value = previous;
      field.dispatchEvent(new Event('input'));
    };
    retry.onclick = () => write(false);
    publish.onclick = () => write(true);
    field.addEventListener('input', edited);
    saveLetterNow = saveNow;
    if (resume && field.value !== savedText()) edited(); else paint();
    return h('div', { class: 'ds-card ds-stack ds-card--draft' }, status, preview, restore, paper, counter, retry, card, publish);
  }

  // cached — перерисовать по последнему ответу (после закрытия переписки), не запрашивая ward.get заново
  // Шторка «Открытка для письма» (038): «без открытки» и мини-открытки набора, выбранная обведена; saved(id из ответа)
  const cardTitle = (id) => (state.looks.postcards.find((c) => c.id === id) || { title: t('app.event.card.none') }).title;

  const renderWard = screen(async ({ cached = false } = {}) => {
    const nav = state.nav, evId = state.evId;
    let d = cached && state.wardCache && state.wardCache.evId === evId ? state.wardCache.d : null;

      showLoader();
      try { d = await api('ward.get', { event: evId }); } catch (e) { if (nav === state.nav) showError(e); return; }

      state.wardCache = { evId, d };

      main.replaceChildren(...[finishedNotice(), d.state === 'no_draw'

      const marks = { '\u0001': h('mark', { class: 'ds-diff-add' }, t('app.event.ward.added')), '\u0002': h('del', { class: 'ds-diff-del' }, t('app.event.ward.removed')) };

    const gift = h('input', { type: 'checkbox', checked: d.gift_ready, disabled: finished() || d.shipment?.sent });
    bindToggle(gift, async (on) => {
      await api('ward.gift', { event: evId, ready: on });
      d.gift_ready = on;
      haptic('success');
      toast(t(on ? 'app.event.gift.done' : 'app.event.markRemoved'));
    });

    main.replaceChildren(...[
      finishedNotice(),
      h('div', { class: 'ds-ward-card' }, h('div', { class: 'ds-ward-card__label' }, t(finished() ? 'app.event.ward.wasSantaFor' : 'app.event.ward.santaFor')), nameEl, toggle),
      ui.section(t('app.event.ward.letterTitle')),
      ui.card(letter, d.ward.updated_at && text.trim() ? ui.hint(t('app.event.ward.updated', { time: fmtTime(d.ward.updated_at) })) : null),
      d.ward.delivery !== undefined ? ui.section(t('app.event.delivery.title')) : null,
      d.ward.delivery !== undefined
        ? ui.card(d.ward.delivery ? h('div', { class: 'ds-letter' }, d.ward.delivery) : h('div', { class: 'ds-text-muted' }, t('app.event.delivery.wardNone')))
        : null,
      h('div', { class: 'ds-card' }, ui.toggleRow(t('app.event.gift.title'), t('app.event.gift.sub'), gift)),
      d.shipment ? outgoingShipment(d, evId) : null,
      ui.section(t('app.event.chat.wardTitle')),
      ui.card(chatHint, maskCtl, chatButton('ward')),
    ].filter(Boolean));
  });

  function shipmentCard(s, ...actions) {
    const rows = [[t('app.event.shipment.date'), fmtDate(s.date)], [t('app.event.shipment.carrier'), s.carrier], [t('app.event.shipment.tracking'), s.tracking]].filter(([, v]) => v);
    const copy = ui.btn(t('app.event.shipment.copy'), 'secondary', 'ds-btn--sm');
    copy.onclick = async () => {
      try { await navigator.clipboard.writeText(s.tracking); toast(t('app.event.shipment.copied')); }
      catch {
        const input = h('input', { class: 'ds-input', readonly: true, value: s.tracking, 'aria-label': t('app.event.shipment.tracking') });
        openSheet(t('app.event.shipment.copyTitle'), input);
        input.focus(); input.select();
      }
    };
    return ui.card(h('div', { class: 'ds-text-strong' }, t(s.sent ? 'app.event.shipment.sent' : 'app.event.shipment.notSent')),
      rows.length ? h('dl', { class: 'ds-info-grid' }, rows.flatMap(([k, v]) => [h('dt', {}, k), h('dd', {}, v)])) : null,
      s.tracking ? copy : null, ...actions);
  }

  function outgoingShipment(d, event) {
    const s = d.shipment;
    if (finished()) return shipmentCard(s);
    const save = async (fields) => {
      const result = await api('ward.shipment', { event, version: s.version, ...fields });
      d.shipment = result.shipment;
      d.gift_ready = true;
      if (state.evId === event && state.tab === 'ward') await backToEvent(event);
    };
    const edit = ui.btn(t(s.sent ? 'app.event.shipment.edit' : 'app.event.shipment.mark'), s.sent ? 'secondary' : '', 'ds-btn--block');
    edit.onclick = () => {
      const today = new Date(Date.now() + d.timezone_offset_hours * 3600000).toISOString().slice(0, 10);
      const date = ui.date(s.sent ? s.date : today);
      date.max = today;
      const carrier = h('input', { class: 'ds-input', value: s.carrier || '', maxlength: 200 });
      const tracking = h('input', { class: 'ds-input', value: s.tracking || '', maxlength: 200 });
      const submit = ui.btn(t(s.sent ? 'app.event.shipment.saveChanges' : 'app.event.shipment.confirm'), '', 'ds-btn--block');
      const dateError = ui.hint(t('app.event.shipment.dateError'));
      dateError.hidden = true;
      date.oninput = () => { dateError.hidden = date.validity.valid; };
      const own = openSheet(t('app.event.shipment.title'), [ui.hint(t('app.event.shipment.hint')),
        field(t('app.event.shipment.date'), date), dateError, field(t('app.event.shipment.carrier'), carrier), field(t('app.event.shipment.tracking'), tracking), submit]);
      submit.onclick = () => busy(submit, async () => {
        if (!date.validity.valid) { dateError.hidden = false; date.focus(); return; }
        await save({ sent: true, date: date.value, carrier: carrier.value, tracking: tracking.value });
        if (own.isConnected) closeSheet();
      });
    };
    const cancel = ui.btn(t('app.event.shipment.cancel'), 'secondary', 'ds-btn--block');
    cancel.onclick = () => busy(cancel, async () => {
      if (await confirmBox(t('app.event.shipment.cancelConfirm'))) await save({ sent: false });
    });
    return shipmentCard(s, edit, s.sent ? cancel : null);
  }

  // ---------- вкладка «Инфо» ----------
  // 037: участники события — имена, роли и общие числа, без отметок по людям (specs/037-info-people/contracts/info-people.md)
  const ROSTER_FOLD = 8;
  function peopleSection() {
    const g = state.ev;
    const r = state.roster;
    if (!r) return [];
    const { people, progress: pr } = r;
    const progress = !pr.of ? null : pr.letters === undefined
      ? t('app.event.people.gifts', { ready: pr.ready, received: pr.received, of: pr.of })
      : t('app.event.people.letters', { letters: pr.letters, of: pr.of });
    const list = h('div', { class: 'ds-card ds-card--flush' });
    if (!people.length) list.append(h('div', { class: 'ds-list-item ds-text-muted' }, t('app.event.people.empty')));
    const folded = state.rosterOpen !== g.id && people.length > ROSTER_FOLD;
    list.append(...(folded ? people.slice(0, ROSTER_FOLD) : people).map(personRow));
    if (folded) {
      const more = h('div', { class: 'ds-list-item ds-list-item--link', role: 'button', onclick: () => {
        state.rosterOpen = g.id;
        more.replaceWith(...people.slice(ROSTER_FOLD).map(personRow));
      } }, h('div', { class: 'ds-grow ds-text-muted' }, t('app.event.people.showAll', { count: people.length })), h('span', { class: 'ds-list-item__chevron' }, '›'));
      list.append(more);
    }
    const nodes = [ui.section(t('app.event.people.title', { count: people.length })), progress ? ui.hint(progress, 'ds-hint--inset') : null, list];
    if (r.invite_link) {
      const share = ui.btn(t('app.event.people.invite'), 'secondary', 'ds-btn--block');
      share.addEventListener('click', () => shareInvite(r.invite_link, g.title));
      nodes.push(ui.card(ui.hint(t(g.approval ? 'app.event.people.inviteRequest' : 'app.event.people.inviteJoin')), share));
    }
    return nodes;
  }
  // Строка участника: имя, «(ты)», роль; при разрешении организатора — @username и чат в Telegram по нажатию
  function personRow(u) {
    const mine = u.id === state.me.id;
    const title = personTitle(u);
    if (mine) title.append(h('span', { class: 'ds-hint ds-text-regular' }, t('app.home.you')));
    const role = u.role === 'owner' ? ui.chip(t('app.home.role.owner')) : u.role === 'admin' ? ui.chip(t('app.home.role.admin')) : null;
    if (!state.ev.contacts_on || !u.username || mine) return h('div', { class: 'ds-list-item' }, h('div', { class: 'ds-grow' }, title), role);
    return h('div', { class: 'ds-list-item ds-list-item--link', role: 'button', onclick: async () => {
      if (await confirmBox(t('app.event.people.openChat', { name: u.name }))) openTgLink('https://t.me/' + encodeURIComponent(u.username));
    } }, h('div', { class: 'ds-grow' }, title), role, h('span', { class: 'ds-list-item__chevron' }, '›'));
  }

  function renderInfo() {
    const g = state.ev;
    const rows = [];
    rows.push([t('app.event.info.status'), t(g.status === 'finished' ? 'app.event.info.finished' : 'app.home.active')]);
    if (g.budget) rows.push([t('app.event.info.budget'), g.budget]);
    if (g.event_date) rows.push([t('app.event.info.date'), fmtDate(g.event_date)]);
    if (g.place) rows.push([t('app.event.info.place'), g.place]);
    if (g.deadline) rows.push([t('app.event.info.deadline'), fmtDate(g.deadline)]);
    rows.push([t('app.event.info.draw'), t(g.draw_done ? 'app.event.info.drawDone' : 'app.event.info.drawPending')]);
    if (!g.draw_done && g.status !== 'finished') rows.push([t('app.event.info.autoDraw'), fmtDate(g.draw_auto_date) || t('app.event.info.autoDrawNone')]);
    rows.push([t('app.event.info.ring'), t(g.ring ? 'app.event.info.ringOn' : 'app.event.info.ringOff')]);
    const managers = state.roster?.organizers ?? [];
    const owner = managers.find((u) => u.owner);
    const admins = managers.filter((u) => !u.owner);
    if (owner) rows.push([t('app.event.info.owner'), owner.name]);
    if (admins.length) rows.push([t('app.event.info.admins'), admins.map((u) => u.name).join(', ')]);
    const nodes = [
      finishedNotice(),
      ui.section(t('app.event.info.title')),
      h('div', { class: 'ds-card' }, h('dl', { class: 'ds-info-grid' }, rows.flatMap(([k, v]) => [h('dt', {}, k), h('dd', {}, v)]))),
    ];
    if (g.notes) nodes.push(h('div', { class: 'ds-card ds-letter' }, g.notes));
    nodes.push(...peopleSection());
    if (state.announcements.length) {
      nodes.push(ui.section(t('app.event.info.announcements')), ...state.announcements.map((a) => ui.card(
        h('div', { class: 'ds-letter' }, a.text),
        ui.hint(`📣 ${a.author_name} · ${fmtTime(a.created_at)}`))));
    }
    // 012: обращение к организатору — участнику эвента, если есть кому написать
    const sup = state.support;
    if (state.member && sup && (sup.available || sup.open || sup.unread)) {
      const btn = ui.btn(t(state.can.chat_write ? 'app.event.support.ask' : 'app.event.support.threads'), 'secondary', 'ds-btn--block');
      if (sup.unread) btn.append(h('span', { class: 'ds-badge' }, String(sup.unread)));
      btn.addEventListener('click', () => openSupport());
      const status = sup.open
        ? (sup.assignee_name ? t('app.event.support.ledBy', { name: sup.assignee_name }) : t('app.event.support.waiting')) : '';
      nodes.push(ui.section(t('app.event.support.section')), ui.card(
        ui.hint(t('app.event.support.hint')),
        status ? h('div', { class: 'ds-text-sm' }, status) : null,
        btn));
    }
    if (!state.member) nodes.push(ui.hint(t('app.event.info.notMember'), 'ds-hint--inset'));
    nodes.push(...donateSection());
    nodes.push(
      ui.section(t('app.event.how.title')),
      h('div', { class: 'ds-card' }, h('ul', { class: 'ds-bullets' },
        h('li', {}, t('app.event.how.letter')),
        h('li', {}, t('app.event.how.ward')),
        h('li', {}, t('app.event.how.after')))),
      ...leaveSection());
    main.replaceChildren(...nodes.filter(Boolean));
  }

  // 035: выход участника — внизу «Инфо», только из идущего события; сам — до жеребьёвки (слово ВЫЙТИ), после неё —
  // через организатора: обращение, если есть кому написать. 036: после жеребьёвки организатора удалить может не всякий —
  // владельца никто (выход только через отмену жеребьёвки, до раскрытия Сант), админа — только владелец
  function leaveSection() {
    const g = state.ev;
    if (!state.member || g.status === 'finished') return [];
    if (g.draw_done && state.me.is_owner) {
      return [ui.section(t('app.event.leave.section')), ui.card(ui.hint(t(g.revealed ? 'app.event.leave.ownerRevealed' : 'app.event.leave.ownerDrawn',
        { owner: ownerWord() })))];
    }
    if (g.draw_done) {
      const sup = state.support;
      const write = sup && (sup.available || sup.open) ? ui.btn(t('app.event.support.ask'), 'secondary', 'ds-btn--block') : null;
      if (write) write.addEventListener('click', () => openSupport());
      return [ui.section(t('app.event.leave.section')), ui.card(ui.hint(state.me.is_admin
        ? t('app.event.leave.adminDrawn', { owner: ownerWord() })
        : t('app.event.leave.memberDrawn')), write)];
    }
    const leave = ui.btn(t('app.event.leave.button'), 'danger', 'ds-btn--block');
    leave.addEventListener('click', () => typedSheet(t('app.event.leave.sheetTitle'), {
      lines: [t('app.event.leave.lostLetter'), t('app.event.leave.lostThreads')],
      keep: t('app.event.leave.keep'),
      word: t('app.core.word.leave'),
      label: t('app.event.leave.confirm'),
      run: (word) => api('event.leave', { event: g.id, confirm: word }),
      after: async () => { toast(t('app.event.leave.done', { title: g.title })); await goHome(); },
    }));
    return [ui.section(t('app.event.leave.section')), ui.card(ui.hint(t('app.event.leave.hint')), leave)];
  }
