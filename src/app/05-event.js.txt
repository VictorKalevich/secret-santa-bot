  function renderHero() {
    const e = state.ev;
    // 038: «‹» — всегда: на главном экране «Профиль» (имя, значок) и «Уведомления» нужны и участнику с одним событием
    heroBack.hidden = false;

    if (e.event_date) parts.push('🎁 ' + fmtDate(e.event_date));
    if (e.budget) parts.push('💰 ' + e.budget);
    // неразрывные пробелы внутри частей: «до 50 BYN» не разрывается на две строки
    heroSub.textContent = parts.map((p) => p.replace(/ /g, '\u00a0')).join(' · ') || 'Тайный Санта';

  function tabHasDot(id) {
    const u = (state.member && state.member.unread) || {};
    if (id === 'my') return state.dirty || u.from_santa > 0;
    if (id === 'ward') return u.from_ward > 0;
    // 012: непрочитанный ответ организатора; у админа — ничьи обращения и новые сообщения во взятых им
    if (id === 'info') return !!(state.support && state.support.unread > 0);
    if (id === 'admin') return !!(state.supportAdmin && state.supportAdmin.waiting + state.supportAdmin.unread > 0);
    return false;
  }

  function tabsList() {
    const t = [];
    if (state.member) t.push(['my', '✉️', 'Письмо'], ['ward', '🎁', 'Подопечный']);
    t.push(['info', 'ℹ️', 'Инфо']);
    if (state.me.is_admin) t.push(['admin', '🛠', 'Управление']);
    return t;
  }

  function renderTabs() {
    const nav = document.getElementById('tabs');
    nav.hidden = false;
    nav.replaceChildren(...tabsList().map(([id, icon, label]) =>
      h('button', {
        class: 'ds-tab' + (state.tab === id ? ' ds-tab--active' : ''),
        onclick: async () => {
          if (state.tab === id) return;
          if (!(await mayLeave())) return;
          if (id === 'my' && state.ev.delivery_on) return openEvent(state.evId, 'my');
          state.tab = id;
          render();
        },
      }, h('span', { class: 'ds-tab__icon' }, icon), h('span', { class: 'ds-tab__label' }, label),
        tabHasDot(id) ? h('span', { class: 'ds-tab__dot' }) : null)
    ));
  }

  // Плашка завершённого эвента; e — данные события (панель передаёт свой ответ, 026), по умолчанию — экрана события
  function finishedNotice(e = state.ev) {
    if (!e || e.status !== 'finished') return null;
    const until = new Date(e.chat_until);
    const chat = state.can.chat_write
      ? `Переписка открыта до ${until.getDate()} ${MONTHS[until.getMonth()]}, потом — только чтение.`
      : 'Переписка — только для чтения.';
    return h('div', { class: 'ds-notice ds-text-sm' }, `🏁 Событие завершено ${fmtTime(e.finished_at)}. Письмо и отметки — только для чтения. ${chat}`);
  }

  // Черновик письма (041; 058 — реже и сразу при уходе): пока пишут — сохранение не чаще раза в DRAFT_GAP, первое после
  // паузы — через DRAFT_PAUSE (draftDelay, мс); при уходе приложения с экрана и уходе с вкладки — сразу (saveLetterNow,
  // mayLeave). Сохранения всех редакторов — по одному (letterQueue): редактор, открытый заново, пока дописывается
  // прежний, ждёт его и берёт версию из его ответа (letterRevisions — по ключу копии) — иначе вторая запись с той же
  // версией получила бы отказ «письмо изменилось»
  const DRAFT_PAUSE = 450;
  const DRAFT_GAP = 5000;
  const draftDelay = (now, last) => Math.max(DRAFT_PAUSE, last + DRAFT_GAP - now);
  let letterQueue = Promise.resolve();
  const letterRevisions = new Map();
  let saveLetterNow = null;
  onVisibility((shown) => { if (!shown && saveLetterNow) saveLetterNow(); });

  // Локальная копия страхует ввод до ответа сервера, включая мгновенное закрытие Telegram.
  function letterDraftEditor(m, markDirty) {
    const event = state.evId, draft = m.draft, editable = state.can.edit_letter;
    const storageKey = ['santa-letter', state.me.id, event, draft.joined_at].join('/');
    const savedText = () => draft.stale ? m.wishlist : (draft.text ?? m.wishlist);
    let backup;
    try { backup = JSON.parse(localStorage.getItem(storageKey)); } catch {}
    if (typeof backup?.text !== 'string' || backup.text.length > state.limits.chars) backup = null;
    const resume = editable && backup?.revision === draft.revision;
    let recovery = backup && !resume && backup.text !== savedText() ? backup.text : null;
    let pending = false, timer = null, failure = '', saving = false, urgent = false, last = 0;
    const field = ui.textarea({ class: 'ds-input ds-input--letter', 'aria-label': 'Текст письма',
      placeholder: 'Дорогой Санта! В этом году я мечтаю о…', value: resume ? backup.text : savedText(),
      maxlength: state.limits.chars, readonly: !editable });
    const title = h('strong'), detail = ui.hint('');
    const status = h('div', { class: 'ds-notice ds-notice--letter', role: 'status' }, title, detail);
    const preview = ui.btn('Посмотреть опубликованное', 'secondary', 'ds-btn--sm');
    preview.onclick = () => openSheet('Опубликованное письмо', h('div', { class: 'ds-letter' }, m.wishlist));
    const restore = ui.btn('Восстановить прежний черновик', 'secondary');
    const retry = ui.btn('Повторить сохранение', 'secondary');
    const publish = ui.btn('', '', 'ds-btn--block ds-btn--letter-publish');
    const paper = h('div', {}, postcard(m.card, field));
    const card = ui.btn('🖼 Открытка: ' + cardTitle(m.card), 'secondary', 'ds-btn--block');
    card.onclick = () => openPostcardSheet((id) => {
      paper.replaceChildren(postcard(id, field));
      card.textContent = '🖼 Открытка: ' + cardTitle(id);
    });
    const counter = ui.hint('');
    function paint() {
      const published = !!m.wishlist.trim(), changed = field.value !== m.wishlist;
      title.textContent = changed ? (published ? '📝 Черновик изменений' : '📝 Черновик')
        : published ? '✅ Опубликовано' : 'Письмо пока пустое';
      const visibility = changed && published ? 'Санте видна предыдущая версия.' : published
        ? 'Это письмо доступно твоему Санте.' : 'Санта пока не видит письмо.';
      detail.textContent = visibility + (failure ? ' ' + failure : pending ? ' Сохраняем…' : '');
      preview.hidden = !published || !changed;
      restore.hidden = recovery === null && !draft.stale;
      restore.textContent = editable ? 'Восстановить прежний черновик' : 'Посмотреть прежний черновик';
      retry.hidden = !failure || !pending;
      card.hidden = publish.hidden = !editable;
      publish.textContent = published ? 'Опубликовать изменения' : 'Опубликовать письмо';
      publish.disabled = saving || !changed || !field.value.trim();
      counter.textContent = field.value.length + ' / ' + state.limits.chars;
      if (field.isConnected) markDirty(pending || saving);
    }
    function remember(text) {
      try { localStorage.setItem(storageKey, JSON.stringify({ text, revision: draft.revision })); } catch {}
    }
    function forget(text, revision) {
      try {
        const cached = JSON.parse(localStorage.getItem(storageKey));
        if (cached?.text === text && cached.revision === revision) localStorage.removeItem(storageKey);
      } catch {}
    }
    // Сохранение по расписанию: уже запланированное новая правка не откладывает
    function schedule() {
      if (timer || saving || !pending || failure) return;
      timer = setTimeout(() => { timer = null; write(false); }, urgent ? 0 : draftDelay(Date.now(), last));
    }
    function edited() {
      pending = true;
      failure = '';
      remember(field.value);
      schedule();
      paint();
    }
    // Сразу: приложение ушло с экрана, уход с вкладки; идёт сохранение — сразу после него
    function saveNow() {
      if (!pending || failure) return;
      urgent = true;
      clearTimeout(timer);
      timer = null;
      if (!saving) write(false);
    }
    // Один запрос за раз во всём приложении. Текст, набранный во время записи, уходит со следующей версией.
    async function write(publishing) {
      if (saving || (!pending && !publishing)) return;
      clearTimeout(timer);
      timer = null;
      urgent = false;
      saving = true;
      paint();
      const turn = letterQueue.then(async () => {
        draft.revision = Math.max(draft.revision, letterRevisions.get(storageKey) || 0);
        const text = field.value, revision = draft.revision;
        last = Date.now();
        try {
          const result = await api(publishing ? 'wishlist.publish' : 'wishlist.draft',
            { event, text, revision, joined_at: draft.joined_at });
          draft.revision = result.revision;
          letterRevisions.set(storageKey, result.revision);
          draft.stale = false;
          if (publishing) { m.wishlist = text; m.wishlist_updated_at = result.updated_at; }
          draft.text = text === m.wishlist ? null : text;
          pending = field.value !== text;
          if (pending) remember(field.value); else forget(text, revision);
          failure = '';
        } catch (e) {
          failure = (publishing ? 'Публикация не удалась. ' : 'Черновик не сохранён. ') + e.message;
        }
      });
      letterQueue = turn;
      await turn;
      saving = false;
      paint();
      schedule();
    }
    restore.onclick = () => {
      const previous = recovery ?? draft.text;
      if (!editable) { openSheet('Прежний черновик', h('div', { class: 'ds-letter' }, previous)); return; }
      recovery = null;
      field.value = previous;
      field.dispatchEvent(new Event('input'));
    };
    retry.onclick = () => write(false);
    publish.onclick = () => write(true);
    field.addEventListener('input', edited);
    saveLetterNow = saveNow;
    if (resume && field.value !== savedText()) edited(); else paint();
    return h('div', { class: 'ds-card ds-stack ds-card--draft' }, status, preview, restore, paper, counter, retry, card, publish);
  }

  const renderWard = screen(async ({ cached = false } = {}) => {
    const nav = state.nav, evId = state.evId;
    let d = cached && state.wardCache && state.wardCache.evId === evId ? state.wardCache.d : null;
    if (!d) {
      showLoader();
      try { d = await api('ward.get', { event: evId }); } catch (e) { if (nav === state.nav) showError(e); return; }
      if (nav !== state.nav) return;
      state.wardCache = { evId, d };
    }
    if (d.state !== 'ok') {
      main.replaceChildren(...[finishedNotice(), d.state === 'no_draw'
        ? ui.empty('🎲', 'Жеребьёвка ещё не проведена', 'После жеребьёвки здесь появится имя и письмо твоего подопечного. Бот пришлёт уведомление.')

      return;
    }
    const nameEl = h('div', { class: 'ds-ward-card__name' + (state.wardNameShown ? '' : ' ds-ward-card__name--hidden') }, d.ward.name);
    const toggle = ui.btn(state.wardNameShown ? '🙈 Скрыть' : '👀 Показать имя', 'sm', 'ds-btn--on-festive');
    toggle.addEventListener('click', () => {
      state.wardNameShown = !state.wardNameShown;
      nameEl.classList.toggle('ds-ward-card__name--hidden', !state.wardNameShown);
      toggle.textContent = state.wardNameShown ? '🙈 Скрыть' : '👀 Показать имя';
    });

    const text = d.ward.wishlist || '';

    if (!text.trim()) {

    } else if (d.previous !== null && d.previous !== undefined) {
      letter = h('div', { class: 'ds-stack' },
        h('div', { class: 'ds-notice ds-text-sm' }, '✨ Письмо изменилось с твоего прошлого просмотра: новое ',
          h('mark', { class: 'ds-diff-add' }, 'выделено'), ', удалённое ', h('del', { class: 'ds-diff-del' }, 'зачёркнуто'), '.'),
        postcard(d.ward.card, renderDiff(d.previous, text)));
    } else {
      letter = postcard(d.ward.card, h('div', { class: 'ds-letter' }, text));
    }

    // 040: маска — пока событие идёт и тайна не раскрыта; потом только показывается
    const chatHint = ui.hint('');
    const maskCtl = !finished() && !state.ev.revealed ? ui.btn('', 'secondary', 'ds-btn--block')
      : maskLabel(d.mask) ? ui.hint('Маска в переписке: ' + maskLabel(d.mask) + '.') : null;
    const showMask = () => {
      chatHint.textContent = `Есть вопрос? Напиши подопечному — для него ты «${maskedSanta(d.mask)}», имя не раскрывается.`;
      if (maskCtl && maskCtl.tagName === 'BUTTON') maskCtl.textContent = '🎭 Маска: ' + (maskLabel(d.mask) || 'без маски');
    };
    if (maskCtl && maskCtl.tagName === 'BUTTON') maskCtl.addEventListener('click', () => openMaskSheet(d.mask, (mask) => { d.mask = mask; showMask(); }));
    showMask();

    const gift = h('input', { type: 'checkbox', checked: d.gift_ready, disabled: finished() || d.shipment?.sent });
    bindToggle(gift, async (on) => {
      await api('ward.gift', { event: evId, ready: on });
      d.gift_ready = on;
      haptic('success');
      toast(on ? '🎉 Отлично! Подарок отмечен как готовый' : 'Отметка снята');
    });

    main.replaceChildren(...[
      finishedNotice(),
      h('div', { class: 'ds-ward-card' }, h('div', { class: 'ds-ward-card__label' }, finished() ? 'Ты был(а) Тайным Сантой для' : 'Ты — Тайный Санта для'), nameEl, toggle),
      ui.section('Письмо подопечного'),
      ui.card(letter, d.ward.updated_at && text.trim() ? ui.hint('Обновлено ' + fmtTime(d.ward.updated_at)) : null),
      d.ward.delivery !== undefined ? ui.section('Как передать подарок') : null,
      d.ward.delivery !== undefined
        ? ui.card(d.ward.delivery ? h('div', { class: 'ds-letter' }, d.ward.delivery) : h('div', { class: 'ds-text-muted' }, 'Подопечный пока не указал адрес — спроси в переписке.'))
        : null,
      h('div', { class: 'ds-card' }, ui.toggleRow('🎁 Подарок готов', 'Отметку видят владелец и администраторы', gift)),
      d.shipment ? outgoingShipment(d, evId) : null,
      ui.section('Переписка с подопечным'),
      ui.card(chatHint, maskCtl, chatButton('ward')),
    ].filter(Boolean));
  });

  function shipmentCard(s, ...actions) {
    const rows = [['Дата отправки', fmtDate(s.date)], ['Служба доставки', s.carrier], ['Трек-номер', s.tracking]].filter(([, v]) => v);
    const copy = ui.btn('Скопировать трек-номер', 'secondary', 'ds-btn--sm');
    copy.onclick = async () => {
      try { await navigator.clipboard.writeText(s.tracking); toast('Трек-номер скопирован'); }
      catch {
        const input = h('input', { class: 'ds-input', readonly: true, value: s.tracking, 'aria-label': 'Трек-номер' });
        openSheet('Скопируй трек-номер', input);
        input.focus(); input.select();
      }
    };
    return ui.card(h('div', { class: 'ds-text-strong' }, s.sent ? '📦 Подарок отправлен' : 'Отправка пока не отмечена'),
      rows.length ? h('dl', { class: 'ds-info-grid' }, rows.flatMap(([k, v]) => [h('dt', {}, k), h('dd', {}, v)])) : null,
      s.tracking ? copy : null, ...actions);
  }

  function outgoingShipment(d, event) {
    const s = d.shipment;
    if (finished()) return shipmentCard(s);
    const save = async (fields) => {
      const result = await api('ward.shipment', { event, version: s.version, ...fields });
      d.shipment = result.shipment;
      d.gift_ready = true;
      if (state.evId === event && state.tab === 'ward') await backToEvent(event);
    };
    const edit = ui.btn(s.sent ? 'Изменить данные' : 'Отметить отправку', s.sent ? 'secondary' : '', 'ds-btn--block');
    edit.onclick = () => {
      const today = new Date(Date.now() + d.timezone_offset_hours * 3600000).toISOString().slice(0, 10);
      const date = ui.date(s.sent ? s.date : today);
      date.max = today;
      const carrier = h('input', { class: 'ds-input', value: s.carrier || '', maxlength: 200 });
      const tracking = h('input', { class: 'ds-input', value: s.tracking || '', maxlength: 200 });
      const submit = ui.btn(s.sent ? 'Сохранить изменения' : 'Подтвердить отправку', '', 'ds-btn--block');
      const dateError = ui.hint('Выбери сегодняшний или прошедший день.');
      dateError.hidden = true;
      date.oninput = () => { dateError.hidden = date.validity.valid; };
      const own = openSheet('Отправка подарка', [ui.hint('Можно отметить отправку без подробностей.'),
        field('Дата отправки', date), dateError, field('Служба доставки', carrier), field('Трек-номер', tracking), submit]);
      submit.onclick = () => busy(submit, async () => {
        if (!date.validity.valid) { dateError.hidden = false; date.focus(); return; }
        await save({ sent: true, date: date.value, carrier: carrier.value, tracking: tracking.value });
        if (own.isConnected) closeSheet();
      });
    };
    const cancel = ui.btn('Отменить отметку', 'secondary', 'ds-btn--block');
    cancel.onclick = () => busy(cancel, async () => {
      if (await confirmBox('Снять отметку об отправке? Подопечный узнает, что отправка отменена.')) await save({ sent: false });
    });
    return shipmentCard(s, edit, s.sent ? cancel : null);
  }

  // ---------- вкладка «Инфо» ----------
  // 037: участники события — имена, роли и общие числа, без отметок по людям (specs/037-info-people/contracts/info-people.md)
  const ROSTER_FOLD = 8;
  function peopleSection() {
    const g = state.ev;
    const r = state.roster;
    if (!r) return [];
    const { people, progress: pr } = r;
    const progress = !pr.of ? null : pr.letters === undefined
      ? `🎁 Подарки готовы: ${pr.ready} из ${pr.of} · 📦 получено: ${pr.received} из ${pr.of}`
      : `✉️ Письма написали ${pr.letters} из ${pr.of}`;
    const list = h('div', { class: 'ds-card ds-card--flush' });
    if (!people.length) list.append(h('div', { class: 'ds-list-item ds-text-muted' }, 'Пока никого нет.'));
    const folded = state.rosterOpen !== g.id && people.length > ROSTER_FOLD;
    list.append(...(folded ? people.slice(0, ROSTER_FOLD) : people).map(personRow));
    if (folded) {
      const more = h('div', { class: 'ds-list-item ds-list-item--link', role: 'button', onclick: () => {
        state.rosterOpen = g.id;
        more.replaceWith(...people.slice(ROSTER_FOLD).map(personRow));
      } }, h('div', { class: 'ds-grow ds-text-muted' }, `Показать всех (${people.length})`), h('span', { class: 'ds-list-item__chevron' }, '›'));
      list.append(more);
    }
    const nodes = [ui.section(`Участники · ${people.length}`), progress ? ui.hint(progress, 'ds-hint--inset') : null, list];
    if (r.invite_link) {
      const share = ui.btn('📨 Пригласить друга', 'secondary', 'ds-btn--block');
      share.addEventListener('click', () => shareInvite(r.invite_link, g.title));
      nodes.push(ui.card(ui.hint(g.approval
        ? 'Позови друга — он сможет отправить заявку на участие.'
        : 'Позови друга — по ссылке он сразу вступит в событие.'), share));
    }
    return nodes;
  }
  // Строка участника: имя, «(ты)», роль; при разрешении организатора — @username и чат в Telegram по нажатию
  function personRow(u) {
    const mine = u.id === state.me.id;
    const title = personTitle(u);
    if (mine) title.append(h('span', { class: 'ds-hint ds-text-regular' }, '  (ты)'));
    const role = u.role === 'owner' ? ui.chip('👑 владелец') : u.role === 'admin' ? ui.chip('🛠 администратор') : null;
    if (!state.ev.contacts_on || !u.username || mine) return h('div', { class: 'ds-list-item' }, h('div', { class: 'ds-grow' }, title), role);
    return h('div', { class: 'ds-list-item ds-list-item--link', role: 'button', onclick: async () => {
      if (await confirmBox(`Открыть чат с ${u.name} в Telegram?`)) openTgLink('https://t.me/' + encodeURIComponent(u.username));
    } }, h('div', { class: 'ds-grow' }, title), role, h('span', { class: 'ds-list-item__chevron' }, '›'));
  }

  function renderInfo() {
    const g = state.ev;
    const rows = [];
    rows.push(['Статус', g.status === 'finished' ? 'завершён 🏁' : 'идёт']);
    if (g.budget) rows.push(['Бюджет', g.budget]);
    if (g.event_date) rows.push(['Вручение', fmtDate(g.event_date)]);
    if (g.place) rows.push(['Где', g.place]);
    if (g.deadline) rows.push(['Письмо до', fmtDate(g.deadline)]);
    rows.push(['Жеребьёвка', g.draw_done ? 'проведена ✅' : 'ещё не проведена']);
    if (!g.draw_done && g.status !== 'finished') rows.push(['Автожеребьёвка', fmtDate(g.draw_auto_date) || 'не установлена']);
    rows.push(['Кольцевая', g.ring ? 'да — взаимных пар нет' : 'нет — взаимные пары возможны']);
    const managers = state.roster?.organizers ?? [];
    const owner = managers.find((u) => u.owner);
    const admins = managers.filter((u) => !u.owner);
    if (owner) rows.push(['Владелец', owner.name]);
    if (admins.length) rows.push(['Администраторы', admins.map((u) => u.name).join(', ')]);
    const nodes = [
      finishedNotice(),
      ui.section('О событии'),
      h('div', { class: 'ds-card' }, h('dl', { class: 'ds-info-grid' }, rows.flatMap(([k, v]) => [h('dt', {}, k), h('dd', {}, v)]))),
    ];
    if (g.notes) nodes.push(h('div', { class: 'ds-card ds-letter' }, g.notes));
    nodes.push(...peopleSection());
    if (state.announcements.length) {
      nodes.push(ui.section('Объявления'), ...state.announcements.map((a) => ui.card(
        h('div', { class: 'ds-letter' }, a.text),
        ui.hint(`📣 ${a.author_name} · ${fmtTime(a.created_at)}`))));
    }
    // 012: обращение к организатору — участнику эвента, если есть кому написать
    const sup = state.support;
    if (state.member && sup && (sup.available || sup.open || sup.unread)) {
      const btn = ui.btn(state.can.chat_write ? '💬 Задать вопрос' : '💬 Обращения', 'secondary', 'ds-btn--block');
      if (sup.unread) btn.append(h('span', { class: 'ds-badge' }, String(sup.unread)));
      btn.addEventListener('click', () => openSupport());
      const status = sup.open ? (sup.assignee_name ? `Твоё обращение ведёт ${sup.assignee_name}.` : 'Обращение отправлено и ждёт ответа.') : '';
      nodes.push(ui.section('Нужна помощь?'), ui.card(
        ui.hint('Задай вопрос по событию. Ответ придёт сюда и в чат бота. Сохраняй имя подопечного в тайне.'),
        status ? h('div', { class: 'ds-text-sm' }, status) : null,
        btn));
    }
    if (!state.member) nodes.push(ui.hint('Ты не участвуешь в этом событии — тебе доступны вкладки «Инфо» и «Управление».', 'ds-hint--inset'));
    nodes.push(...donateSection());
    nodes.push(
      ui.section('Как это работает'),
      h('div', { class: 'ds-card' }, h('ul', { class: 'ds-bullets' },
        h('li', {}, 'Напиши письмо Санте — его увидит тот, кто будет дарить тебе подарок.'),
        h('li', {}, 'После жеребьёвки во вкладке «Подопечный» появится, кому даришь ты.'),
        h('li', {}, 'После завершения события письмо остаётся для чтения, переписка открыта ещё 7 дней. Событие сохранится в истории.'))),
      ...leaveSection());
    main.replaceChildren(...nodes.filter(Boolean));
  }

  // 035: выход участника — внизу «Инфо», только из идущего события; сам — до жеребьёвки (слово ВЫЙТИ), после неё —
  // через организатора: обращение, если есть кому написать. 036: после жеребьёвки организатора удалить может не всякий —
  // владельца никто (выход только через отмену жеребьёвки, до раскрытия Сант), админа — только владелец
  function leaveSection() {
    const g = state.ev;
    if (!state.member || g.status === 'finished') return [];
    if (g.draw_done && state.me.is_owner) {
      return [ui.section('Выход из события'), ui.card(ui.hint(g.revealed
        ? `Тайна уже раскрыта — жеребьёвку не отменить, поэтому выйти из события нельзя: удалить ${ownerWord().gen} некому.`
        : `Ты ${ownerWord().nom} — удалить тебя из события некому. Выйти можно, только отменив жеребьёвку: «Управление» → «Опасная зона» → ` +
          '«↩️ Отменить жеребьёвку» (пары удалятся у всех, участники получат уведомление). Потом выключи «Я участвую» и проведи жеребьёвку заново.'))];
    }
    if (g.draw_done) {
      const sup = state.support;
      const write = sup && (sup.available || sup.open) ? ui.btn('💬 Задать вопрос', 'secondary', 'ds-btn--block') : null;
      if (write) write.addEventListener('click', () => openSupport());
      return [ui.section('Выход из события'), ui.card(ui.hint(state.me.is_admin
        ? `Жеребьёвка уже прошла — удалить из события администратора может только ${ownerWord().nom}. Отправь обращение, чтобы владелец мог помочь.`
        : 'Жеребьёвка уже прошла. Чтобы выйти, обратись к владельцу или администраторам.'), write)];
    }
    const leave = ui.btn('🚪 Выйти из события', 'danger', 'ds-btn--block');
    leave.addEventListener('click', () => typedSheet('Выйти из события', {
      lines: ['твоё письмо Санте и адрес доставки;', 'твои обращения по этому событию.'],
      keep: 'Владелец и администраторы получат уведомление о твоём выходе. Вернуться можно по пригласительной ссылке, пока регистрация открыта.',
      word: 'ВЫЙТИ',
      label: '🚪 Выйти',
      run: (word) => api('event.leave', { event: g.id, confirm: word }),
      after: async () => { toast(`Ты больше не участвуешь в событии «${g.title}»`); await goHome(); },
    }));
    return [ui.section('Выход из события'), ui.card(ui.hint('Не получается участвовать? Выйти можно, пока не проведена жеребьёвка.'), leave)];
  }
