
  // ---------- вкладка «Управление» ----------
  // Панель по секциям (030): renderAdmin читает ответ своего события и собирает объект панели p, секции — функции
  // adminXxx(p) → узлы. Контракт — specs/030-admin-panel-parts/contracts/admin-panel.md
  const renderAdmin = screen(async ({ quiet = false } = {}) => {
    // Экран строится для эвента evId (009): все действия — над ним; ответ для покинутого экрана не рисуется.
    // Событие — только из своего ответа d (026): не из state.ev, которое могло устареть.
    // quiet (026) — тихая перестройка: без загрузчика, прежняя панель на экране до ответа, прокрутка сохраняется
    const nav = state.nav, evId = state.evId;
    const live = () => nav === state.nav;
    const xapi = (action, body) => api(action, { ...(body || {}), event: evId });
    const again = (opts) => (live() ? renderAdmin(opts) : undefined);
    if (!quiet) showLoader();
    let d;
    try { d = await xapi('admin.state'); } catch (e) { if (live()) showError(e); return; }
    if (!live()) return;
    const s = d.settings;
    const refresh = async (opts) => {
      if (!live()) return;
      await reloadMe().catch(() => {});
      await reloadEvent(evId).catch(() => {});
      if (!live()) return;
      renderHero();
      if (!state.member && (state.tab === 'my' || state.tab === 'ward')) state.tab = 'admin';
      renderTabs();
      syncBack();
      await again(opts);
    };
    // Параметр, от которого зависят другие секции, меняется через set: подписанные watch обновляются на месте
    const watchers = {};
    const p = {
      d, s, evId, xapi, again, refresh, live,
      users: d.users,
      byId: Object.fromEntries(d.users.map((u) => [u.id, u])),
      players: d.users.filter((u) => u.registered),
      me: state.me,
      done: s.status === 'finished',
      drafts: state.drafts[evId] || (state.drafts[evId] = {}),
      watch: (key, fn) => (watchers[key] || (watchers[key] = [])).push(fn),
      set: (key, value) => { s[key] = value; for (const fn of watchers[key] || []) fn(); },
    };
    // Обращения (012) — после всего, что касается приглашения, перед объявлением. Завершённый эвент: заявки (только
    // отклонить, 025) и обращения — на том же месте, перед участниками
    const sections = p.done
      ? [adminRequests, adminSupport]
      : [adminInvite, adminRequests, adminRemoved, adminInviteFrom, adminSupport, adminAnnounce];
    // Владелец бота без участия и организаторства — гость (063): по своему ответу, не по state.ev (026)
    const guest = state.botOwner && !d.me_member && !d.organizers.some((u) => u.id === state.me.id);
    const nodes = [guestNotice(guest), finishedNotice(s)];
    for (const section of [...sections, adminMembers, adminExclusions, adminFixed, adminDraw, adminParams, adminLog, adminSwap, adminVault, adminOrganizers, adminDanger]) {
      nodes.push(...section(p));
    }
    const y = window.scrollY;
    main.replaceChildren(...nodes.filter(Boolean));
    if (quiet) window.scrollTo(0, y);
  });

  // Заголовок строки списка: имя и @username (заявки, удалённые, участники)
  const personTitle = (u) => h('div', { class: 'ds-list-item__title' }, u.name || t('app.core.noName'), u.username ? h('span', { class: 'ds-hint ds-text-regular' }, '  @' + u.username) : null);
  // Санта не открыл подопечного — «Участники» и шторка обмена (049)
  const unseenChip = () => ui.chip(t('app.admin.unseen'), 'warning');

  // --- приглашение (только для идущего эвента)
  function adminInvite(p) {
    const { s, d, xapi, again, refresh } = p;
    const regToggle = h('input', { type: 'checkbox', checked: s.registration_open });
    bindToggle(regToggle, async (on) => {
      await xapi('admin.registration', { open: on });
      toast(on ? t('app.admin.invite.regOpen') : t('app.admin.invite.regClosed'));
      p.set('registration_open', on); // «Пригласить участников» и подсказка у заявок — на месте, без перестройки (030)
    });
    const linkInput = h('input', { class: 'ds-input', type: 'text', value: s.invite_link || t('app.admin.invite.noLink'), readonly: true });
    const copyBtn = ui.btn(t('app.admin.invite.copy'), 'secondary', 'ds-btn--sm');
    const shareBtn = ui.btn(t('app.core.link.share'), 'secondary', 'ds-btn--sm');
    // Без ссылки копировать и пересылать нечего (009)
    copyBtn.disabled = shareBtn.disabled = !s.invite_link;
    copyBtn.addEventListener('click', async () => {
      try { await navigator.clipboard.writeText(s.invite_link); toast(t('app.core.link.copied')); }
      catch (e) { linkInput.select(); toast(t('app.core.link.copyByHand')); }
    });
    shareBtn.addEventListener('click', () => shareInvite(s.invite_link, s.title));
    const regenBtn = ui.btn(t('app.admin.invite.regen'), 'danger', 'ds-btn--sm');
    regenBtn.addEventListener('click', async () => {
      if (!(await confirmBox(t('app.admin.invite.regenConfirm')))) return;
      busy(regenBtn, async () => { await xapi('admin.invite.regen'); toast(t('app.admin.invite.regenDone')); await again(); });
    });
    // 064: регистрация закрыта — ссылка неактивна, кнопок нет; переключатель меняет это на месте (p.set)
    const linkRow = h('div', { class: 'ds-row ds-row--wrap' }, copyBtn, shareBtn, regenBtn);
    const syncLink = () => { linkInput.disabled = linkRow.hidden = !s.registration_open; };
    syncLink();
    p.watch('registration_open', syncLink);
    // 010: возврат удалённых организатором по ссылке
    const rejoinToggle = h('input', { type: 'checkbox', checked: s.rejoin });
    bindToggle(rejoinToggle, async (on) => {
      await xapi('admin.game', { rejoin: on });
      toast(on ? t('app.admin.invite.rejoin') : t('app.admin.invite.rejoinOff'));
      await again();
    });
    // 008: вступление по одобрению
    const apprToggle = h('input', { type: 'checkbox', checked: s.approval });
    bindToggle(apprToggle, async (on) => {
      await xapi('admin.game', { approval: on });
      toast(on ? t('app.admin.invite.approvalOn') : t('app.admin.invite.approvalOff'));
    });
    const regCard = ui.card(
      ui.toggleRow(t('app.admin.invite.regOpen'), null, regToggle),
      linkInput,
      linkRow,
      ui.toggleRow(t('app.admin.invite.approval'), null, apprToggle, 'ds-divided'),
      ui.toggleRow(t('app.admin.invite.rejoin'), null, rejoinToggle, 'ds-divided'));
    if (!s.draw_done) {
      const partToggle = h('input', { type: 'checkbox', checked: d.me_member });
      bindToggle(partToggle, async (on) => {
        if (!on && !(await confirmBox(t('app.admin.participate.leaveConfirm')))) return false;
        await xapi('admin.participate', { on });
        toast(on ? t('app.admin.participate.on') : t('app.admin.participate.off'));
        await refresh();
      });
      regCard.append(ui.toggleRow(t('app.admin.participate.title'), t('app.admin.participate.hint'), partToggle, 'ds-divided'));
    }
    return [ui.section(t('app.admin.invite.title')), regCard];
  }

  // --- заявки на вступление (008): в идущем эвенте — после приглашения; в завершённом (025) — только «Отклонить»:
  // принять в него нельзя, а человек ждёт ответа (то же, что кнопка бота)
  function adminRequests(p) {
    const { s, d, done, xapi, refresh } = p;
    if (!d.requests.length) return [];
    const reqList = h('div', { class: 'ds-card ds-card--flush' });
    for (const r of d.requests) {
      // Кнопки заявки недоступны, пока идёт решение (024)
      const pick = h('div', { class: 'ds-row ds-row--tight' });
      const decide = (approve) => busy([...pick.children], async () => {
        await xapi('admin.request.decide', { id: r.id, approve });
        toast(approve ? t('app.admin.requests.approved', { name: r.name }) : t('app.admin.requests.rejected'));
        await refresh();
      });
      if (!done) pick.append(h('button', { class: 'ds-icon-btn', title: t('app.admin.requests.approve'), onclick: () => decide(true) }, '✅'));
      pick.append(h('button', { class: 'ds-icon-btn', title: t('app.admin.requests.reject'), onclick: () => decide(false) }, '❌'));
      reqList.append(h('div', { class: 'ds-list-item' },
        h('div', { class: 'ds-grow' },
          personTitle(r),
          ui.hint(fmtTime(r.created_at))),
        pick));
    }
    const hint = ui.hint(done
      ? t('app.admin.requests.finishedHint')
      : t('app.admin.requests.closedHint'), 'ds-hint--inset');
    if (!done) {
      const sync = () => { hint.hidden = !!s.registration_open; };
      sync();
      p.watch('registration_open', sync);
    }
    return [ui.section(t('app.admin.requests.title', { n: d.requests.length })), hint, reqList];
  }

  // 010: удалённые организатором — при запрете возврата им можно точечно разрешить вернуться
  function adminRemoved(p) {
    const { s, d, xapi, again } = p;
    if (s.rejoin || !d.removed.length) return [];
    const remList = h('div', { class: 'ds-card ds-card--flush' });
    for (const r of d.removed) {
      const allow = ui.btn(t('app.admin.removed.allow'), 'secondary', 'ds-btn--sm');
      allow.addEventListener('click', () => busy(allow, async () => {
        await xapi('admin.removed.allow', { id: r.id });
        toast(t('app.admin.removed.allowed', { name: r.name || t('app.admin.removed.noName') }));
        await again();
      }));
      remList.append(h('div', { class: 'ds-list-item' },
        h('div', { class: 'ds-grow' },
          personTitle(r),
          ui.hint(t('app.admin.removed.at', { time: fmtTime(r.removed_at) }))),
        allow));
    }
    return [ui.section(t('app.admin.removed.title', { n: d.removed.length })),
      ui.hint(t('app.admin.removed.hint'), 'ds-hint--inset'),
      remList];
  }

  // Пригласить участников прошлого события
  function adminInviteFrom(p) {
    const { s, d, xapi } = p;
    if (!d.other_events.length) return [];
    const sel = h('select', { class: 'ds-input' }, h('option', { value: '' }, t('app.admin.inviteFrom.pick')), ...d.other_events.map((e) => h('option', { value: e.id }, e.status === 'finished' ? t('app.admin.inviteFrom.finished', { title: e.title }) : e.title)));
    const inv = ui.btn(t('app.admin.inviteFrom.button'), 'secondary', 'ds-btn--block');
    const hint = ui.hint('');
    // Закрытая регистрация — значит никто (010): приглашения не рассылаются. После рассылки кнопка — снова по
    // регистрации (её могли закрыть, пока шла рассылка), а не просто доступна
    const sync = () => {
      inv.disabled = !s.registration_open;
      hint.textContent = s.registration_open
        ? t('app.admin.inviteFrom.hint')
        : t('app.admin.inviteFrom.closedHint');
    };
    inv.addEventListener('click', async () => {
      if (!sel.value) return toast(t('app.admin.inviteFrom.pickFirst'));
      if (!(await confirmBox(t('app.admin.inviteFrom.confirm')))) return;
      busy(inv, async () => { const r = await xapi('admin.event.invite_from', { from: +sel.value }); toast(r.count ? t('app.admin.inviteFrom.sent', { n: r.count }) : t('app.admin.inviteFrom.allHere')); }).then(sync);
    });
    sync();
    p.watch('registration_open', sync);
    return [ui.card(hint, sel, inv)];
  }

  // --- обращения участников (012): без текстов; переписка — в шторке обращения
  function adminSupport(p) {
    const { d, me, evId, again } = p;
    const sup = d.support || { threads: [] };
    const openCount = sup.threads.filter((th) => th.status === 'open').length;
    const list = h('div', { class: 'ds-card ds-card--flush' });
    if (!sup.threads.length) list.append(h('div', { class: 'ds-list-item ds-text-muted' }, t('app.admin.support.empty')));
    for (const th of sup.threads) {
      const chip = th.status !== 'open' ? ui.chip(t('app.admin.support.closed'))
        : !th.assignee ? ui.chip(t('app.admin.support.waiting'), 'warning')
        : th.assignee === me.id ? ui.chip(t('app.admin.support.mine'), 'success') : ui.chip(t('app.admin.support.ledBy', { name: th.assignee_name }));
      list.append(h('div', { class: 'ds-list-item ds-list-item--link', role: 'button', onclick: () => openSupportThread(th.id, evId, again) },
        h('div', { class: 'ds-grow' },
          h('div', { class: 'ds-list-item__title' }, th.name || t('app.core.noName')),
          h('div', { class: 'ds-chips' }, chip),
          ui.hint(fmtTime(th.last_at))),
        th.unread ? h('span', { class: 'ds-badge' }, String(th.unread)) : null,
        h('span', { class: 'ds-list-item__chevron' }, '›')));
    }
    return [ui.section(openCount ? t('app.admin.support.titleCount', { n: openCount }) : t('app.admin.support.title')), list];
  }

  // --- объявление (007); черновик переживает перерисовку панели (009)
  function adminAnnounce(p) {
    const { evId, drafts, xapi } = p;
    const annText = ui.textarea({ placeholder: t('app.admin.announce.placeholder'), rows: 2, maxlength: 2000, value: drafts.announce || '' });
    const annBtn = ui.btn(t('app.admin.announce.send'), 'secondary', 'ds-btn--block');
    const annUpdate = () => { drafts.announce = annText.value; annBtn.disabled = !annText.value.trim(); };
    annText.addEventListener('input', annUpdate);
    annUpdate();
    annBtn.addEventListener('click', async () => {
      if (!annText.value.trim()) return toast(t('app.admin.announce.empty'));
      if (!(await confirmBox(t('app.admin.announce.confirm')))) return;
      const text = annText.value;
      busy(annBtn, async () => {
        const r = await xapi('admin.announce', { text });
        if (annText.value === text) { annText.value = ''; annText.dispatchEvent(new Event('input')); }
        await reloadEvent(evId).catch(() => {});
        toast(t('app.admin.announce.sent', { n: r.count }));
      }).then(annUpdate);
    });
    return [ui.section(t('app.admin.announce.title')),
      ui.card(ui.hint(t('app.admin.announce.hint')), annText, annBtn)];
  }

  // --- меню участника (004)
  const canRemove = (p, u) => !p.done && u.id !== p.me.id && !u.is_owner && (!u.is_admin || p.me.is_owner);
  // Переносит участие только владелец (010): админ мог бы перенести игрока на свой второй аккаунт и увидеть его пары
  // Перенос участия — только владелец бота (051): владельцу компании кнопки нет
  const canTransfer = (p, u) => state.botOwner && u.registered && !u.is_owner;
  // На аккаунт администратора переносить нельзя: он получил бы чужие пары (006)
  const transferTargets = (p, u) => p.users.filter((x) => x.id !== u.id && x.registered && !x.in_draw && !x.is_admin);
  function adminMemberMenu(p, u, chips) {
    const { s, me, xapi, again, refresh } = p;
    const items = [];
    if (u.registered) {
      const b = ui.btn(t('app.admin.member.openLetter'), 'secondary', 'ds-btn--block');
      b.addEventListener('click', async () => {
        if (!(await confirmBox(t('app.admin.member.openLetterConfirm', { name: u.name })))) return;
        busy(b, async () => {
          const w = await xapi('admin.wishlist', { id: u.id });
          openSheet(t('app.admin.member.letterTitle', { name: w.name }), !w.wishlist ? h('div', { class: 'ds-card ds-text-muted' }, t('app.admin.member.letterEmpty'))
            : w.card ? postcard(w.card, h('div', { class: 'ds-letter' }, w.wishlist)) : h('div', { class: 'ds-card ds-letter' }, w.wishlist));
        });
      });
      items.push(b);
      // Переименовывает только владелец бота (063): имя одно во всех событиях; себя — в «Настройках»
      if (state.botOwner) {
        const r = ui.btn(t('app.admin.member.rename'), 'secondary', 'ds-btn--block');
        r.addEventListener('click', () => {
          const inp = h('input', { class: 'ds-input', type: 'text', value: u.name, maxlength: 40 });
          const ok = ui.btn(t('app.core.save'), '', 'ds-btn--block');
          ok.addEventListener('click', () => busy(ok, async () => { await api('admin.rename', { id: u.id, name: inp.value }); closeSheet(); toast(t('app.admin.member.renamed')); await refresh(); }));
          openSheet(t('app.admin.member.nameTitle'), [ui.hint(t('app.admin.member.nameHint')), clearable(inp, t('app.admin.member.nameTitle')), ok]);
        });
        items.push(r);
      }
    }
    if (canTransfer(p, u) && transferTargets(p, u).length) {
      const b = ui.btn(t('app.admin.member.transfer'), 'secondary', 'ds-btn--block');
      b.addEventListener('click', () => adminTransfer(p, u));
      items.push(b);
    }
    if (canRemove(p, u)) {
      const b = ui.btn(t('app.admin.member.remove'), 'danger', 'ds-btn--block');
      b.addEventListener('click', async () => {
        const tail = s.draw_done && u.in_draw ? t('app.admin.member.removeRedraw') : '';
        if (!(await confirmBox(t('app.admin.member.removeConfirm', { name: u.name || t('app.core.noName'), title: s.title, tail })))) return;
        busy(b, async () => { await xapi('admin.remove', { id: u.id }); closeSheet(); toast(t('app.admin.member.removed')); await again(); });
      });
      items.push(b);
    }
    // Переносит участие только владелец бота (051, 063)
    if (u.is_owner) items.push(ui.hint(t('app.admin.member.ownerHint')));
    else if (!me.is_owner) items.push(ui.hint(t(u.is_admin ? 'app.admin.member.adminHint' : 'app.admin.member.transferBotOwner', { owner: ownerWord() })));
    else if (!state.botOwner && u.registered) items.push(ui.hint(t('app.admin.member.transferBotOwner')));
    openSheet(u.name || t('app.core.noName'), [
      h('div', {}, u.username ? ui.hint('@' + u.username) : null, h('div', { class: 'ds-chips' }, chips)),
      ...(items.length ? items : [ui.hint(t('app.admin.member.noActions'))]),
    ]);
  }
  function adminTransfer(p, u) {
    const targets = transferTargets(p, u);
    const sel = h('select', { class: 'ds-input' }, h('option', { value: '' }, t('app.admin.transfer.pick')),
      ...targets.map((x) => h('option', { value: x.id }, x.name + (x.username ? ' (@' + x.username + ')' : ''))));
    const go = ui.btn(t('app.admin.transfer.go'), '', 'ds-btn--block');
    go.addEventListener('click', async () => {
      const to = targets.find((x) => String(x.id) === sel.value);
      if (!to) return toast(t('app.admin.transfer.pickFirst'));
      if (!(await confirmBox(t('app.admin.transfer.confirm', { name: u.name, to: to.name })))) return;
      busy(go, async () => { await api('admin.transfer', { from: u.id, to: to.id }); closeSheet(); toast(t('app.admin.transfer.done')); await p.refresh(); });
    });
    openSheet(t('app.admin.transfer.title'), [
      ui.hint(t('app.admin.transfer.hint', { name: u.name || t('app.admin.transfer.noName') })),
      sel, go]);
  }

  // --- участники
  function adminMembers(p) {
    const { s, users, players, me, done, xapi, again } = p;
    const filled = players.filter((u) => u.has_wishlist).length;
    const list = h('div', { class: 'ds-card ds-card--flush' });
    if (!users.length) list.append(h('div', { class: 'ds-list-item ds-text-muted' }, t('app.event.people.empty')));
    for (const u of users) {
      const chips = [];
      if (!u.registered) chips.push(ui.chip(t('app.admin.members.registering')));
      else {
        chips.push(u.has_wishlist ? ui.chip(t('app.admin.members.letter'), 'success') : ui.chip(t('app.admin.members.noLetter'), 'warning'));
        if (s.draw_done && !u.in_draw) chips.push(ui.chip(t('app.admin.members.notInDraw'), 'warning'));
        if (u.ward_seen === false) chips.push(unseenChip());
      }
      if (u.blocked) chips.push(ui.chip(t('app.admin.members.blocked'), 'warning'));
      if (u.is_owner) chips.push(ui.chip(t('app.core.role.owner')));
      else if (u.is_admin) chips.push(ui.chip(t('app.core.role.admin')));
      const actions = h('div', { class: 'ds-row ds-row--tight' },
        h('button', { class: 'ds-icon-btn', title: t('app.admin.members.actions'), onclick: () => adminMemberMenu(p, u, chips.map((c) => c.cloneNode(true))) }, '⋯'));
      list.append(h('div', { class: 'ds-list-item' },
        h('div', { class: 'ds-grow' },
          personTitle(u),
          h('div', { class: 'ds-chips' }, chips)),
        actions));
      // Вписывает только владелец (012): админ по отметке «не открыл» понял бы, кто дарит новичку
      if (!done && s.draw_done && u.registered && !u.in_draw) {
        if (me.is_owner) {
          const ins = ui.btn(t('app.admin.members.insert'), '', 'ds-btn--sm');
          ins.addEventListener('click', async () => {
            if (!(await confirmBox(t('app.admin.members.insertConfirm', { name: u.name })))) return;
            busy(ins, async () => { await xapi('admin.insert', { id: u.id }); toast(t('app.admin.members.inserted')); await again(); });
          });
          list.append(h('div', { class: 'ds-list-item ds-list-item--sub' }, ins));
        } else {
          list.append(h('div', { class: 'ds-list-item ds-list-item--sub' }, ui.hint(t('app.admin.members.insertOwnerOnly', { owner: ownerWord() }))));
        }
      }
    }
    const nodes = [
      ui.section(t('app.admin.members.title')),
      ui.hint(t('app.admin.members.letters', { filled, total: players.length }), 'ds-hint--inset'),
      list];
    if (!done && players.length > filled) {
      const remindBtn = ui.btn(t('app.admin.members.remind'), 'secondary', 'ds-btn--block');
      remindBtn.addEventListener('click', () => busy(remindBtn, async () => {
        const r = await xapi('admin.remind.unfilled');
        toast(r.count ? t('app.admin.remindSent', { n: r.count }) : t('app.admin.members.allWrote'));
      }));
      nodes.push(remindBtn);
    }
    return nodes;
  }

  // --- владелец и администраторы (063; до 063 — на главном экране комнаты): владельцу — назначить из участников, снять

  // --- правила пар (026): исключения и закреплённые пары устроены одинаково — список «кто — кому» (взаимные —
  // одной строкой, ✕ — убрать) и форма «Кто дарит… / кому… / и наоборот»; разные — тексты, запросы и что можно менять
  const nameOf = (p, id) => p.byId[id]?.name || '?';
  function rulePairs(rows) {
    const seen = new Set();
    const pairs = [];
    for (const e of rows) {
      const key = e.giver + ':' + e.receiver;
      if (seen.has(key)) continue;
      const mutual = rows.some((x) => x.giver === e.receiver && x.receiver === e.giver);
      seen.add(key);
      if (mutual) seen.add(e.receiver + ':' + e.giver);
      pairs.push({ ...e, mutual });
    }
    return pairs;
  }
  // Строка пары (060): текст словаря key, {giver} и {receiver} — имена жирным. Подстановки t() — строки, поэтому на их
  // место ставятся метки, а по меткам текст режется на части между узлами
  function rulePairText(p, key, pair) {
    const names = { giver: nameOf(p, pair.giver), receiver: nameOf(p, pair.receiver) };
    return t(key, { giver: '\u0001giver\u0001', receiver: '\u0001receiver\u0001' }).split(/\u0001(giver|receiver)\u0001/)
      .map((part, i) => (i % 2 ? h('b', {}, names[part]) : part)).filter((part) => part !== '');
  }
  // both / one — ключи словаря строки взаимной и односторонней пары; del — кнопка ✕ { title, run(pair) }, без неё
  // список только для чтения
  function ruleList(p, pairs, { empty, both, one, del }) {
    const list = h('div', { class: 'ds-card ds-card--flush' });
    if (!pairs.length) list.append(h('div', { class: 'ds-list-item ds-text-muted' }, empty));
    for (const pair of pairs) {
      list.append(h('div', { class: 'ds-list-item' },
        h('div', { class: 'ds-grow' }, rulePairText(p, pair.mutual ? both : one, pair)),
        del ? h('button', { class: 'ds-icon-btn', title: del.title, onclick: (ev) => busy(ev.currentTarget, async () => { await del.run(pair); await p.again(); }) }, '✕') : null));
    }
    return list;
  }
  // Форма: «Кто дарит…» и to, флажок «и наоборот» (mutual — включён ли сразу), кнопка label; add(кто, кому, взаимно)
  function ruleForm(p, { to, label, mutual: on, add }) {
    const opts = (ph) => [h('option', { value: '' }, ph), ...p.players.map((u) => h('option', { value: u.id }, u.name))];
    const selA = h('select', { class: 'ds-input' }, opts(t('app.admin.rules.giver')));
    const selB = h('select', { class: 'ds-input' }, opts(to));
    const mutual = h('input', { type: 'checkbox', checked: on });
    const btn = ui.btn(label, 'secondary', 'ds-btn--block');
    btn.addEventListener('click', () => busy(btn, async () => {
      if (!selA.value || !selB.value) return toast(t('app.admin.rules.pickTwo'));
      if (selA.value === selB.value) return toast(t('app.admin.rules.different'));
      await add(+selA.value, +selB.value, mutual.checked);
      await p.again();
    }));
    return { selA, selB, mutualRow: h('label', { class: 'ds-check' }, mutual, t('app.admin.rules.mutual')), btn };
  }

  // --- исключения
  function adminExclusions(p) {
    const { s, d, me, done, xapi } = p;
    if (done) return [];
    // 009: после жеребьёвки исключения только для чтения — по заменам можно было бы вычислить пары.
    // Владелец пары видит и так — для него заморозки нет (010)
    const exFrozen = s.draw_done && !me.is_owner;

    });
    const exForm = exFrozen ? null : ruleForm(p, {
      to: t('app.admin.exclusions.to'), label: t('app.admin.exclusions.add'), mutual: true,
      add: async (giver, receiver, mutual) => {

        toast(t('app.admin.exclusions.added'));
      },
    });
    return [
      ui.section(t('app.admin.exclusions.title')),
      exList,
      exFrozen ? null : ui.card(

  // --- закреплённые пары (014): только владелец — админу поле fixed не приходит. Те же помощники, что у исключений.
  function adminFixed(p) {
    const { s, d, done, xapi } = p;
    // После жеребьёвки блок нужен, только если пары были закреплены (показывается для чтения)
    if (done || !d.fixed || (s.draw_done && !d.fixed.length)) return [];
    const fxPairs = rulePairs(d.fixed);
    const nodes = [ui.section(t('app.admin.fixed.title')), ruleList(p, fxPairs, {
      empty: t('app.admin.fixed.empty'), both: 'app.admin.fixed.both', one: 'app.admin.fixed.one',
      del: s.draw_done ? null : {
        title: t('app.admin.fixed.del'),
        run: async (f) => { await xapi('admin.fixed.del', f); toast(t('app.admin.fixed.removed')); },
      },
    })];
    if (s.draw_done) {
      nodes.push(ui.hint(t('app.admin.fixed.frozen'), 'ds-hint--inset'));
      return nodes;
    }
    const fx = ruleForm(p, {
      to: t('app.admin.fixed.to'), label: t('app.admin.fixed.add'), mutual: false,
      add: async (giver, receiver, mutual) => {
        await xapi('admin.fixed.add', { giver, receiver, mutual: !s.ring && mutual });
        toast(t('app.admin.fixed.added'));
      },
    });
    const fxForm = ui.card();
    // При кольцевой — только в одну сторону: флажка «и наоборот» нет, взаимные пары — предупреждение. Кольцевую
    // переключают в «Жеребьёвке» — форма перестраивается по p.watch
    const sync = () => {
      const clash = s.ring ? fxPairs.filter((f) => f.mutual) : [];
      // replaceChildren, в отличие от h(), вывел бы null текстом — пустые места отбрасываются
      fxForm.replaceChildren(...[
        ui.hint(t('app.admin.fixed.hint', { ring: s.ring ? t('app.admin.fixed.hintRing') : '' })),
        clash.length ? h('div', { class: 'ds-notice ds-notice--error' }, t('app.admin.fixed.clash', {
          pairs: clash.map((f) => t('app.admin.fixed.clashPair', { giver: nameOf(p, f.giver), receiver: nameOf(p, f.receiver) })).join('; '),
        })) : null,
        fx.selA, fx.selB,
        s.ring ? null : fx.mutualRow,
        fx.btn,
      ].filter(Boolean));
    };
    sync();
    p.watch('ring', sync);
    nodes.push(fxForm);
    return nodes;
  }

  // --- жеребьёвка
  function adminDraw(p) {
    const { s, d, me, players, done, evId, xapi, refresh } = p;
    const drawCard = ui.card();
    if (!s.draw_done) {
      const ring = h('input', { type: 'checkbox', checked: s.ring });
      bindToggle(ring, async (on) => {
        await xapi('admin.ring', { on });
        toast(on ? t('app.admin.draw.ringOn') : t('app.admin.draw.ringOff'));
        p.set('ring', on); // 014: флажок «и наоборот» и предупреждение закреплённых пар зависят от кольцевой
        await reloadEvent(evId).catch(() => {});
      });
      drawCard.append(ui.toggleRow(t('app.admin.draw.ring'),
        t('app.admin.draw.ringHint'), ring));
      if (d.other_events.length) {
        const av = h('input', { type: 'checkbox', checked: s.avoid_previous });
        bindToggle(av, async (on) => { await xapi('admin.avoid_previous', { on }); });
        drawCard.append(h('label', { class: 'ds-check' }, av, t('app.admin.draw.avoidPrevious')));
      }
      const drawBtn = ui.btn(t('app.admin.draw.run'), 'festive', 'ds-btn--block');
      const result = h('div');
      drawBtn.addEventListener('click', () => busy(drawBtn, async () => {
        const c = await xapi('admin.draw.check');
        result.replaceChildren();
        if (!c.possible) {
          haptic('error');
          result.append(h('div', { class: 'ds-notice ds-notice--error' }, h('b', {}, t('app.admin.draw.impossible')), h('ul', { class: 'ds-bullets' }, c.problems.map((x) => h('li', {}, x)))));
          return;
        }
        // Окно подтверждения Telegram — до 250 символов (009): важное сначала, вместо списка имён — число
        const k = c.empty_wishlists.length;
        let text = t('app.admin.draw.confirm', { n: c.count, ring: c.ring ? t('app.admin.draw.ringMark') : '' });
        if (k) text += `\n\n${t('app.admin.draw.confirmNoLetters', { n: k })}`;
        if (!(await confirmBox(text))) return;
        const r = await xapi('admin.draw');
        haptic('success');
        toast(t('app.admin.draw.doneToast', { n: r.count }));
        await refresh();
      }));
      drawCard.append(
        ui.hint(t('app.admin.draw.hint', { tail: me.is_owner ? t('app.admin.draw.hintOwner') : t('app.admin.draw.hintAdmin') })),
        drawBtn, result);
    } else {
      drawCard.append(
        h('div', {}, h('div', { class: 'ds-text-strong' }, t('app.admin.draw.doneTitle', { ring: s.ring ? t('app.admin.draw.ringMark') : '' })), s.draw_at ? ui.hint(fmtTime(s.draw_at)) : null));
      if (s.draw_cancelable) {
        // 007: отменить можно, пока никто не открыл подопечного и переписки нет
        const cancel = ui.btn(t('app.admin.draw.cancel'), 'secondary', 'ds-btn--block');
        cancel.addEventListener('click', async () => {
          if (!(await confirmBox(t('app.admin.draw.cancelConfirm')))) return;
          busy(cancel, async () => { await xapi('admin.draw.cancel'); toast(t('app.admin.draw.canceled')); await refresh(); });
        });
        drawCard.append(ui.hint(t('app.admin.draw.cancelHint')), cancel);
      }
    }
    return [ui.section(t('app.admin.draw.title')), drawCard];
  }

  // --- параметры эвента
  function adminParams(p) {
    const { s, done, drafts, xapi, refresh, live } = p;
    if (done) return [];
    // Черновик параметров (009): несохранённые правки переживают перерисовку панели после других действий
    const g = drafts.game || {};
    const val = (k) => (k in g ? g[k] : s[k]);
    const f = {
      title: h('input', { class: 'ds-input', type: 'text', value: val('title'), maxlength: 80 }),
      budget: h('input', { class: 'ds-input', type: 'text', value: val('budget'), placeholder: t('app.admin.params.budgetPlaceholder'), maxlength: 100 }),
      event_date: ui.date(val('event_date')),
      place: h('input', { class: 'ds-input', type: 'text', value: val('place'), placeholder: t('app.admin.params.placePlaceholder'), maxlength: 200 }),
      deadline: ui.date(val('deadline')),
      notes: ui.textarea({ value: val('notes'), placeholder: t('app.admin.params.notesPlaceholder'), rows: 2, maxlength: 2000 }),
    };
    // Расписание (007): автожеребьёвка (до жеребьёвки) и авто-раскрытие (064: завершение — само, на 3-й день после даты
    // вручения). У автожеребьёвки свой календарь
    // (027): системный сам выделяет сегодня и подставляет его по «Готово» — жеребьёвка включалась бы случайно
    if (!s.draw_done) {
      f.draw_auto_date = ui.datePick(val('draw_auto_date'), {
        title: t('app.admin.params.autoDraw'), hint: t('app.admin.params.autoDrawHint') });
    }
    const checks = {
      letter_drafts: h('input', { type: 'checkbox', checked: val('letter_drafts') }),
      reveal_auto: h('input', { type: 'checkbox', checked: val('reveal_auto') }),
    };
    const saveGame = ui.btn(t('app.core.save'), '', 'ds-btn--block');
    // Авто-раскрытие считается от даты вручения: без неё не сработает (009)
    const noDate = ui.hint(t('app.admin.params.noDate'));
    const values = () => ({
      ...Object.fromEntries(Object.entries(f).map(([k, el]) => [k, el.value])),
      ...Object.fromEntries(Object.entries(checks).map(([k, el]) => [k, el.checked])),
    });
    const onEdit = () => {
      drafts.game = values();
      noDate.hidden = !(checks.reveal_auto.checked && !f.event_date.value);
    };
    for (const el of [...Object.values(f), ...Object.values(checks)]) {
      el.addEventListener('input', onEdit);
      el.addEventListener('change', onEdit);
    }
    noDate.hidden = !(checks.reveal_auto.checked && !f.event_date.value);
    saveGame.addEventListener('click', () => busy(saveGame, async () => {
      await xapi('admin.game', values());
      delete drafts.game;
      // Панель — заново по сохранённым данным (026): новое название во всех её текстах (подтверждения, «Отправить в
      // чат», служебный просмотр); тихо — без загрузчика, экран остаётся у «Параметров»
      await refresh({ quiet: true });
      if (!live()) return;
      haptic('success');
      toast(t('app.core.saved'));
    }));
    return [
      ui.section(t('app.admin.params.title')),
      ui.card(
        field(t('app.admin.params.name'), f.title),
        field(t('app.admin.params.budget'), f.budget),
        field(t('app.admin.params.eventDate'), f.event_date),
        field(t('app.admin.params.place'), f.place),
        field(t('app.admin.params.deadline'), f.deadline),
        field(t('app.admin.params.notes'), f.notes),
        h('label', { class: 'ds-check' }, checks.letter_drafts, t('app.admin.params.letterDrafts')),
        f.draw_auto_date ? field(t('app.admin.params.autoDraw'), f.draw_auto_date) : null,
        h('label', { class: 'ds-check' }, checks.reveal_auto, t('app.admin.params.revealAuto')),
        noDate,
        saveGame),
    ];
  }
