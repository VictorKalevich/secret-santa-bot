
  // ---------- состояние ----------
  // view: 'home' (мои события, 063), 'settings' («Настройки», 063) или 'event' (вкладки одного эвента).
  // botOwner — владелец бота (me.is_owner; в событии state.me.is_owner — «владелец этого события», enterEvent); open —
  // открытый бот; create — можно ли создавать события (063: { bot } владельцу бота, { left, limit, next } по лимиту).
  // nav — номер экрана (009): растёт при каждом переходе; ответ сервера для прежнего экрана не рисуется.
  // drafts[evId] — несохранённые объявление и параметры эвента (переживают перерисовку панели).
  // dirty — несохранённое на «Письме» (вопрос Telegram при закрытии, точка на вкладке): true — адрес или письмо без
  // черновиков, перед уходом спросить; 'draft' — только черновик письма, при уходе он сохранится сам (058).
  // wardCache — последний ответ ward.get: после закрытия переписки «Подопечный» перерисовывается без нового запроса,
  // иначе подсветка изменений письма пропала бы (сервер считает письмо просмотренным).
  const state = {
    me: null, events: [], limits: { chars: 4000 }, view: 'home', nav: 0, drafts: {}, wardCache: null,
    evId: null, ev: null, member: null, can: {}, tab: 'my', dirty: false, wardNameShown: false,
    announcements: [], // 007: объявления события (вкладка «Инфо»); все поля состояния — здесь (026)
    support: null, supportAdmin: null, // 012: обращения — участнику его статус, админу — ничьи и непрочитанные
    roster: null, rosterOpen: null, // 037: участники на «Инфо»; rosterOpen — id события, где список раскрыт
    looks: { postcards: [] }, // набор открыток — с сервера (me)
    botOwner: false, open: false, create: null,
    lang: 'ru', textsVersion: '', // 060: язык и версия словаря приложения (ответ me)
    langs: { choice: '', auto: 'ru', list: [] }, // 061: выбор языка ('' — как в Telegram), язык по Telegram, языки бота
  };
  const main = document.getElementById('main');
  const heroTitle = document.getElementById('heroTitle');
  const heroSub = document.getElementById('heroSub');
  const heroBack = document.getElementById('heroBack');
  const heroAction = document.getElementById('heroAction');
  // Уход с события или вкладки (058): черновик письма — сохранить сразу, спросить — только о том, что само не сохранится
  async function mayLeave() {
    if (saveLetterNow) saveLetterNow();
    if (state.dirty === true && !(await confirmBox(t('app.home.unsaved')))) return false;
    if (state.dirty) setDirty(false);
    return true;
  }
  async function leaveEvent() {
    if (!(await mayLeave())) return;
    await goHome();
  }
  // «‹» и системная «Назад» (063): из события и «Настроек» — к списку событий
  function goBack() {
    if (state.view === 'event') return leaveEvent();
    if (state.view === 'settings') return goHome();
  }
  heroBack.addEventListener('click', goBack);
  heroAction.addEventListener('click', () => goSettings());

  // Запросы в пределах открытого эвента
  const eapi = (action, body) => api(action, { ...(body || {}), event: state.evId });
  const finished = () => state.ev && state.ev.status === 'finished';

  function setDirty(v) {
    state.dirty = v;
    try { if (tg && ver('6.2')) v ? tg.enableClosingConfirmation() : tg.disableClosingConfirmation(); } catch (e) {}
    if (state.view === 'event') renderTabs();
  }

  async function reloadMe() {
    // Словарь — только если у приложения его нет или он другой (060); до любого построения экрана
    const d = await api('me', { texts: state.textsVersion || undefined });
    setTexts(d);
    state.langs = { choice: d.lang_choice || '', auto: d.lang_auto || 'ru', list: d.languages || [] };
    if (!d.registered) return d;
    state.me = d.me;
    state.botOwner = !!d.me.is_owner;
    state.open = !!d.open;
    state.create = d.create || null;
    state.events = d.events;
    state.limits = d.limits;
    state.looks = d.looks;
    if (state.ev) enterEvent(state.ev.role);
    return d;
  }

  // Права в открытом событии (063): роль из event.get пишется в state.me.is_admin / is_owner — остальные части читают их
  // как раньше; владелец бота — права владельца в любом событии
  function enterEvent(role) {
    state.me.is_admin = state.botOwner || role === 'owner' || role === 'admin';
    state.me.is_owner = state.botOwner || role === 'owner';
  }
  // Владелец бота в чужом событии — гость (063): плашка над «Инфо» и «Управлением»
  const guestNotice = (on) => (on ? h('div', { class: 'ds-notice ds-text-sm' }, t('app.home.guest')) : null);

  // Данные эвента evId; если пользователь уже в другом эвенте — ничего не меняет (009).
  // Участие обновляется на месте: открытые экраны держат ссылку на state.member и видят свежие данные.
  async function reloadEvent(evId = state.evId) {
    const d = await api('event.get', { event: evId });
    if (evId !== state.evId) return null;
    if (state.member && d.member && state.ev && state.ev.id === d.event.id) Object.assign(state.member, d.member);
    else state.member = d.member;
    state.ev = d.event;
    state.can = d.can;
    state.announcements = d.announcements || [];
    state.support = d.support || null;
    state.supportAdmin = d.support_admin || null;
    state.roster = d.roster || null;
    enterEvent(d.event.role);
    return d;
  }

  function showLoader() { main.replaceChildren(h('div', { class: 'ds-loader' }, h('div', { class: 'ds-spinner' }), t('app.boot.loading'))); }
  // retry — повтор именно неудавшейся загрузки (009); на экране эвента — ещё выход к списку
  function showError(e, retry) {
    const again = h('button', { class: 'ds-btn ds-btn--secondary ds-empty__action', onclick: () => (retry || render)() }, t('app.boot.retry'));
    const home = state.view === 'event' && state.me
      ? h('button', { class: 'ds-btn ds-btn--secondary ds-empty__action', onclick: () => { setDirty(false); goHome(); } }, t('app.home.toEvents'))
      : null;
    main.replaceChildren(ui.empty('⚠️', t('app.boot.loadFailed'), e.message, [again, home]));
  }
  // Асинхронный экран («Подопечный», «Управление»; 026): отказ при построении — экран ошибки с повтором, а не вечный
  // загрузчик, откуда бы экран ни вызвали (render, закрытие шторки, перестройка панели); для покинутого экрана — ничего
  function screen(build) {
    return async (...args) => {
      const nav = state.nav;
      try { return await build(...args); }
      catch (e) { if (nav === state.nav) showError(e); }
    };
  }

  function render() {
    state.nav++;
    window.scrollTo(0, 0);
    // Нижняя панель и ⚙️ — только у главного экрана (063): его render ставит их снова
    main.classList.remove('ds-page--bar');
    heroAction.hidden = true;
    try {
      if (state.view === 'home') renderHome();
      else if (state.view === 'settings') renderSettings();
      else {
        renderHero();
        renderTabs();
        ({ my: renderMy, ward: renderWard, info: renderInfo, admin: renderAdmin }[state.tab] || renderInfo)();
      }
    } catch (e) {
      showError(e); // ошибка вкладки — экран ошибки, а не прежний экран или загрузчик (026)
    }
    syncBack();
  }

  // ---------- главный экран: мои события (005, 063) ----------
  async function goHome() {
    const nav = ++state.nav;
    state.view = 'home';
    state.evId = null;
    state.ev = null;
    state.wardCache = null;
    try { history.replaceState(null, '', location.pathname + location.hash); } catch (e) {}
    syncBack();
    showLoader();
    try { await reloadMe(); } catch (e) { if (nav === state.nav) showError(e, goHome); return; }
    if (nav === state.nav) render();
  }

  async function openEvent(id, tab, chat, support) {
    const nav = ++state.nav;
    state.view = 'event';
    state.evId = id;
    state.member = null;
    state.wardCache = null;
    state.wardNameShown = false;
    state.rosterOpen = null;
    syncBack();
    showLoader();
    try { await reloadEvent(id); } catch (e) { if (nav === state.nav) showError(e, () => openEvent(id, tab, chat, support)); return; }
    if (nav !== state.nav) return;
    const m = state.member;
    // Админу «Подопечный» по умолчанию не открывается: просмотр подопечного запрещает отмену жеребьёвки (009)
    // 012: из кнопок бота «💬 Открыть» — обращение участника (support=my) или обращение в панели админа (support=<id>)
    const supportThread = state.me.is_admin && /^\d+$/.test(support || '') ? Number(support) : 0;
    if (support === 'my' && m) tab = 'info';
    if (supportThread) tab = 'admin';
    if (tab === 'my' && !m) tab = 'info';
    state.tab = tab || (state.me.is_admin ? (m ? 'my' : 'admin')
      : m ? (state.ev.draw_done && m.has_ward && m.wishlist.trim() ? 'ward' : 'my') : 'info');
    render();
    if (chat === 'santa' && m && m.has_santa) openChat('santa');
    if (chat === 'ward' && m && m.has_ward) openChat('ward');
    if (support === 'my' && m) openSupport();
    if (supportThread) openSupportThread(supportThread, id, () => (state.evId === id && state.tab === 'admin' ? render() : undefined));
  }

  // Системная кнопка «Назад» Telegram (009): закрывает шторку, иначе — как «‹» (063: из события и «Настроек» — к списку)
  function syncBack() {
    try {
      if (!tg || !ver('6.1')) return;
      if (sheetEl || ((state.view === 'event' || state.view === 'settings') && !heroBack.hidden)) tg.BackButton.show();
      else tg.BackButton.hide();
    } catch (e) {}
  }
  try { if (tg && ver('6.1')) tg.BackButton.onClick(() => (sheetEl ? requestCloseSheet() : goBack())); } catch (e) {}

  // Списки главного экрана (063): «Идут» — по дате вручения, без даты — в конце, при равных — новые первыми;
  // «Завершённые» — недавно завершённые первыми
  function homeLists(events) {
    const newer = (a, b) => b.id - a.id;
    const byDate = (a, b) => (a.event_date && b.event_date ? a.event_date.localeCompare(b.event_date)
      : a.event_date ? -1 : b.event_date ? 1 : 0) || newer(a, b);
    return {
      active: events.filter((e) => e.status !== 'finished').sort(byDate),
      finished: events.filter((e) => e.status === 'finished').sort((a, b) => (b.finished_at || 0) - (a.finished_at || 0) || newer(a, b)),
    };
  }

  function eventRow(e) {
    const chips = [];
    if (e.role === 'owner' || e.role === 'admin') chips.push(ui.chip(t('app.core.role.' + e.role)));
    if (e.event_date) chips.push(ui.chip('🎁 ' + fmtDate(e.event_date)));
    if (!e.member) chips.push(ui.chip(t('app.home.notMember')));
    return h('div', { class: 'ds-list-item ds-list-item--link', role: 'button', onclick: () => openEvent(e.id) },
      h('div', { class: 'ds-grow' }, h('div', { class: 'ds-list-item__title' }, e.title), chips.length ? h('div', { class: 'ds-chips' }, chips) : null),
      e.unread ? h('span', { class: 'ds-badge' }, String(e.unread)) : null,
      h('span', { class: 'ds-list-item__chevron' }, '›'));
  }

  function renderHome() {
    heroBack.hidden = true;
    heroAction.hidden = false;
    heroTitle.textContent = t('app.boot.title');
    heroSub.textContent = t('app.home.greet', { name: state.me.name });
    document.getElementById('tabs').hidden = true;
    const { active, finished } = homeLists(state.events);
    const nodes = [];
    if (active.length) nodes.push(ui.section(t('app.home.going')), h('div', { class: 'ds-card ds-card--flush' }, active.map(eventRow)));
    if (finished.length) {
      nodes.push(h('details', { class: 'ds-details ds-stack--sm' },
        h('summary', { class: 'ds-details__summary ds-section-title' }, t('app.home.done', { n: finished.length })),
        h('div', { class: 'ds-card ds-card--flush' }, finished.map(eventRow))));
    }
    if (!active.length && !finished.length) nodes.push(ui.hint(t(state.create ? 'app.home.empty.create' : 'app.home.empty.join'), 'ds-hint--inset'));
    // «Создать событие» — закреплена внизу (063), только тем, кто может создавать
    if (state.create) {
      const create = ui.btn(t('app.home.create.button'), '', 'ds-btn--block');
      create.addEventListener('click', openCreateSheet);
      nodes.push(h('div', { class: 'ds-bottom-bar' }, create));
      main.classList.add('ds-page--bar');
    }
    main.replaceChildren(...nodes);
  }

  // Шторка «Новое событие» (063): название, образец (события, где он владелец или администратор), «Я тоже участвую»;
  // лимит — подсказкой, при пределе кнопка недоступна. После создания — новое событие на «Управлении»
  function openCreateSheet() {
    const limit = state.create;
    const title = h('input', { class: 'ds-input', type: 'text', maxlength: 80, placeholder: t('app.home.create.placeholder', { year: new Date().getFullYear() + 1 }) });
    // Образец (007): параметры и исключения прошлого эвента, по желанию — приглашение его участников
    const templates = state.events.filter((e) => e.role === 'owner' || e.role === 'admin');
    const from = h('select', { class: 'ds-input' }, h('option', { value: '' }, t('app.home.create.noTemplate')),
      ...templates.map((e) => h('option', { value: e.id }, t('app.home.create.fromTemplate', { title: e.title }))));
    const invite = h('input', { type: 'checkbox', checked: true, disabled: true });
    from.addEventListener('change', () => { invite.disabled = !from.value; });
    const participate = h('input', { type: 'checkbox', checked: true });
    const ok = ui.btn(t('app.core.create'), '', 'ds-btn--block');
    ok.disabled = !limit.bot && limit.left === 0;
    ok.addEventListener('click', () => busy(ok, async () => {
      if (!title.value.trim()) return toast(t('app.core.needTitle'));
      const r = await api('event.create', {
        title: title.value, from: from.value ? +from.value : undefined, invite: !!from.value && invite.checked, participate: participate.checked,
      });
      closeSheet();
      toast(r.invited ? t('app.home.create.doneInvited', { count: r.invited })
        : r.invite_error ? t('app.home.create.doneInviteError', { error: r.invite_error }) : t('app.home.create.done'));
      await reloadMe();
      await openEvent(r.event.id, 'admin');
    }));
    const sheet = [ui.hint(t('app.home.create.hint')), clearable(title, t('app.home.create.name'))];
    if (templates.length) {
      sheet.push(from, h('label', { class: 'ds-check' }, invite, t('app.home.create.invite')), ui.hint(t('app.home.create.templateHint')));
    }
    sheet.push(h('label', { class: 'ds-check' }, participate, t('app.home.create.participate')));
    if (!limit.bot) {
      const next = new Date(limit.next || 0);
      const iso = isoDay(next.getFullYear(), next.getMonth() + 1, next.getDate());
      sheet.push(ui.hint(limit.left ? t('app.home.create.left', { n: limit.left })
        : t('app.home.create.full', { limit: limit.limit, date: t('app.core.dateYear', { date: fmtDate(iso), year: next.getFullYear() }) })));
    }
    openSheet(t('app.home.create.title'), [...sheet, ok]);
  }
