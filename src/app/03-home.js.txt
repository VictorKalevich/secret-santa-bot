
  // ---------- состояние ----------
  // view: 'companies' (список компаний, 051), 'home' (события одной компании) или 'event' (вкладки одного эвента).
  // companies — видимые компании из me; companyId, company — компания открытого экрана (enterCompany); botOwner — владелец
  // бота (me.is_owner до входа в компанию: в компании state.me.is_owner — «владелец этой компании»); open — открытый бот,
  // create — можно ли создать компанию (052: { bot } владельцу бота, { left } человеку открытого бота).
  // nav — номер экрана (009): растёт при каждом переходе; ответ сервера для прежнего экрана не рисуется.
  // drafts[evId] — несохранённые объявление и параметры эвента (переживают перерисовку панели).
  // dirty — несохранённое на «Письме» (вопрос Telegram при закрытии, точка на вкладке): true — адрес или письмо без
  // черновиков, перед уходом спросить; 'draft' — только черновик письма, при уходе он сохранится сам (058).
  // wardCache — последний ответ ward.get: после закрытия переписки «Подопечный» перерисовывается без нового запроса,
  // иначе подсветка изменений письма пропала бы (сервер считает письмо просмотренным).
  const state = {
    me: null, events: [], limits: { chars: 4000 }, view: 'home', nav: 0, drafts: {}, wardCache: null,
    evId: null, ev: null, member: null, can: {}, tab: 'my', dirty: false, wardNameShown: false,
    announcements: [], // 007: объявления события (вкладка «Инфо»); все поля состояния — здесь (026)
    support: null, supportAdmin: null, // 012: обращения — участнику его статус, админу — ничьи и непрочитанные
    roster: null, rosterOpen: null, // 037: участники на «Инфо»; rosterOpen — id события, где список раскрыт
    looks: { postcards: [], masks: [] }, // наборы открыток и масок — с сервера (me)
    companies: [], companyId: null, company: null, botOwner: false, open: false, create: null,
    lang: 'ru', textsVersion: '', // 060: язык и версия словаря приложения (ответ me)
  };
  const main = document.getElementById('main');
  const heroTitle = document.getElementById('heroTitle');
  const heroSub = document.getElementById('heroSub');
  const heroBack = document.getElementById('heroBack');
  // Уход с события или вкладки (058): черновик письма — сохранить сразу, спросить — только о том, что само не сохранится
  async function mayLeave() {
    if (saveLetterNow) saveLetterNow();
    if (state.dirty === true && !(await confirmBox(t('app.home.unsaved')))) return false;
    if (state.dirty) setDirty(false);
    return true;
  }
  async function leaveEvent() {
    if (!(await mayLeave())) return;
    await goHome();
  }
  // «‹» и системная «Назад» (051): из события — на главный экран его компании, оттуда — к «Компаниям», если список есть
  // (withList, 052)
  function goBack() {
    if (state.view === 'event') return leaveEvent();
    if (state.view === 'home' && withList()) return goCompanies();
  }
  heroBack.addEventListener('click', goBack);

  // Запросы в пределах открытого эвента
  const eapi = (action, body) => api(action, { ...(body || {}), event: state.evId });
  const finished = () => state.ev && state.ev.status === 'finished';

  function setDirty(v) {
    state.dirty = v;
    try { if (tg && ver('6.2')) v ? tg.enableClosingConfirmation() : tg.disableClosingConfirmation(); } catch (e) {}
    if (state.view === 'event') renderTabs();
  }

  async function reloadMe() {
    // Словарь — только если у приложения его нет или он другой (060); до любого построения экрана
    const d = await api('me', { texts: state.textsVersion || undefined });
    setTexts(d);
    if (!d.registered) return d;
    state.me = d.me;
    state.botOwner = !!d.me.is_owner;
    state.open = !!d.open;
    state.create = d.create || null;
    state.companies = d.companies || [];
    state.events = d.events;
    state.limits = d.limits;
    state.looks = d.looks;
    enterCompany(state.companyId);
    return d;
  }

  // Компания открытого экрана (051, R13): права человека в ней — в state.me.is_admin / is_owner, остальные части читают
  // их как раньше. Компании нет (удалена, больше не видна) — экран без компании
  function enterCompany(id) {
    const c = state.companies.find((x) => x.id === id) || null;
    state.companyId = c ? c.id : null;
    state.company = c;
    if (c) {
      state.me.is_admin = c.organizer;
      state.me.is_owner = c.owner;
    }
  }

  // Данные эвента evId; если пользователь уже в другом эвенте — ничего не меняет (009).
  // Участие обновляется на месте: открытые экраны держат ссылку на state.member и видят свежие данные.
  async function reloadEvent(evId = state.evId) {
    const d = await api('event.get', { event: evId });
    if (evId !== state.evId) return null;
    if (state.member && d.member && state.ev && state.ev.id === d.event.id) Object.assign(state.member, d.member);
    else state.member = d.member;
    state.ev = d.event;
    state.can = d.can;
    state.announcements = d.announcements || [];
    state.support = d.support || null;
    state.supportAdmin = d.support_admin || null;
    state.roster = d.roster || null;
    return d;
  }

  function showLoader() { main.replaceChildren(h('div', { class: 'ds-loader' }, h('div', { class: 'ds-spinner' }), t('app.boot.loading'))); }
  // retry — повтор именно неудавшейся загрузки (009); на экране эвента — ещё выход к списку
  function showError(e, retry) {
    const again = h('button', { class: 'ds-btn ds-btn--secondary ds-empty__action', onclick: () => (retry || render)() }, t('app.boot.retry'));
    const home = state.view === 'event' && state.me
      ? h('button', { class: 'ds-btn ds-btn--secondary ds-empty__action', onclick: () => { setDirty(false); goHome(); } }, t('app.home.toEvents'))
      : null;
    main.replaceChildren(ui.empty('⚠️', t('app.boot.loadFailed'), e.message, [again, home]));
  }
  // Асинхронный экран («Подопечный», «Управление»; 026): отказ при построении — экран ошибки с повтором, а не вечный
  // загрузчик, откуда бы экран ни вызвали (render, закрытие шторки, перестройка панели); для покинутого экрана — ничего
  function screen(build) {
    return async (...args) => {
      const nav = state.nav;
      try { return await build(...args); }
      catch (e) { if (nav === state.nav) showError(e); }
    };
  }

  function render() {
    state.nav++;
    window.scrollTo(0, 0);
    try {
      if (state.view === 'companies') renderCompanies();
      else if (state.view === 'home') renderHome();
      else {
        renderHero();
        renderTabs();
        ({ my: renderMy, ward: renderWard, info: renderInfo, admin: renderAdmin }[state.tab] || renderInfo)();
      }
    } catch (e) {
      showError(e); // ошибка вкладки — экран ошибки, а не прежний экран или загрузчик (026)
    }
    syncBack();
  }

  // ---------- главный экран: список эвентов (005) ----------
  async function goHome() {
    const nav = ++state.nav;
    state.view = 'home';
    state.evId = null;
    state.ev = null;
    state.wardCache = null;
    try { history.replaceState(null, '', location.pathname + location.hash); } catch (e) {}
    syncBack();
    showLoader();
    try { await reloadMe(); } catch (e) { if (nav === state.nav) showError(e, goHome); return; }
    if (nav === state.nav) render();
  }

  // Экран «Компании» (051): корневой у человека с несколькими компаниями
  async function goCompanies() {
    const nav = ++state.nav;
    state.view = 'companies';
    state.evId = null;
    state.ev = null;
    state.wardCache = null;
    state.companyId = null;
    try { history.replaceState(null, '', location.pathname + location.hash); } catch (e) {}
    syncBack();
    showLoader();
    try { await reloadMe(); } catch (e) { if (nav === state.nav) showError(e, goCompanies); return; }
    if (nav === state.nav) render();
  }
  // Главный экран компании
  function openCompany(id) {
    state.companyId = id;
    return goHome();
  }

  async function openEvent(id, tab, chat, support) {
    const nav = ++state.nav;
    state.view = 'event';
    state.evId = id;
    state.member = null;
    state.wardCache = null;
    state.wardNameShown = false;
    state.rosterOpen = null;
    syncBack();
    showLoader();
    try { await reloadEvent(id); } catch (e) { if (nav === state.nav) showError(e, () => openEvent(id, tab, chat, support)); return; }
    if (nav !== state.nav) return;
    enterCompany(state.ev.company); // права в компании события (051)
    const m = state.member;
    // Админу «Подопечный» по умолчанию не открывается: просмотр подопечного запрещает отмену жеребьёвки (009)
    // 012: из кнопок бота «💬 Открыть» — обращение участника (support=my) или обращение в панели админа (support=<id>)
    const supportThread = state.me.is_admin && /^\d+$/.test(support || '') ? Number(support) : 0;
    if (support === 'my' && m) tab = 'info';
    if (supportThread) tab = 'admin';
    if (tab === 'my' && !m) tab = 'info';
    state.tab = tab || (state.me.is_admin ? (m ? 'my' : 'admin')
      : m ? (state.ev.draw_done && m.has_ward && m.wishlist.trim() ? 'ward' : 'my') : 'info');
    render();
    if (chat === 'santa' && m && m.has_santa) openChat('santa');
    if (chat === 'ward' && m && m.has_ward) openChat('ward');
    if (support === 'my' && m) openSupport();
    if (supportThread) openSupportThread(supportThread, id, () => (state.evId === id && state.tab === 'admin' ? render() : undefined));
  }

  // Системная кнопка «Назад» Telegram (009): закрывает шторку, иначе — как «‹» (на экране эвента — к его компании, на
  // главном экране — к «Компаниям», 051)
  function syncBack() {
    try {
      if (!tg || !ver('6.1')) return;
      if (sheetEl || ((state.view === 'event' || (state.view === 'home' && withList())) && !heroBack.hidden)) tg.BackButton.show();
      else tg.BackButton.hide();
    } catch (e) {}
  }
  try { if (tg && ver('6.1')) tg.BackButton.onClick(() => (sheetEl ? requestCloseSheet() : goBack())); } catch (e) {}

  function eventRow(e) {
    const chips = [e.status === 'finished' ? ui.chip(t('app.home.finished')) : ui.chip(t('app.home.active'), 'success')];
    if (e.event_date) chips.push(ui.chip('🎁 ' + fmtDate(e.event_date)));
    if (!e.member) chips.push(ui.chip(t('app.home.notMember')));
    return h('div', { class: 'ds-list-item ds-list-item--link', role: 'button', onclick: () => openEvent(e.id) },
      h('div', { class: 'ds-grow' }, h('div', { class: 'ds-list-item__title' }, e.title), h('div', { class: 'ds-chips' }, chips)),
      e.unread ? h('span', { class: 'ds-badge' }, String(e.unread)) : null,
      h('span', { class: 'ds-list-item__chevron' }, '›'));
  }

  function renderHome() {
    const me = state.me;
    const nav = state.nav;
    const many = state.companies.length >= 2;
    heroBack.hidden = !withList();
    heroTitle.textContent = companyTitle();
    const greet = () => { heroSub.textContent = t('app.home.greet', { name: me.name }); };
    greet();
    // Разделы с именами людей (039): после смены своего имени перечитываются на месте
    const people = [];
    document.getElementById('tabs').hidden = true;
    // События — только этой компании (051)
    const events = state.events.filter((e) => !state.companyId || e.company === state.companyId);
    const mine = events.filter((e) => e.member);
    const others = events.filter((e) => !e.member);
    const nodes = [];
    // 051, R9: владелец бота в чужой компании — гость
    if (state.company && state.company.role === 'guest') {
      nodes.push(h('div', { class: 'ds-notice ds-text-sm' }, t('app.home.guest')));
    }
    nodes.push(ui.section(t('app.home.myEvents')));
    if (mine.length) nodes.push(h('div', { class: 'ds-card ds-card--flush' }, mine.map(eventRow)));
    else nodes.push(ui.hint(t(me.is_admin ? 'app.home.noEvents.admin' : 'app.home.noEvents.member'), 'ds-hint--inset'));

    if (me.is_admin) {
      if (others.length) nodes.push(ui.section(t('app.home.otherEvents')), h('div', { class: 'ds-card ds-card--flush' }, others.map(eventRow)));
      const create = ui.btn(t('app.home.create.button'), 'festive', 'ds-btn--block');
      create.addEventListener('click', () => {
        const title = h('input', { class: 'ds-input', type: 'text', maxlength: 80, placeholder: t('app.home.create.placeholder', { year: new Date().getFullYear() + 1 }) });
        // Образец (007): параметры и исключения прошлого эвента, по желанию — приглашение его участников
        const from = h('select', { class: 'ds-input' }, h('option', { value: '' }, t('app.home.create.noTemplate')),
          ...events.map((e) => h('option', { value: e.id }, t('app.home.create.fromTemplate', { title: e.title }))));
        const invite = h('input', { type: 'checkbox', checked: true, disabled: true });
        const inviteRow = h('label', { class: 'ds-check' }, invite, t('app.home.create.invite'));
        from.addEventListener('change', () => { invite.disabled = !from.value; });
        const ok = ui.btn(t('app.core.create'), '', 'ds-btn--block');
        ok.addEventListener('click', () => busy(ok, async () => {
          if (!title.value.trim()) return toast(t('app.core.needTitle'));
          const r = await capi('admin.events.create', { title: title.value, from: from.value ? +from.value : undefined, invite: !!from.value && invite.checked });
          closeSheet();
          toast(r.invited ? t('app.home.create.doneInvited', { count: r.invited })
            : r.invite_error ? t('app.home.create.doneInviteError', { error: r.invite_error }) : t('app.home.create.done'));
          await reloadMe();
          await openEvent(r.event.id, 'admin');
        }));
        const sheet = [ui.hint(t('app.home.create.hint')), clearable(title, t('app.home.create.name'))];
        if (events.length) {
          sheet.push(from, inviteRow, ui.hint(t('app.home.create.templateHint')));
        }
        openSheet(t('app.home.create.title'), [...sheet, ok]);
      });
      nodes.push(h('div', { class: 'ds-card' }, create));
      for (const section of [renderFamiliesSection(), renderAdminsSection()]) {
        nodes.push(section.box);
        people.push(section.load);
      }
    }

    // «Профиль» и «Уведомления» — на корневом экране (051, FR-5125): у человека с несколькими компаниями — на «Компаниях»
    if (!many) nodes.push(...personalSections(nav, greet, people));
    main.replaceChildren(...nodes);
  }

  // «Профиль» и «Уведомления» корневого экрана. Новое имя сразу отображается на экране (039): приветствие и разделы с
  // именами людей (people — их load) перечитываются на месте
  function personalSections(nav, greet, people = []) {
    const me = state.me;
    const nameText = h('div', { class: 'ds-text-strong' }, me.name);
    const profileChanged = () => {
      if (nav !== state.nav) return;
      nameText.textContent = me.name;
      greet();
      for (const load of people) load();
    };
    const editName = ui.btn(t('app.home.profile.editName'), 'secondary', 'ds-btn--block');
    editName.addEventListener('click', () => openNameSheet(profileChanged));

    // Уведомления — общие для всех эвентов
    const modes = [['sound', t('app.home.notify.sound')], ['silent', t('app.home.notify.silent')], ['off', t('app.home.notify.off')]];
    const seg = h('div', { class: 'ds-segmented' });
    const draw = () => seg.replaceChildren(...modes.map(([k, label]) => h('button', {
      class: 'ds-segmented__item' + (me.notify_mode === k ? ' ds-segmented__item--on' : ''),
      onclick: () => busy([...seg.children], async () => { await api('notify.set', { mode: k }); me.notify_mode = k; draw(); toast(t('app.core.saved')); }),
    }, label)));
    draw();
    return [ui.section(t('app.home.profile.title')), ui.card(nameText, editName), ui.section(t('app.home.notify.title')),
      ui.card(ui.hint(t('app.home.notify.hint')), seg)];
  }

  // Шторка «Твоё имя» (035): правило — как при регистрации (до 40 символов, пробелы схлопывает сервер); saved(имя из
  // ответа сервера)
  function openNameSheet(saved) {
    const input = h('input', { class: 'ds-input', type: 'text', maxlength: 40, value: state.me.name, autocomplete: 'off' });
    const save = ui.btn(t('app.core.save'), '', 'ds-btn--block');
    save.addEventListener('click', () => busy(save, async () => {
      if (!input.value.trim()) return toast(t('app.home.name.empty'));
      const r = await api('name.set', { name: input.value });
      state.me.name = r.name;
      closeSheet();
      toast(t('app.home.name.saved'));
      saved(r.name);
    }));
    openSheet(t('app.home.name.title'), [ui.hint(t('app.home.name.hint')), clearable(input, t('app.home.name.title')), save]);
  }

  // Раздел «Семьи» (008): постоянные исключения для всех эвентов. → { box, load }: повторный load заменяет содержимое по
  // ответу, без загрузчика (039)
  function renderFamiliesSection() {
    const box = h('div', {}, ui.section(t('app.home.families.title')), h('div', { class: 'ds-loader' }, h('div', { class: 'ds-spinner' })));
    const load = () => capi('admin.families').then((d) => {
      const list = h('div', { class: 'ds-card ds-card--flush' });
      if (!d.families.length) list.append(h('div', { class: 'ds-list-item ds-text-muted' }, t('app.home.families.empty')));
      for (const f of d.families) {
        list.append(h('div', { class: 'ds-list-item' },
          h('div', { class: 'ds-grow' }, h('div', { class: 'ds-list-item__title' }, f.name), ui.hint(f.members.map((m) => m.name).join(', '))),
          h('button', { class: 'ds-icon-btn', title: t('app.home.families.edit'), onclick: () => openFamily(d, f) }, '✏️')));
      }
      const add = ui.btn(t('app.home.families.new'), 'secondary', 'ds-btn--block');
      add.addEventListener('click', () => openFamily(d, null));
      box.replaceChildren(ui.section(t('app.home.families.title')), list,
        ui.card(ui.hint(t('app.home.families.hint')), add));
    }).catch((e) => box.replaceChildren(ui.section(t('app.home.families.title')), ui.hint('⚠️ ' + e.message, 'ds-hint--inset')));
    function openFamily(d, f) {
      const name = h('input', { class: 'ds-input', type: 'text', maxlength: 40, value: f ? f.name : '', placeholder: t('app.home.families.placeholder') });
      const free = d.users.filter((u) => !u.family_id || (f && u.family_id === f.id));
      const boxes = free.map((u) => {
        const c = h('input', { type: 'checkbox', checked: !!f && u.family_id === f.id });
        c.dataset.id = u.id;
        return [c, h('div', { class: 'ds-row' }, h('label', { class: 'ds-check' }, c, u.name))];
      });
      const save = ui.btn(t('app.core.save'), '', 'ds-btn--block');
      save.addEventListener('click', () => busy(save, async () => {
        const members = boxes.filter(([c]) => c.checked).map(([c]) => +c.dataset.id);
        await capi('admin.family.save', { id: f ? f.id : undefined, name: name.value, members });
        closeSheet();
        toast(t('app.home.families.saved'));
        await load();
      }));
      const items = [clearable(name, t('app.home.families.name')), ui.hint(t('app.home.families.members')), ...boxes.map(([, l]) => l), save];
      if (f) {
        const del = ui.btn(t('app.home.families.delete'), 'danger', 'ds-btn--block');
        del.addEventListener('click', async () => {
          if (!(await confirmBox(t('app.home.families.deleteConfirm', { name: f.name })))) return;
          busy(del, async () => { await capi('admin.family.delete', { id: f.id }); closeSheet(); toast(t('app.home.families.deleted')); await load(); });
        });
        items.push(del);
      }
      openSheet(t(f ? 'app.home.families.family' : 'app.home.families.new'), items);
    }
    load();
    return { box, load };
  }

  // Раздел «Владелец и администраторы» (002): люди и права компании открытого экрана. → { box, load }, как у «Семей»
  function renderAdminsSection() {
    const box = h('div', {}, ui.section(t('app.home.admins.title')), h('div', { class: 'ds-loader' }, h('div', { class: 'ds-spinner' })));
    // 👑 (051): владелец компании передаёт свои права; владелец бота делает человека владельцем компании
    const makeOwner = async (u, btns) => {
      if (!(await confirmBox(state.botOwner
        ? t('app.home.admins.makeOwnerConfirm', { name: u.name })
        : t('app.home.admins.transferConfirm', { name: u.name })))) return;
      busy(btns, async () => { await capi('admin.owner.transfer', { id: u.id }); toast(t('app.home.admins.transferred')); await goHome(); });
    };
    const load = () => capi('admin.users').then((d) => {
      const me = state.me;
      const admins = d.users.filter((u) => u.is_admin).sort((a, b) => (b.is_owner ? 1 : 0) - (a.is_owner ? 1 : 0));
      const list = h('div', { class: 'ds-card ds-card--flush' });
      for (const u of admins) {
        const actions = h('div', { class: 'ds-row ds-row--tight' });
        if (me.is_owner && !u.is_owner) {
          actions.append(
            h('button', { class: 'ds-icon-btn', title: t('app.home.admins.transfer'), onclick: () => makeOwner(u, [...actions.children]) }, '👑'),
            h('button', { class: 'ds-icon-btn', title: t('app.home.admins.revoke'), onclick: async () => {
              if (!(await confirmBox(t('app.home.admins.revokeConfirm', { name: u.name })))) return;
              busy([...actions.children], async () => { await capi('admin.role.revoke', { id: u.id }); toast(t('app.home.admins.revoked')); await goHome(); });
            } }, '✕'));
        }
        list.append(h('div', { class: 'ds-list-item' },
          h('div', { class: 'ds-grow' },
            h('div', { class: 'ds-list-item__title' }, u.name || t('app.core.noName'), u.id === me.id ? h('span', { class: 'ds-hint ds-text-regular' }, t('app.home.you')) : null),
            h('div', { class: 'ds-chips' }, u.is_owner ? ui.chip(t('app.core.role.owner')) : ui.chip(t('app.core.role.admin')))),
          actions));
      }
      const parts = [ui.section(t('app.home.admins.title')), list];
      if (me.is_owner) {
        const candidates = d.users.filter((u) => !u.is_admin);
        if (candidates.length) {
          const sel = h('select', { class: 'ds-input' }, h('option', { value: '' }, t('app.home.admins.grantPick')), ...candidates.map((u) => h('option', { value: u.id }, u.name)));
          const grant = ui.btn(t('app.home.admins.grant'), 'secondary', 'ds-btn--block');
          grant.addEventListener('click', async () => {
            const u = candidates.find((x) => String(x.id) === sel.value);
            if (!u) return toast(t('app.core.pickPerson'));
            if (!(await confirmBox(t('app.home.admins.grantConfirm', { name: u.name })))) return;
            busy(grant, async () => { await capi('admin.role.grant', { id: u.id }); toast(t('app.home.admins.granted')); await goHome(); });
          });
          parts.push(ui.card(ui.hint(t('app.home.admins.ownerHint')), sel, grant));
          // Владельцем можно сделать и не администратора (051)
          const pick = h('select', { class: 'ds-input' }, h('option', { value: '' }, t('app.home.admins.ownerPick')), ...candidates.map((u) => h('option', { value: u.id }, u.name)));
          const crown = ui.btn(t('app.home.admins.makeOwner'), 'secondary', 'ds-btn--block');
          crown.addEventListener('click', () => {
            const u = candidates.find((x) => String(x.id) === pick.value);
            return u ? makeOwner(u, crown) : toast(t('app.core.pickPerson'));
          });
          parts.push(ui.card(pick, crown));
        }
      } else {
        parts.push(ui.hint(d.owner_name ? t('app.home.admins.onlyOwnerNamed', { name: d.owner_name }) : t('app.home.admins.onlyOwner'), 'ds-hint--inset'));
      }
      const log = ui.btn(t('app.core.auditLog'), 'secondary', 'ds-btn--block');
      log.addEventListener('click', () => openAudit(null));
      parts.push(ui.card(ui.hint(t('app.home.admins.logHint')), log));
      box.replaceChildren(...parts);
    }).catch((e) => box.replaceChildren(ui.section(t('app.home.admins.title')), ui.hint('⚠️ ' + e.message, 'ds-hint--inset')));
    load();
    return { box, load };
  }
