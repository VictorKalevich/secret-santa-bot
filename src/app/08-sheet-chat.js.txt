
  // ---------- шторка ----------
  // beforeClose — проверка перед закрытием пользователем (✕, фон, «Назад», драгер): false — шторка остаётся (009).
  // Открытую шторку (sheetEl) знает только ядро шторки (026): openSheet возвращает свою шторку, заголовок может быть
  // узлом — содержимое меняет свой заголовок и опрашивает сервер (poll), только пока открыта именно его шторка.
  // У каждой шторки драгер (033) — его ставит openSheet; пока шторка открыта, жест Telegram «свайп вниз сворачивает
  // приложение» выключен (Bot API 7.7), иначе свайп за полосу свернул бы всё приложение; после закрытия — как было
  let sheetEl = null;
  let sheetOnClose = null;
  let sheetBeforeClose = null;
  let sheetSwipes = false;
  function openSheet(title, content, { full = false, onClose = null, beforeClose = null } = {}) {
    closeSheet();
    const header = h('div', { class: 'ds-row ds-row--between' }, h('h2', { class: 'ds-sheet__title' }, title),
      h('button', { class: 'ds-icon-btn', title: t('app.core.calendar.close'), onclick: requestCloseSheet }, '✕'));
    // Закрытие клавиатуры может сдвинуть шторку между pointerup и click. Учитываем, где начался тап.
    let pressedBackdrop = false;
    sheetEl = h('div', {
      class: 'ds-sheet-backdrop',
      onpointerdown: (e) => { pressedBackdrop = e.isPrimary && e.target === e.currentTarget; },
      onpointercancel: () => { pressedBackdrop = false; },
      onclick: (e) => {
        const close = pressedBackdrop && e.target === e.currentTarget;
        pressedBackdrop = false;
        if (close) requestCloseSheet();
      },
    }, h('div', { class: full ? 'ds-sheet ds-sheet--full' : 'ds-sheet ds-stack' }, header, ...[].concat(content)));
    sheetHandle(sheetEl);
    sheetOnClose = onClose;
    sheetBeforeClose = beforeClose;
    document.body.append(sheetEl);
    document.body.classList.add('ds-root--locked'); // страница под шторкой не прокручивается (011)
    try { if (tg && tg.isVerticalSwipesEnabled) { tg.disableVerticalSwipes(); sheetSwipes = true; } } catch (e) {}
    syncBack();
    return sheetEl;
  }
  // Драгер шторки own (028; у каждой шторки — 033): полоса сверху; шторка едет за пальцем вниз (вверх — нет). Отпустили
  // дальше четверти высоты или быстрым свайпом — закрытие, как ✕: сначала вопрос beforeClose (шторка ждёт ответа там,
  // где её отпустили), отказ — возврат на место, иначе уходит вниз и закрывается — если открыта всё ещё она
  function sheetHandle(own) {
    const sheet = own.firstElementChild;
    const handle = h('div', { class: 'ds-sheet__handle', 'aria-hidden': 'true' });
    sheet.classList.add('ds-sheet--handle');
    sheet.prepend(handle);
    let drag = null;
    const shift = (dy) => { sheet.style.transform = dy ? `translateY(${dy}px)` : ''; };
    handle.addEventListener('pointerdown', (e) => {
      drag = { id: e.pointerId, y: e.clientY, t: Date.now() };
      handle.setPointerCapture(e.pointerId);
      sheet.classList.remove('ds-sheet--settle');
    });
    handle.addEventListener('pointermove', (e) => {
      if (drag && e.pointerId === drag.id) shift(Math.max(0, e.clientY - drag.y));
    });
    const release = async (e) => {
      if (!drag || e.pointerId !== drag.id) return;
      const dy = Math.max(0, e.clientY - drag.y);
      const fast = dy > 30 && dy / Math.max(1, Date.now() - drag.t) > 0.5;
      drag = null;
      sheet.classList.add('ds-sheet--settle');
      if (e.type !== 'pointerup' || !(dy > sheet.offsetHeight / 4 || fast)) return shift(0);
      if (sheetEl !== own || !(await mayCloseSheet())) return shift(0);
      let gone = false;
      const close = () => { if (!gone) { gone = true; if (sheetEl === own) closeSheet(); } };
      sheet.addEventListener('transitionend', close, { once: true });
      setTimeout(close, 400);
      sheet.style.transform = 'translateY(100%)';
    };
    handle.addEventListener('pointerup', release);
    handle.addEventListener('pointercancel', release);
  }
  // Опрос шторки own (026, 058). load() — запрос и отрисовка, истина — пришло новое. Промежуток — pollDelay: в открытом
  // боте без нового растёт (POLL_STEPS, секунды), после POLL_IDLE секунд тишины опрос спит; в закрытом — 6 секунд.
  // Пока приложение не на экране, запросов нет. now() — запрос сразу и расписание с начала (первая загрузка, своя отправка,
  // возвращение в приложение, касание спящей шторки); отказ load из now() — вызывающему, при опросе — проглатывается.
  // Ответ запроса, начатого до now(), расписание не трогает (gen). Шторка закрыта или заменена другой — опрос
  // останавливается сам; stop() — раньше
  const POLL_STEPS = [6, 10, 20, 30, 60];
  const POLL_IDLE = 300;
  function pollDelay(empty, quiet, open) {
    if (!open) return POLL_STEPS[0];
    return quiet >= POLL_IDLE ? null : POLL_STEPS[Math.min(empty, POLL_STEPS.length - 1)];
  }
  function poll(own, load) {
    let timer = null, empty = 0, quiet = 0, asleep = false, stopped = false, gen = 0;
    const alive = () => !stopped && sheetEl === own;
    const plan = () => {
      clearTimeout(timer);
      timer = null;
      asleep = false;
      if (!alive()) return stop();
      if (!appVisible()) return;
      const s = pollDelay(empty, quiet, state.open);
      const g = gen;
      if (s === null) asleep = true;
      else timer = setTimeout(() => tick(s, g), s * 1000);
    };
    const tick = async (s, g) => {
      timer = null;
      if (!alive()) return stop();
      let fresh = false;
      try { fresh = !!(await load()); } catch (e) {}
      if (g !== gen) return;
      if (fresh) { empty = 0; quiet = 0; } else { empty += 1; quiet += s; }
      plan();
    };
    const now = async () => {
      gen += 1;
      clearTimeout(timer);
      timer = null;
      asleep = false;
      empty = 0;
      quiet = 0;
      const g = gen;
      try { return await load(); } finally { if (g === gen) plan(); }
    };
    const unwatch = onVisibility((shown) => {
      if (!alive()) return stop();
      if (shown) now().catch(() => {});
      else { clearTimeout(timer); timer = null; }
    });
    own.addEventListener('pointerdown', () => { if (asleep && alive()) now().catch(() => {}); });
    function stop() {
      stopped = true;
      clearTimeout(timer);
      timer = null;
      unwatch();
    }
    return { now, stop };
  }
  async function mayCloseSheet() {
    return !sheetBeforeClose || !!(await sheetBeforeClose());
  }
  async function requestCloseSheet() {
    if (await mayCloseSheet()) closeSheet();
  }
  function closeSheet() {
    if (sheetEl) { sheetEl.remove(); sheetEl = null; }
    document.body.classList.remove('ds-root--locked');
    if (sheetSwipes) { sheetSwipes = false; try { tg.enableVerticalSwipes(); } catch (e) {} }
    sheetBeforeClose = null;
    syncBack();
    const cb = sheetOnClose;
    sheetOnClose = null;
    // onClose бывает асинхронной (перечитать событие, перерисовать экран): отказ не теряется молча (026); вызов —
    // сразу, как раньше (обещание с исполнителем начинает его синхронно)
    if (cb) new Promise((done) => done(cb())).catch((e) => toast('⚠️ ' + e.message));
  }

  // Шторка со словом подтверждения (035; вынесена из «Опасной зоны»): что удалится, что останется (keep, необязательно),
  // поле «Введи слово …», кнопка — только при введённом слове; sure — ещё окно «точно?». run(word) — запрос, после
  // ответа шторка закрывается и вызывается after(ответ)
  function typedSheet(title, { lines, keep = '', word, label, sure = false, run, after }) {
    const inp = h('input', { class: 'ds-input', type: 'text', placeholder: t('app.sheet.typed.placeholder', { word }), autocomplete: 'off' });
    const go = ui.btn(label, 'danger', 'ds-btn--block');
    go.disabled = true;
    inp.addEventListener('input', () => { go.disabled = inp.value.trim().toUpperCase() !== word; });
    go.addEventListener('click', async () => {
      if (sure && !(await confirmBox(t('app.sheet.typed.sure', { title })))) return;
      busy(go, async () => { const r = await run(word); closeSheet(); await after(r); });
    });
    openSheet(title, [
      h('div', { class: 'ds-notice ds-notice--error' }, h('b', {}, t('app.sheet.typed.willDelete')), h('ul', { class: 'ds-bullets' }, lines.map((l) => h('li', {}, l)))),
      keep ? ui.hint(keep) : null,
      clearable(inp, t('app.sheet.typed.field')), go]);
  }

  // ---------- поле ответа: переписка Санты и подопечного, обращения ----------
  // Текст, «Отправить», счётчик символов (виден за 500 до предела); одно на все переписки (026)
  function composerParts(placeholder) {
    const max = state.limits.chars;
    const ta = ui.textarea({ class: 'ds-input ds-composer__input', rows: 1, placeholder });
    const send = ui.btn(t('app.sheet.send'), '', 'ds-composer__send');
    const counter = h('div', { class: 'ds-counter ds-composer__counter', hidden: true });
    const composer = h('div', { class: 'ds-composer' }, ta, send);
    const update = () => {
      const n = ta.value.length;
      counter.hidden = n < max - 500;
      counter.textContent = `${n} / ${max}`;
      counter.classList.toggle('ds-counter--over', n > max);
      send.disabled = !ta.value.trim() || n > max;
    };
    ta.addEventListener('input', update);
    update();
    const clear = (text) => { if (ta.value === text) { ta.value = ''; ta.style.height = 'auto'; } update(); };
    // Скрытое поле не измерить: высота, посчитанная до показа, обрезала бы его — пересчитываем при показе
    const show = (on) => {
      const was = composer.hidden;
      composer.hidden = !on;
      if (on && was) { ta.style.height = 'auto'; if (ta.value) ta.style.height = ta.scrollHeight + 2 + 'px'; }
    };
    return { ta, send, counter, composer, update, clear, show };
  }

  // После шторки события (переписка, обращения участника): событие — заново, экран — если пользователь ещё на нём.
  // «Подопечный» — по прежнему ответу: подсветка изменений письма сохраняется (009)
  async function backToEvent(evId) {
    const nav = state.nav;
    await reloadEvent(evId).catch(() => {});
    if (nav !== state.nav || state.evId !== evId) return;
    if (state.dirty) renderTabs();
    else if (state.tab === 'ward') { renderHero(); renderTabs(); renderWard({ cached: true }); }
    else render();
  }

  // ---------- переписка Санты и подопечного (003) ----------
  function openChat(withWhom) {
    const list = h('div', { class: 'ds-chat' }, h('div', { class: 'ds-chat__empty' }, t('app.boot.loading')));
    const c = composerParts(t('app.sheet.chat.placeholder'));
    const closedNote = h('div', { class: 'ds-notice ds-text-sm', hidden: true }, t('app.sheet.chatReadOnly'));
    const evId = state.evId;
    // lastId — только из ответа chat.get (009): своё отправленное сообщение его не двигает, иначе ответ собеседника,
    // пришедший в те же секунды (с меньшим id), не загрузился бы никогда
    let lastId = 0;
    let live = null;
    const shown = new Set();
    // 040: маска пары — подопечному в заголовке, Санте в подсказке; сначала из экрана события, затем из каждого ответа
    const title = h('span', {}, withWhom === 'ward' ? t('app.event.chat.wardTitle') : t('app.sheet.chat.yourSanta'));
    const note = ui.hint(withWhom === 'ward' ? '' : t('app.sheet.chat.santaKnows'));
    const showMask = (mask) => {
      if (withWhom === 'ward') note.textContent = t('app.sheet.chat.youAre', { santa: maskedSanta(mask) });
      else title.textContent = t('app.sheet.chat.yourMasked', { santa: maskedSanta(mask) });
    };
    showMask(withWhom === 'ward' ? (state.wardCache && state.wardCache.evId === evId ? state.wardCache.d.mask : '') : state.member && state.member.santa_mask);

    const bubble = (m) => {
      const el = h('div', { class: 'ds-chat__msg ' + (m.mine ? 'ds-chat__msg--mine' : 'ds-chat__msg--theirs') },
        m.text, h('div', { class: 'ds-chat__meta' }, fmtTime(m.created_at)));
      el.dataset.id = m.id;
      return el;
    };
    // Вставка по порядку id: сообщение собеседника может прийти после уже показанного своего; истина — что-то добавлено
    const add = (msgs) => {
      const fresh = msgs.filter((m) => !shown.has(m.id));
      if (!fresh.length) return false;
      if (!shown.size) list.replaceChildren();
      for (const m of fresh) {
        shown.add(m.id);
        const next = [...list.children].find((el) => Number(el.dataset.id) > m.id);
        list.insertBefore(bubble(m), next || null);
      }
      list.scrollTop = list.scrollHeight;
      return true;
    };
    const setWritable = (w) => { c.show(w); closedNote.hidden = w; if (!w) c.counter.hidden = true; };
    const load = async () => {
      const d = await api('chat.get', { event: evId, with: withWhom, after: lastId });
      if (d.state !== 'ok') { list.replaceChildren(h('div', { class: 'ds-chat__empty' }, t('app.sheet.chat.noDraw'))); return; }
      showMask(d.mask);
      setWritable(d.writable);
      for (const m of d.messages) lastId = Math.max(lastId, m.id);
      if (!shown.size && !d.messages.length) {
        list.replaceChildren(h('div', { class: 'ds-chat__empty' }, !d.writable ? t('app.sheet.chat.noneWere')
          : withWhom === 'ward' ? t('app.sheet.chat.emptyWard') : t('app.sheet.chat.emptySanta')));
      }
      return add(d.messages);
    };
    c.send.addEventListener('click', () => busy(c.send, async () => {
      const text = c.ta.value;
      const r = await api('message.send', { event: evId, to: withWhom, text });
      c.clear(text);
      add([r.message]);
      haptic('success');
      await live.now().catch(() => {});
    }).then(c.update));
    setWritable(state.can.chat_write);

    const own = openSheet(title, [note, list, c.composer, c.counter, closedNote],
      {
        full: true,
        beforeClose: async () => !c.ta.value.trim() || !!(await confirmBox(t('app.sheet.chat.unsent'))),
        onClose: async () => {
          live.stop();
          await backToEvent(evId);
        },
      });
    live = poll(own, load);
    live.now().catch((e) => toast('⚠️ ' + e.message));
  }

  // Участник: все его обращения в эвенте одной перепиской; закрытые отделены служебной строкой
  function openSupport() {
    const evId = state.evId;
    const c = composerParts(t('app.sheet.support.placeholder'));
    const list = h('div', { class: 'ds-chat' }, h('div', { class: 'ds-chat__empty' }, t('app.boot.loading')));
    const status = ui.hint('');
    const closedNote = h('div', { class: 'ds-notice ds-text-sm', hidden: true }, t('app.sheet.supportReadOnly'));
    // Под полем ввода: в шторке на весь экран нет ds-stack — отступ сверху даёт ds-sheet__action (027)
    const resolve = ui.btn(t('app.sheet.support.resolve'), 'secondary', 'ds-btn--block ds-sheet__action');
    resolve.hidden = true;
    let live = null;
    let sig = '';
    const load = async () => {
      const d = await api('support.get', { event: evId });
      const open = d.threads.find((th) => th.status === 'open');
      const key = JSON.stringify(d.threads.map((th) => [th.id, th.status, th.assignee_name, th.messages.length]));
      c.show(d.writable && (d.available || !!open));
      closedNote.hidden = d.writable;
      resolve.hidden = !open || !d.writable;
      status.textContent = open
        ? (open.assignee_name ? t('app.sheet.support.ledBy', { name: open.assignee_name }) : t('app.sheet.support.sent'))
        : t('app.sheet.support.intro');
      if (key === sig) return false;
      sig = key;
      const nodes = [];
      for (const th of d.threads) {
        nodes.push(...th.messages.map(supportBubble));
        if (th.status !== 'open') nodes.push(h('div', { class: 'ds-chat__note' }, t('app.sheet.support.closedAt', { time: fmtTime(th.closed_at) })));
      }
      list.replaceChildren(...(nodes.length ? nodes : [h('div', { class: 'ds-chat__empty' }, d.available
        ? t('app.sheet.support.empty')
        : t('app.sheet.support.nobody'))]));
      list.scrollTop = list.scrollHeight;
      return true;
    };
    c.send.addEventListener('click', () => busy(c.send, async () => {
      const text = c.ta.value;
      await api('support.send', { event: evId, text });
      c.clear(text);
      haptic('success');
      await live.now().catch(() => {});
    }).then(c.update));
    resolve.addEventListener('click', async () => {
      if (!(await confirmBox(t('app.sheet.support.resolveConfirm')))) return;
      busy(resolve, async () => { await api('support.close', { event: evId }); toast(t('app.sheet.support.closed')); await live.now(); });
    });
    const own = openSheet(t('app.sheet.support.title'), [status, list, c.composer, c.counter, closedNote, resolve], {
      full: true,
      beforeClose: async () => !c.ta.value.trim() || !!(await confirmBox(t('app.sheet.support.unsent'))),
      onClose: async () => {
        live.stop();
        await backToEvent(evId);
      },
    });
    live = poll(own, load);
    live.now().catch((e) => toast('⚠️ ' + e.message));
  }

  // Админ: одно обращение. Переписка — если её можно видеть (ничьё, своё, владелец); действия — взять, передать, закрыть

        live.stop();
        await reloadEvent(evId).catch(() => {});
        if (state.evId !== evId) return;
        renderTabs();
        if (after) await after();
      },
    });
    live = poll(own, load);
    live.now().catch((e) => toast('⚠️ ' + e.message));
  }
