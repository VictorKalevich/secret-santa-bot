  function slotNames(value) {

  // Формы числа — по локали словаря (app.core.locale): код языка не обязан быть локалью Intl
  function pluralCategory(n) {
    if (!pluralRules) {
      const locale = own(TEXTS, 'app.core.locale') ? TEXTS['app.core.locale'] : textsLang;
      try { pluralRules = new Intl.PluralRules(locale); } catch (e) { pluralRules = new Intl.PluralRules('ru'); }

  // Текст по ключу: {имя} — подстановка, {имя.форма} — форма из набора (ownerWord); набор форм числа — по params.n.
  // Нет ключа — сам ключ и console.error (тесты в Chrome собирают такие ошибки)
  function t(key, params) {
    const value = own(TEXTS, key) ? TEXTS[key] : undefined;
    if (value === undefined) {
      i18nError('нет ключа ' + key);
      return key;
    }
    if (Array.isArray(value)) {
      i18nError(key + ' — список, а не текст');
      return value.join(', ');
    }
    let text = value;
    if (typeof value === 'object') {
      const n = params && params.n;
      if (typeof n !== 'number') i18nError(key + ': у набора форм нет числа n');
      text = value[typeof n === 'number' ? pluralCategory(n) : 'other'] ?? value.other ?? Object.values(value)[0];
    }
    if (params) {
      const used = slotNames(value);
      for (const name of Object.keys(params)) {
        if (!used.has(name) && !(name === 'n' && typeof value === 'object')) i18nError(key + ': лишняя подстановка ' + name);
      }
    }
    return text.replace(SLOT, (m, name, form) => {
      let v = params && own(params, name) ? params[name] : undefined;
      if (form !== undefined && v !== undefined) v = v && typeof v === 'object' && own(v, form) ? v[form] : undefined;
      if (v !== undefined) return String(v);
      i18nError(key + ': нет подстановки ' + (form ? name + '.' + form : name));
      return m;
    });
  }

  function h(tag, attrs, ...children) {
    const el = document.createElement(tag);
    for (const [k, v] of Object.entries(attrs || {})) {
      if (v === undefined || v === null || v === false) continue;
      if (k === 'class') el.className = v;
      else if (k.startsWith('on')) el.addEventListener(k.slice(2), v);
      else if (k === 'value') el.value = v;
      else if (k === 'checked') el.checked = !!v;
      else el.setAttribute(k, v === true ? '' : v);
    }
    for (const c of children.flat()) {
      if (c === null || c === undefined || c === false) continue;
      el.append(c instanceof Node ? c : document.createTextNode(String(c)));
    }
    return el;
  }

  // Тап мимо поля прячет клавиатуру (039): WebKit (iOS) снимает фокус с поля, только когда тапнули другое поле, а клик по
  // «некликабельному» месту до документа не доходит — поэтому события указателя. Тап — касание без сдвига (прокрутку iOS
  // отменяет: pointercancel). Поля, кнопки, ссылки, списки и подписи работают как раньше: спрятанная клавиатура меняет
  // высоту окна, и нажатие кнопки попало бы не туда.
  const TAP_SLOP = 10;
  const KEEPS_FOCUS = 'input, textarea, select, button, a, label, [role="button"], [contenteditable]';
  function blurOnTap() {
    let down = null;
    let sheetTouch = null;
    // При автокоррекции снятие фокуса завершает ввод и может сдвинуть шторку.
    // Сначала отменяем системный click этого касания, затем прячем клавиатуру.
    document.addEventListener('touchstart', (e) => {
      sheetTouch = null;
      const field = document.activeElement;
      const target = e.target instanceof Element ? e.target : e.target.parentElement;
      if (e.touches.length !== 1 || !field?.matches('input.ds-input:not([readonly]), textarea') ||
          !field.closest('.ds-sheet-backdrop')?.contains(target) || target.closest(KEEPS_FOCUS)) return;
      const touch = e.touches[0];
      sheetTouch = { field, id: touch.identifier, x: touch.screenX, y: touch.screenY };
    }, { capture: true, passive: true });
    document.addEventListener('touchcancel', () => { sheetTouch = null; }, true);
    document.addEventListener('touchend', (e) => {
      const start = sheetTouch;
      sheetTouch = null;
      const touch = e.changedTouches[0];
      if (!start || touch.identifier !== start.id || Math.hypot(touch.screenX - start.x, touch.screenY - start.y) > TAP_SLOP) return;
      if (e.cancelable) e.preventDefault();
      start.field.blur();
    }, { capture: true, passive: false });
    document.addEventListener('pointerdown', (e) => { down = e.isPrimary ? { id: e.pointerId, x: e.clientX, y: e.clientY } : null; }, true);
    document.addEventListener('pointercancel', () => { down = null; sheetTouch = null; }, true);
    document.addEventListener('pointerup', (e) => {
      const tap = !!down && down.id === e.pointerId && Math.hypot(e.clientX - down.x, e.clientY - down.y) <= TAP_SLOP;
      down = null;
      const field = document.activeElement;
      if (sheetTouch || !tap || !field || !field.matches('input, textarea') || !(e.target instanceof Element) || e.target.closest(KEEPS_FOCUS)) return;
      field.blur();
    }, true);
  }

  // Страница видна и Telegram не свернул мини-приложение: iOS сворачивает его в плашку без visibilitychange, для этого —

  // onVisibility(fn) — fn(на экране) при каждой смене, возвращает отписку: опрос шторок встаёт на паузу, черновик письма
  // сохраняется сразу
  const appVisible = () => !document.hidden && !(tg && tg.isActive === false);
  const visibilityWatchers = new Set();
  function onVisibility(fn) {
    visibilityWatchers.add(fn);
    return () => visibilityWatchers.delete(fn);
  }
  function watchVisibility() {
    let shown = appVisible();
    const changed = () => {
      if (appVisible() === shown) return;
      shown = !shown;
      for (const fn of [...visibilityWatchers]) fn(shown);
    };
    document.addEventListener('visibilitychange', changed);
    try { if (tg) { tg.onEvent('activated', changed); tg.onEvent('deactivated', changed); } } catch (e) {}
  }

  // ---------- свой календарь (027) ----------
  // Системный календарь сам выделяет сегодняшний день и по «Готово» подставляет его (iOS) — там, где дата сама включает
  // действие (автожеребьёвка), календарь свой: без даты ничего не выделено, «Готово» — только после выбора дня.
  const isoDay = (y, m, d) => `${y}-${String(m).padStart(2, '0')}-${String(d).padStart(2, '0')}`;
  // Клетки месяца m (1–12) года y: null — пустые до 1-го числа (неделя с понедельника), затем { iso, day } по порядку
  function calendarCells(y, m) {
    const cells = [];
    const lead = (new Date(y, m - 1, 1).getDay() + 6) % 7;
    for (let i = 0; i < lead; i++) cells.push(null);
    const days = new Date(y, m, 0).getDate();
    for (let d = 1; d <= days; d++) cells.push({ iso: isoDay(y, m, d), day: d });
    return cells;
  }
  // Поле — кнопка в виде поля ввода; value — 'YYYY-MM-DD' или '' (панель читает его, как у полей ввода); выбор шлёт change
  function datePick(value, { title, hint } = {}) {
    const el = h('button', { type: 'button', class: 'ds-input ds-input--pick' });
    const show = (iso) => {
      const m = /^(\d{4})-\d{2}-\d{2}$/.exec(iso || '');
      el.value = m ? iso : '';
      el.textContent = m ? t('app.core.dateYear', { date: fmtDate(iso), year: m[1] }) : t('app.core.dateNone');
      el.classList.toggle('ds-input--placeholder', !m);
    };
    const set = (iso) => { show(iso); el.dispatchEvent(new Event('change', { bubbles: true })); };
    show(value);
    el.addEventListener('click', () => openCalendar(el.value, set, { title, hint }));
    return el;
  }
  // Шторка календаря: месяц даты (без неё — текущий); дни раньше сегодня недоступны (кроме уже заданной даты); назад —
  // не раньше текущего месяца и месяца даты. «Готово» → onPick(день), «Сбросить» (только при дате) → onPick('')
  function openCalendar(value, onPick, { title = t('app.core.calendar.title'), hint } = {}) {
    const now = new Date();
    const today = isoDay(now.getFullYear(), now.getMonth() + 1, now.getDate());
    let picked = value;
    const start = (value || today).slice(0, 7);
    const min = start < today.slice(0, 7) ? start : today.slice(0, 7);
    let view = { y: +start.slice(0, 4), m: +start.slice(5, 7) };

    const prev = h('button', { type: 'button', class: 'ds-icon-btn', title: t('app.core.calendar.prev'), 'aria-label': t('app.core.calendar.prev') }, '‹');
    const next = h('button', { type: 'button', class: 'ds-icon-btn', title: t('app.core.calendar.next'), 'aria-label': t('app.core.calendar.next') }, '›');
    const grid = h('div', { class: 'ds-calendar__grid' });

    done.disabled = !picked;

      prev.disabled = isoDay(view.y, view.m, 1).slice(0, 7) <= min;

    };
    prev.addEventListener('click', () => shift(-1));
    next.addEventListener('click', () => shift(1));
    done.addEventListener('click', () => { closeSheet(); onPick(picked); });

  }

  async function api(action, body) {
    let res;
    try {
      res = await fetch('/api/' + action, {
        method: 'POST',
        headers: { 'content-type': 'application/json', 'x-init-data': initData },
        body: JSON.stringify(body || {}),
      });
    } catch (e) {
      throw new Error(t('app.boot.offline'));
    }
    const data = await res.json().catch(() => ({ ok: false, error: t('app.boot.serverError') }));
    if (!data.ok) throw new Error(data.error || t('app.boot.error'));
    return data;
  }

  function bindToggle(input, fn) {
    input.addEventListener('change', async () => {
      const on = input.checked;
      input.disabled = true;
      try {
        if ((await fn(on)) === false) input.checked = !on;
      } catch (e) {
        input.checked = !on;
        haptic('error');
        toast('⚠️ ' + e.message);
      } finally {
        input.disabled = false;
      }
    });
    return input;
  }
  // Даты (060): «25 ноября» — app.core.date с месяцем в родительном падеже; заголовок календаря (027) — месяц в
  // именительном, неделя с понедельника (app.core.weekdays)
  function fmtDate(iso) {
    const m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(iso || '');
    return m ? t('app.core.date', { day: +m[3], month: tl('app.core.monthsGen')[+m[2] - 1] }) : '';
  }
  function fmtTime(ms) {
    if (!ms) return '';
    const d = new Date(ms);
    const now = new Date();
    const time = d.toLocaleTimeString(t('app.core.locale'), { hour: '2-digit', minute: '2-digit' });
    if (d.toDateString() === now.toDateString()) return t('app.core.todayAt', { time });
    const y = new Date(now); y.setDate(now.getDate() - 1);
    if (d.toDateString() === y.toDateString()) return t('app.core.yesterdayAt', { time });
    return t('app.core.dateAt', { date: t('app.core.date', { day: d.getDate(), month: tl('app.core.monthsGen')[d.getMonth()] }), time });
  }
  function autosize(el) {
    const fit = () => { el.style.height = 'auto'; el.style.height = el.scrollHeight + 2 + 'px'; };
    el.addEventListener('input', fit);
    requestAnimationFrame(fit);
    return el;
  }

  // ---------- сравнение версий письма (по словам) ----------
  function diffTokens(a, b) {
    let start = 0;
    while (start < a.length && start < b.length && a[start] === b[start]) start++;
    let endA = a.length, endB = b.length;
    while (endA > start && endB > start && a[endA - 1] === b[endB - 1]) { endA--; endB--; }
    const A = a.slice(start, endA), B = b.slice(start, endB);
    const n = A.length, m = B.length;
    if (n * m > 4e6) return null;
    const W = m + 1;
    const dp = new Uint16Array((n + 1) * W);
    for (let i = n - 1; i >= 0; i--)
      for (let j = m - 1; j >= 0; j--)
        dp[i * W + j] = A[i] === B[j] ? dp[(i + 1) * W + j + 1] + 1 : Math.max(dp[(i + 1) * W + j], dp[i * W + j + 1]);
    const ops = [];
    const push = (kind, s) => { const last = ops[ops.length - 1]; if (last && last.t === kind) last.s += s; else ops.push({ t: kind, s }); };
    if (start) push('=', a.slice(0, start).join(''));
    let i = 0, j = 0;
    while (i < n && j < m) {
      if (A[i] === B[j]) { push('=', A[i]); i++; j++; }
      else if (dp[(i + 1) * W + j] >= dp[i * W + j + 1]) { push('-', A[i]); i++; }
      else { push('+', B[j]); j++; }
    }
    while (i < n) push('-', A[i++]);
    while (j < m) push('+', B[j++]);
    if (endA < a.length) push('=', a.slice(endA).join(''));
    return ops;
  }
  function renderDiff(oldText, newText) {
    const tok = (s) => s.match(/\s+|[^\s]+/g) || [];
    const ops = diffTokens(tok(oldText), tok(newText));
    const box = h('div', { class: 'ds-letter' });
    if (!ops) { box.textContent = newText; return box; }
    for (const op of ops) {
      if (op.t === '=') { box.append(op.s); continue; }
      // Пробелы и переносы по краям — вне подсветки; удалённые пробелы не показываем
      const [, lead, core, trail] = /^(\s*)([\s\S]*?)(\s*)$/.exec(op.s);
      if (op.t === '+') box.append(lead);
      if (core) box.append(h(op.t === '+' ? 'mark' : 'del', { class: op.t === '+' ? 'ds-diff-add' : 'ds-diff-del' }, core));
      if (op.t === '+') box.append(trail);
    }
    return box;
  }
