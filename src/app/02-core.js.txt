  // ---------- утилиты ----------
  // Создание элементов: пользовательский текст всегда вставляется как текст, не как HTML.
  function h(tag, attrs, ...children) {
    const el = document.createElement(tag);
    for (const [k, v] of Object.entries(attrs || {})) {
      if (v === undefined || v === null || v === false) continue;
      if (k === 'class') el.className = v;
      else if (k.startsWith('on')) el.addEventListener(k.slice(2), v);
      else if (k === 'value') el.value = v;
      else if (k === 'checked') el.checked = !!v;
      else el.setAttribute(k, v === true ? '' : v);
    }
    for (const c of children.flat()) {
      if (c === null || c === undefined || c === false) continue;
      el.append(c instanceof Node ? c : document.createTextNode(String(c)));
    }
    return el;
  }
  const ui = {
    section: (title) => h('div', { class: 'ds-section-title' }, title),
    card: (...c) => h('div', { class: 'ds-card ds-stack' }, ...c),
    hint: (text, extra) => h('div', { class: 'ds-hint' + (extra ? ' ' + extra : '') }, text),
    btn: (label, variant, extra) => h('button', { class: 'ds-btn' + (variant ? ' ds-btn--' + variant : '') + (extra ? ' ' + extra : '') }, label),
    switchEl: (input) => h('label', { class: 'ds-switch' }, input, h('span', { class: 'ds-switch__track' })),
    toggleRow: (title, sub, input, extra) => {
      input.setAttribute('aria-label', title); // имя переключателя для экранного диктора (009)
      return h('div', { class: 'ds-row ds-row--between' + (extra ? ' ' + extra : '') },
        h('div', {}, h('div', { class: 'ds-text-strong' }, title), sub ? ui.hint(sub) : null), ui.switchEl(input));
    },
    empty: (icon, title, text, action) => h('div', { class: 'ds-card ds-empty' },
      h('div', { class: 'ds-empty__icon' }, icon), h('h3', { class: 'ds-empty__title' }, title),
      h('div', { class: 'ds-text-muted' }, text), action || null),
    chip: (text, kind) => h('span', { class: 'ds-chip' + (kind ? ' ds-chip--' + kind : '') }, text),
    textarea: (attrs) => autosize(h('textarea', { class: 'ds-input', ...attrs })),
    date: (value) => closeOnReset(h('input', { class: 'ds-input', type: 'date', value })),
  };

  // Поле даты (011): «Сбросить» в системном календаре очищает поле и закрывает календарь (iOS сам его не закрывает).
  // Календарь закрывается вместе с фокусом поля. Очистка с клавиатуры (ПК: Backspace по частям даты) фокус не снимает.
  function closeOnReset(el) {
    let keyAt = 0;
    el.addEventListener('keydown', () => { keyAt = Date.now(); });
    const onClear = () => {
      if (el.value || Date.now() - keyAt < 1000 || document.activeElement !== el) return;
      setTimeout(() => el.blur(), 0);
    };
    el.addEventListener('input', onClear);
    el.addEventListener('change', onClear);
    return el;
  }

  async function api(action, body) {
    let res;
    try {
      res = await fetch('/api/' + action, {
        method: 'POST',
        headers: { 'content-type': 'application/json', 'x-init-data': initData },
        body: JSON.stringify(body || {}),
      });
    } catch (e) {
      throw new Error('Нет связи с сервером. Проверь интернет.');
    }
    const data = await res.json().catch(() => ({ ok: false, error: 'Ошибка сервера' }));
    if (!data.ok) throw new Error(data.error || 'Ошибка');
    return data;
  }

  let toastTimer;
  function toast(text) {
    const t = document.getElementById('toast');
    t.textContent = text;
    t.classList.add('ds-toast--visible');
    clearTimeout(toastTimer);
    toastTimer = setTimeout(() => t.classList.remove('ds-toast--visible'), 2600);
  }
  function haptic(kind) {
    try { tg && tg.HapticFeedback && tg.HapticFeedback.notificationOccurred(kind); } catch (e) {}
  }
  function confirmBox(text) {
    return new Promise((resolve) => {
      if (tg && ver('6.2')) {
        try { tg.showConfirm(text.slice(0, 250), (ok) => resolve(!!ok)); return; } catch (e) {}
      }
      resolve(window.confirm(text));
    });
  }
  async function busy(btn, fn) {
    if (btn) btn.disabled = true;
    try { return await fn(); }
    catch (e) { haptic('error'); toast('⚠️ ' + e.message); }
    finally { if (btn) btn.disabled = false; }
  }
  // Переключатель с немедленным сохранением (009): пока идёт запрос — недоступен; при ошибке (или если fn вернула
  // false — например, отказ в подтверждении) возвращается в прежнее положение.
  function bindToggle(input, fn) {
    input.addEventListener('change', async () => {
      const on = input.checked;
      input.disabled = true;
      try {
        if ((await fn(on)) === false) input.checked = !on;
      } catch (e) {
        input.checked = !on;
        haptic('error');
        toast('⚠️ ' + e.message);
      } finally {
        input.disabled = false;
      }
    });
    return input;
  }
  const MONTHS = ['января', 'февраля', 'марта', 'апреля', 'мая', 'июня', 'июля', 'августа', 'сентября', 'октября', 'ноября', 'декабря'];
  function fmtDate(iso) {
    const m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(iso || '');
    return m ? `${+m[3]} ${MONTHS[+m[2] - 1]}` : '';
  }
  function fmtTime(ms) {
    if (!ms) return '';
    const d = new Date(ms);
    const now = new Date();
    const hm = d.toLocaleTimeString('ru-RU', { hour: '2-digit', minute: '2-digit' });
    if (d.toDateString() === now.toDateString()) return 'сегодня в ' + hm;
    const y = new Date(now); y.setDate(now.getDate() - 1);
    if (d.toDateString() === y.toDateString()) return 'вчера в ' + hm;
    return `${d.getDate()} ${MONTHS[d.getMonth()]} в ${hm}`;
  }
  function autosize(el) {
    const fit = () => { el.style.height = 'auto'; el.style.height = el.scrollHeight + 2 + 'px'; };
    el.addEventListener('input', fit);
    requestAnimationFrame(fit);
    return el;
  }
  function plural(n, one, few, many) {
    const m10 = n % 10, m100 = n % 100;
    if (m10 === 1 && m100 !== 11) return one;
    if (m10 >= 2 && m10 <= 4 && (m100 < 10 || m100 >= 20)) return few;
    return many;
  }

  // ---------- сравнение версий письма (по словам) ----------
  function diffTokens(a, b) {
    let start = 0;
    while (start < a.length && start < b.length && a[start] === b[start]) start++;
    let endA = a.length, endB = b.length;
    while (endA > start && endB > start && a[endA - 1] === b[endB - 1]) { endA--; endB--; }
    const A = a.slice(start, endA), B = b.slice(start, endB);
    const n = A.length, m = B.length;
    if (n * m > 4e6) return null;
    const W = m + 1;
    const dp = new Uint16Array((n + 1) * W);
    for (let i = n - 1; i >= 0; i--)
      for (let j = m - 1; j >= 0; j--)
        dp[i * W + j] = A[i] === B[j] ? dp[(i + 1) * W + j + 1] + 1 : Math.max(dp[(i + 1) * W + j], dp[i * W + j + 1]);
    const ops = [];
    const push = (t, s) => { const last = ops[ops.length - 1]; if (last && last.t === t) last.s += s; else ops.push({ t, s }); };
    if (start) push('=', a.slice(0, start).join(''));
    let i = 0, j = 0;
    while (i < n && j < m) {
      if (A[i] === B[j]) { push('=', A[i]); i++; j++; }
      else if (dp[(i + 1) * W + j] >= dp[i * W + j + 1]) { push('-', A[i]); i++; }
      else { push('+', B[j]); j++; }
    }
    while (i < n) push('-', A[i++]);
    while (j < m) push('+', B[j++]);
    if (endA < a.length) push('=', a.slice(endA).join(''));
    return ops;
  }
  function renderDiff(oldText, newText) {
    const tok = (s) => s.match(/\s+|[^\s]+/g) || [];
    const ops = diffTokens(tok(oldText), tok(newText));
    const box = h('div', { class: 'ds-letter' });
    if (!ops) { box.textContent = newText; return box; }
    for (const op of ops) {
      if (op.t === '=') { box.append(op.s); continue; }
      // Пробелы и переносы по краям — вне подсветки; удалённые пробелы не показываем
      const [, lead, core, trail] = /^(\s*)([\s\S]*?)(\s*)$/.exec(op.s);
      if (op.t === '+') box.append(lead);
      if (core) box.append(h(op.t === '+' ? 'mark' : 'del', { class: op.t === '+' ? 'ds-diff-add' : 'ds-diff-del' }, core));
      if (op.t === '+') box.append(trail);
    }
    return box;
  }
